-- ==========================================
-- 10-1. notifications 테이블 (시스템 알림 이력 관리)
-- ==========================================
CREATE TABLE notifications (
    notification_id BIGSERIAL PRIMARY KEY,                    -- 알림 고유 ID
    account_id BIGINT,                                        -- 알림을 받을 특정 사용자 ID (계정별 알림인 경우)
    company_id BIGINT,                                        -- 알림을 받을 특정 기업 ID (기업 단위 알림인 경우)
    title VARCHAR(150) NOT NULL,                              -- 알림 제목
    message TEXT NOT NULL,                                    -- 알림 상세 내용
    notification_type VARCHAR(50) NOT NULL,                   -- 알림 유형 (예: CORPORATE_APPROVAL, OVERLOAD_WARNING, NOTICE 등)
    reference_id BIGINT,                                      -- 연관된 엔티티 ID (예: 게시글 ID, 과적검사 ID 등 선택적 활용)
    is_read BOOLEAN NOT NULL DEFAULT FALSE,                   -- 읽음 여부 (FALSE: 안 읽음, TRUE: 읽음)
    read_at TIMESTAMP WITH TIME ZONE,                         -- 읽은 일시
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP, -- 알림 생성일시

    CONSTRAINT fk_notifications_account FOREIGN KEY (account_id)
        REFERENCES accounts(account_id) ON DELETE CASCADE,
    CONSTRAINT fk_notifications_company FOREIGN KEY (company_id)
        REFERENCES companies(company_id) ON DELETE CASCADE,
    CONSTRAINT chk_notification_type CHECK (notification_type IN ('CORPORATE_APPROVAL', 'OVERLOAD_WARNING', 'NOTICE', 'SYSTEM'))
);

-- ==========================================
-- 10-2. notifications 인덱스 (성능 최적화)
-- ==========================================
-- 특정 사용자의 안 읽은 알림 목록 조회 등을 빠르게 처리하기 위한 인덱스
CREATE INDEX idx_notifications_account_read ON notifications(account_id, is_read, created_at DESC);
CREATE INDEX idx_notifications_company_read ON notifications(company_id, is_read, created_at DESC);
