-- ==========================================
-- scargo DB 통합 스키마 및 더미 데이터 최종본 (게이트 마스터 반영)
-- ==========================================

-- ==========================================
-- 0. 기존 테이블 삭제 (외래키 제약조건 고려 역순)
-- ==========================================
DROP TABLE IF EXISTS notifications CASCADE;
DROP TABLE IF EXISTS Vehicle_Checklist CASCADE;
DROP TABLE IF EXISTS vehicle_daily_logs CASCADE;
DROP TABLE IF EXISTS tire_management CASCADE;
DROP TABLE IF EXISTS gate_logs CASCADE;
DROP TABLE IF EXISTS gates CASCADE; -- 추가된 게이트 마스터 테이블 삭제 순서 포함
DROP TABLE IF EXISTS overload_checks CASCADE;
DROP TABLE IF EXISTS loading_records CASCADE;
DROP TABLE IF EXISTS containers CASCADE;
DROP TABLE IF EXISTS trucks CASCADE;
DROP TABLE IF EXISTS loading_locations CASCADE;
DROP TABLE IF EXISTS yards CASCADE;
DROP TABLE IF EXISTS attachments CASCADE;
DROP TABLE IF EXISTS post_comments CASCADE;
DROP TABLE IF EXISTS posts CASCADE;
DROP TABLE IF EXISTS accounts CASCADE;
DROP TABLE IF EXISTS companies CASCADE;

-- ==========================================
-- 1. companies 테이블 (업체 정보)
-- ==========================================
CREATE TABLE companies (
    company_id BIGSERIAL PRIMARY KEY,
    company_name VARCHAR(100) NOT NULL,
    address TEXT NOT NULL,
    business_no VARCHAR(20) UNIQUE,
    industry_type VARCHAR(50),
    representative_name VARCHAR(50),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT uk_company_name_address UNIQUE (company_name, address),
    CONSTRAINT chk_companies_biz_no_format CHECK (business_no ~ '^[0-9]{3}-[0-9]{2}-[0-9]{5}$')
);

-- ==========================================
-- 2. accounts 테이블 (사용자 계정 정보)
-- ==========================================
CREATE TABLE accounts (
    account_id BIGSERIAL PRIMARY KEY,
    user_name VARCHAR(30),
    user_id VARCHAR(50) NOT NULL UNIQUE,
    user_pw VARCHAR(255) NOT NULL,
    user_type VARCHAR(30) NOT NULL DEFAULT 'GENERAL',
    company_id BIGINT,
    phone_num VARCHAR(20),
    business_no VARCHAR(20),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_accounts_company FOREIGN KEY (company_id)
        REFERENCES companies(company_id) ON DELETE SET NULL,
    CONSTRAINT chk_user_type CHECK (user_type IN ('GENERAL', 'CORPORATE_PENDING', 'CORPORATE_APPROVED', 'ADMIN'))
);

-- ==========================================
-- 2-1. posts 테이블 (게시글)
-- ==========================================
CREATE TABLE posts (
    post_id BIGSERIAL PRIMARY KEY,
    account_id BIGINT,
    title VARCHAR(200) NOT NULL,
    content_text TEXT NOT NULL,
    category VARCHAR(50) NOT NULL DEFAULT 'FREE',
    view_count INT NOT NULL DEFAULT 0,
    is_pinned BOOLEAN NOT NULL DEFAULT FALSE,
    is_deleted BOOLEAN NOT NULL DEFAULT FALSE,
    deleted_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_posts_account FOREIGN KEY (account_id)
        REFERENCES accounts(account_id) ON DELETE SET NULL,
    CONSTRAINT chk_post_category CHECK (category IN ('NOTICE', 'FREE', 'QNA', 'FAQ'))
);

-- ==========================================
-- 2-2. post_comments 테이블 (댓글 & 대댓글)
-- ==========================================
CREATE TABLE post_comments (
    comment_id BIGSERIAL PRIMARY KEY,
    post_id BIGINT NOT NULL,
    account_id BIGINT,
    parent_id BIGINT,
    content_text TEXT NOT NULL,
    is_deleted BOOLEAN NOT NULL DEFAULT FALSE,
    deleted_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_comments_post FOREIGN KEY (post_id)
        REFERENCES posts(post_id) ON DELETE CASCADE,
    CONSTRAINT fk_comments_account FOREIGN KEY (account_id)
        REFERENCES accounts(account_id) ON DELETE SET NULL,
    CONSTRAINT fk_comments_parent FOREIGN KEY (parent_id)
        REFERENCES post_comments(comment_id) ON DELETE CASCADE
);

-- ==========================================
-- 2-3. attachments 테이블 (첨부파일)
-- ==========================================
CREATE TABLE attachments (
    attachment_id BIGSERIAL PRIMARY KEY,
    post_id BIGINT NOT NULL,
    original_name VARCHAR(255) NOT NULL,
    stored_name VARCHAR(255) NOT NULL,
    file_path VARCHAR(500) NOT NULL,
    file_size BIGINT NOT NULL,
    file_type VARCHAR(100),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_attachments_post FOREIGN KEY (post_id)
        REFERENCES posts(post_id) ON DELETE CASCADE
);

-- ==========================================
-- 3-1. yards 테이블 (상위 야드 관리)
-- ==========================================
CREATE TABLE yards (
    yard_id BIGSERIAL PRIMARY KEY,
    yard_name VARCHAR(50) NOT NULL,
    yard_type VARCHAR(30) NOT NULL DEFAULT 'GENERAL',
    latitude NUMERIC(9, 6),
    longitude NUMERIC(9, 6),
    status VARCHAR(20) DEFAULT 'AVAILABLE',
    is_available BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- ==========================================
-- 3-2. loading_locations 테이블 (야드 하위 세부 위치)
-- ==========================================
CREATE TABLE loading_locations (
    location_id BIGSERIAL PRIMARY KEY,
    yard_id BIGINT NOT NULL,
    sector VARCHAR(50) NOT NULL,
    latitude NUMERIC(9, 6),
    longitude NUMERIC(9, 6),
    status VARCHAR(20) DEFAULT 'AVAILABLE',
    is_available BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_locations_yard FOREIGN KEY (yard_id)
        REFERENCES yards(yard_id) ON DELETE CASCADE
);

-- ==========================================
-- 4. trucks 테이블 (물류 화물차 정보 관리)
-- ==========================================
CREATE TABLE trucks (
    vehicle_no VARCHAR(20) PRIMARY KEY,
    company_id BIGINT NOT NULL,
    is_semi_trailer BOOLEAN NOT NULL DEFAULT FALSE,
    trailer_no VARCHAR(20),
    truck_type VARCHAR(30),
    max_load_weight NUMERIC(8, 2),
    planned_route JSONB,
    status VARCHAR(20) DEFAULT 'OUTSIDE',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_trucks_company FOREIGN KEY (company_id)
        REFERENCES companies(company_id),
    CONSTRAINT chk_truck_status CHECK (status IN ('OUTSIDE', 'INSIDE', 'IN_TRANSIT'))
);

-- ==========================================
-- 5. containers 테이블 (컨테이너 정보)
-- ==========================================
CREATE TABLE containers (
    container_no VARCHAR(11) PRIMARY KEY,
    company_id BIGINT,
    iso_size_type_code VARCHAR(4) NOT NULL,
    container_type VARCHAR(20) NOT NULL,
    is_high_cube BOOLEAN NOT NULL DEFAULT FALSE,
    max_gross_kg NUMERIC(8, 1) NOT NULL,
    tare_kg NUMERIC(8, 1) NOT NULL,
    net_kg NUMERIC(8, 1) NOT NULL,
    cubic_capacity_cbm NUMERIC(6, 2) NOT NULL,
    csc_approval_no VARCHAR(30),
    reserved_cargo_info JSONB,
    loading_location_id BIGINT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_containers_company FOREIGN KEY (company_id)
        REFERENCES companies(company_id) ON DELETE SET NULL,
    CONSTRAINT fk_containers_location FOREIGN KEY (loading_location_id)
        REFERENCES loading_locations(location_id) ON DELETE SET NULL,
    CONSTRAINT chk_containers_no_format CHECK (container_no ~ '^[A-Z]{4}[0-9]{7}$'),
    CONSTRAINT chk_containers_weight CHECK (max_gross_kg >= tare_kg + net_kg)
);

-- ==========================================
-- 6. loading_records 테이블 (컨테이너 적재 이력)
-- ==========================================
CREATE TABLE loading_records (
    record_id BIGSERIAL PRIMARY KEY,
    vehicle_no VARCHAR(20) NOT NULL,
    container_no VARCHAR(11) NOT NULL,
    location_id BIGINT NOT NULL,
    -- 첫 게이트 OCR 인식 전 상하차 작업 대기 상태
    status VARCHAR(20) NOT NULL DEFAULT 'PENDING',
    loaded_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_records_truck FOREIGN KEY (vehicle_no)
        REFERENCES trucks(vehicle_no),
    CONSTRAINT fk_records_container FOREIGN KEY (container_no)
        REFERENCES containers(container_no),
    CONSTRAINT fk_records_location FOREIGN KEY (location_id)
        REFERENCES loading_locations(location_id),
    CONSTRAINT chk_loading_status CHECK (status IN ('PENDING', 'IN_PROGRESS', 'COMPLETED', 'CANCELED'))
);

-- ==========================================
-- 7. overload_checks 테이블 (과적/VGM 검사 결과)
-- ==========================================
CREATE TABLE overload_checks (
    check_id BIGSERIAL PRIMARY KEY,
    vehicle_no VARCHAR(20) NOT NULL,
    container_no VARCHAR(11),
    
    usage_purpose VARCHAR(50),
    empty_vehicle_weight INT,
    total_weight INT,
    max_payload INT,
    vgm_weight INT,
    
    tire_count INT,
    vehicle_axle_count INT,
    axle_count INT,
    
    axle1_weight INT DEFAULT 0, axle2_weight INT DEFAULT 0,
    axle3_weight INT DEFAULT 0, axle4_weight INT DEFAULT 0,
    axle5_weight INT DEFAULT 0, axle6_weight INT DEFAULT 0,
    axle7_weight INT DEFAULT 0, axle8_weight INT DEFAULT 0,
    
    axle1_wheel_left INT DEFAULT 0, axle1_wheel_right INT DEFAULT 0,
    axle2_wheel_left INT DEFAULT 0, axle2_wheel_right INT DEFAULT 0,
    axle3_wheel_left INT DEFAULT 0, axle3_wheel_right INT DEFAULT 0,
    axle4_wheel_left INT DEFAULT 0, axle4_wheel_right INT DEFAULT 0,
    axle5_wheel_left INT DEFAULT 0, axle5_wheel_right INT DEFAULT 0,
    axle6_wheel_left INT DEFAULT 0, axle6_wheel_right INT DEFAULT 0,
    axle7_wheel_left INT DEFAULT 0, axle7_wheel_right INT DEFAULT 0,
    axle8_wheel_left INT DEFAULT 0, axle8_wheel_right INT DEFAULT 0,

    is_violation BOOLEAN NOT NULL,
    violation_reason TEXT,
    retry_count INT NOT NULL DEFAULT 0,
    is_passed BOOLEAN NOT NULL,
    checked_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_checks_truck FOREIGN KEY (vehicle_no)
        REFERENCES trucks(vehicle_no),
    CONSTRAINT fk_checks_container FOREIGN KEY (container_no)
        REFERENCES containers(container_no)
);

-- ==========================================
-- 8-1. gates 테이블 (게이트 마스터 정보 관리) [신규 추가]
-- ==========================================
CREATE TABLE gates (
    gate_id SERIAL PRIMARY KEY,
    gate_code VARCHAR(50) NOT NULL UNIQUE,          -- 게이트 관리용 코드 (예: 'GATE_IN_01')
    gate_name VARCHAR(100) NOT NULL,                 -- 게이트 이름 (예: '제1 물류문(정문)')
    gate_type VARCHAR(10) NOT NULL,                  -- 게이트 유형 ('IN': 입문, 'OUT': 출문)
    latitude NUMERIC(10, 7),                         -- 위도 (GPS 좌표)
    longitude NUMERIC(10, 7),                        -- 경도 (GPS 좌표)
    location_description TEXT,                       -- 상세 위치 설명
    is_active BOOLEAN NOT NULL DEFAULT TRUE,         -- 사용 여부 (활성/비활성)
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT chk_master_gate_type CHECK (gate_type IN ('IN', 'OUT', 'BOTH'))
);

-- ==========================================
-- 8-2. gate_logs 테이블 (OCR 게이트 통과 이력) [수정 반영]
-- ==========================================
CREATE TABLE gate_logs (
    gate_log_id BIGSERIAL PRIMARY KEY,
    gate_id INT NOT NULL,                            -- gates 테이블 참조 외래 키
    recognized_plate_no VARCHAR(20),
    recognized_trailer_no VARCHAR(20),
    actual_vehicle_no VARCHAR(20),
    plate_confidence NUMERIC(5, 2),
    recognition_status VARCHAR(20) NOT NULL DEFAULT 'SUCCESS',
    front_image_url TEXT,
    rear_image_url TEXT,
    ocr_raw_data JSONB,
    vehicle_type VARCHAR(30) DEFAULT 'UNKNOWN',
    pass_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_gate_logs_gate FOREIGN KEY (gate_id)
        REFERENCES gates(gate_id) ON DELETE RESTRICT,
    CONSTRAINT fk_gate_logs_truck FOREIGN KEY (actual_vehicle_no)
        REFERENCES trucks(vehicle_no) ON DELETE SET NULL,
    CONSTRAINT chk_recognition_status CHECK (recognition_status IN ('SUCCESS', 'FAILED'))
);

-- ==========================================
-- 9. tire_management 테이블 (타이어 관리)
-- ==========================================
CREATE TABLE tire_management (
    tire_id BIGSERIAL PRIMARY KEY,
    vehicle_no VARCHAR(20) NOT NULL,
    tire_position VARCHAR(50) NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',
    installation_date TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    installation_mileage INT NOT NULL,
    memo TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_tire_truck FOREIGN KEY (vehicle_no)
        REFERENCES trucks(vehicle_no) ON DELETE CASCADE,
    CONSTRAINT chk_tire_status CHECK (status IN ('ACTIVE', 'REPLACED', 'DISCARDED'))
);

-- ==========================================
-- 10. vehicle_daily_logs 테이블 (일일 운행 기록)
-- ==========================================
CREATE TABLE vehicle_daily_logs (
    log_id BIGSERIAL PRIMARY KEY,
    vehicle_no VARCHAR(20) NOT NULL,
    driving_date DATE NOT NULL,
    daily_distance INT NOT NULL,
    accumulated_mileage INT NOT NULL,
    memo VARCHAR(255),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_log_truck FOREIGN KEY (vehicle_no)
        REFERENCES trucks(vehicle_no) ON DELETE CASCADE
);

-- ==========================================
-- 11. Vehicle_Checklist 테이블 (운수종사자 일상점검표 - 법정 서식 반영)
-- ==========================================
CREATE TABLE Vehicle_Checklist (
    inspection_id BIGSERIAL PRIMARY KEY,                     
    vehicle_no VARCHAR(20) NOT NULL,                         
    inspector_account_id BIGINT,                             
    inspection_date DATE NOT NULL DEFAULT CURRENT_DATE,        
    
    light_status VARCHAR(20) NOT NULL DEFAULT 'PASS',          
    brake_status VARCHAR(20) NOT NULL DEFAULT 'PASS',          
    air_brake_status VARCHAR(20) NOT NULL DEFAULT 'PASS',      
    tire_status VARCHAR(20) NOT NULL DEFAULT 'PASS',           
    steering_status VARCHAR(20) NOT NULL DEFAULT 'PASS',       
    cargo_securement_status VARCHAR(20) NOT NULL DEFAULT 'PASS', 
    engine_oil_status VARCHAR(20) NOT NULL DEFAULT 'PASS',     
    seatbelt_status VARCHAR(20) NOT NULL DEFAULT 'PASS',       
    fire_extinguisher_status VARCHAR(20) NOT NULL DEFAULT 'PASS', 
    safety_triangle_status VARCHAR(20) NOT NULL DEFAULT 'PASS',    
    
    overall_status VARCHAR(20) NOT NULL DEFAULT 'PASS',        
    memo TEXT,                                                 
    
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_inspection_truck FOREIGN KEY (vehicle_no)
        REFERENCES trucks(vehicle_no) ON DELETE CASCADE,
    CONSTRAINT fk_inspection_account FOREIGN KEY (inspector_account_id)
        REFERENCES accounts(account_id) ON DELETE SET NULL,
        
    CONSTRAINT chk_light_chk_status CHECK (light_status IN ('PASS', 'FAIL', 'NA')),
    CONSTRAINT chk_brake_chk_status CHECK (brake_status IN ('PASS', 'FAIL', 'NA')),
    CONSTRAINT chk_air_brake_status CHECK (air_brake_status IN ('PASS', 'FAIL', 'NA')),
    CONSTRAINT chk_tire_chk_status CHECK (tire_status IN ('PASS', 'FAIL', 'NA')),
    CONSTRAINT chk_steering_status CHECK (steering_status IN ('PASS', 'FAIL', 'NA')),
    CONSTRAINT chk_cargo_chk_status CHECK (cargo_securement_status IN ('PASS', 'FAIL', 'NA')),
    CONSTRAINT chk_oil_chk_status CHECK (engine_oil_status IN ('PASS', 'FAIL', 'NA')),
    CONSTRAINT chk_seatbelt_status CHECK (seatbelt_status IN ('PASS', 'FAIL', 'NA')),
    CONSTRAINT chk_fire_ext_status CHECK (fire_extinguisher_status IN ('PASS', 'FAIL', 'NA')),
    CONSTRAINT chk_safety_tri_status CHECK (safety_triangle_status IN ('PASS', 'FAIL', 'NA')),
    CONSTRAINT chk_overall_status CHECK (overall_status IN ('PASS', 'FAIL'))
);

-- ==========================================
-- 12. notifications 테이블 (시스템 알림 이력 관리)
-- ==========================================
CREATE TABLE notifications (
    notification_id BIGSERIAL PRIMARY KEY,                    
    account_id BIGINT,                                        
    company_id BIGINT,                                        
    title VARCHAR(150) NOT NULL,                              
    message TEXT NOT NULL,                                    
    notification_type VARCHAR(50) NOT NULL,                   
    reference_id BIGINT,                                      
    is_read BOOLEAN NOT NULL DEFAULT FALSE,                   
    read_at TIMESTAMP WITH TIME ZONE,                         
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP, 

    CONSTRAINT fk_notifications_account FOREIGN KEY (account_id)
        REFERENCES accounts(account_id) ON DELETE CASCADE,
    CONSTRAINT fk_notifications_company FOREIGN KEY (company_id)
        REFERENCES companies(company_id) ON DELETE CASCADE,
    CONSTRAINT chk_notification_type CHECK (notification_type IN ('CORPORATE_APPROVAL', 'OVERLOAD_WARNING', 'NOTICE', 'SYSTEM'))
);

-- ==========================================
-- 13. 성능 향상을 위한 인덱스 정의
-- ==========================================
CREATE INDEX idx_accounts_company_id ON accounts(company_id);
CREATE INDEX idx_locations_yard_id ON loading_locations(yard_id);
CREATE INDEX idx_trucks_company_id ON trucks(company_id);
CREATE INDEX idx_containers_company_id ON containers(company_id);
CREATE INDEX idx_containers_location_id ON containers(loading_location_id);
CREATE INDEX idx_loading_records_vehicle ON loading_records(vehicle_no);
CREATE INDEX idx_loading_records_container ON loading_records(container_no);
CREATE INDEX idx_loading_records_location ON loading_records(location_id);
CREATE INDEX idx_loading_records_status ON loading_records(status);
CREATE INDEX idx_loading_records_container_status ON loading_records(container_no, status);

CREATE INDEX idx_overload_checks_vehicle ON overload_checks(vehicle_no);
CREATE INDEX idx_overload_checks_container ON overload_checks(container_no);

-- 게이트 관련 인덱스 수정/추가
CREATE INDEX idx_gate_logs_gate_id ON gate_logs(gate_id);
CREATE INDEX idx_gate_logs_actual_vehicle ON gate_logs(actual_vehicle_no);
CREATE INDEX idx_gate_logs_status ON gate_logs(recognition_status);
CREATE INDEX idx_gate_logs_vehicle_pass ON gate_logs(actual_vehicle_no, pass_at DESC);

CREATE INDEX idx_tire_management_vehicle ON tire_management(vehicle_no);
CREATE INDEX idx_tire_management_status ON tire_management(status);

CREATE INDEX idx_vehicle_daily_logs_vehicle ON vehicle_daily_logs(vehicle_no);
CREATE INDEX idx_vehicle_daily_logs_date ON vehicle_daily_logs(vehicle_no, driving_date DESC);

CREATE INDEX idx_vehicle_checklist_vehicle ON Vehicle_Checklist(vehicle_no);
CREATE INDEX idx_vehicle_checklist_date ON Vehicle_Checklist(vehicle_no, inspection_date DESC);
CREATE INDEX idx_vehicle_checklist_account ON Vehicle_Checklist(inspector_account_id);

CREATE INDEX idx_notifications_account_read ON notifications(account_id, is_read, created_at DESC);
CREATE INDEX idx_notifications_company_read ON notifications(company_id, is_read, created_at DESC);

CREATE INDEX idx_posts_category_created ON posts(category, created_at DESC) WHERE is_deleted = FALSE;
CREATE INDEX idx_posts_account_id ON posts(account_id);
CREATE INDEX idx_posts_search ON posts USING gin (to_tsvector('simple', coalesce(title, '') || ' ' || coalesce(content_text, '')));
CREATE INDEX idx_comments_post_id ON post_comments(post_id) WHERE is_deleted = FALSE;
CREATE INDEX idx_attachments_post_id ON attachments(post_id);
