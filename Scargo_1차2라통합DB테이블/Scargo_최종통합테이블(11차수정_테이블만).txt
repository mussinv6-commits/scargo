-- ============================================================
-- 1. 회원 · 게시판 · 알림
-- ============================================================

-- ------------------------------------------------------------
-- companies : 업체 정보
-- ------------------------------------------------------------
CREATE TABLE companies (
    company_id          BIGSERIAL,
    company_name        VARCHAR(100) NOT NULL,
    address             TEXT NOT NULL,
    business_no         VARCHAR(20),
    industry_type       VARCHAR(50),
    representative_name VARCHAR(50),
    created_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT companies_pkey PRIMARY KEY (company_id),
    CONSTRAINT companies_business_no_key UNIQUE (business_no),
    CONSTRAINT uk_company_name_address UNIQUE (company_name, address),
    CONSTRAINT chk_companies_biz_no_format CHECK ((business_no ~ '^[0-9]{3}-[0-9]{2}-[0-9]{5}$'))
);

-- ------------------------------------------------------------
-- accounts : 회원 계정 (GENERAL 일반 / ADMIN 관리자)
-- ------------------------------------------------------------
CREATE TABLE accounts (
    account_id  BIGSERIAL,
    user_name   VARCHAR(30),
    user_id     VARCHAR(50) NOT NULL,                    -- 로그인 ID (문자열, 중복 불가)
    user_pw     VARCHAR(255) NOT NULL,                   -- 비밀번호 해시 (bcrypt 등, 평문 저장 금지)
    user_type   VARCHAR(30) NOT NULL DEFAULT 'GENERAL',  -- GENERAL 일반 / ADMIN 관리자
    company_id  BIGINT,                                  -- 소속 업체 (선택)
    phone_num   VARCHAR(20),
    business_no VARCHAR(20),
    created_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT accounts_pkey PRIMARY KEY (account_id),
    CONSTRAINT accounts_user_id_key UNIQUE (user_id),
    CONSTRAINT chk_user_type CHECK ((user_type IN ('GENERAL', 'ADMIN'))),
    CONSTRAINT fk_accounts_company FOREIGN KEY (company_id) REFERENCES companies(company_id) ON DELETE SET NULL
);
CREATE INDEX idx_accounts_company_id ON accounts (company_id);

-- ------------------------------------------------------------
-- app_tokens : 앱 로그인 토큰
-- ------------------------------------------------------------
CREATE TABLE app_tokens (
    token      VARCHAR(64) NOT NULL,                   -- 앱 로그인 토큰
    account_id BIGINT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    expires_at TIMESTAMPTZ NOT NULL,                   -- 만료 시각

    CONSTRAINT app_tokens_pkey PRIMARY KEY (token),
    CONSTRAINT fk_app_tokens_account FOREIGN KEY (account_id) REFERENCES accounts(account_id) ON DELETE CASCADE
);
CREATE INDEX idx_app_tokens_account ON app_tokens (account_id);

-- ------------------------------------------------------------
-- posts : 게시글
-- ------------------------------------------------------------
CREATE TABLE posts (
    post_id      BIGSERIAL,
    account_id   BIGINT,
    title        VARCHAR(200) NOT NULL,
    content_text TEXT NOT NULL,
    category     VARCHAR(50) NOT NULL DEFAULT 'FREE',
    view_count   INTEGER NOT NULL DEFAULT 0,
    is_pinned    BOOLEAN NOT NULL DEFAULT FALSE,
    is_deleted   BOOLEAN NOT NULL DEFAULT FALSE,
    deleted_at   TIMESTAMPTZ,
    created_at   TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at   TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT posts_pkey PRIMARY KEY (post_id),
    CONSTRAINT chk_post_category CHECK ((category IN ('NOTICE', 'FREE', 'QNA', 'FAQ'))),
    CONSTRAINT fk_posts_account FOREIGN KEY (account_id) REFERENCES accounts(account_id) ON DELETE SET NULL
);
CREATE INDEX idx_posts_account_id ON posts (account_id);
CREATE INDEX idx_posts_category_created ON posts (category, created_at DESC) WHERE (is_deleted = false);
CREATE INDEX idx_posts_search ON posts USING gin (to_tsvector('simple'::regconfig, (((COALESCE(title, ''))::text || ' ') || COALESCE(content_text, ''))));

-- ------------------------------------------------------------
-- post_comments : 댓글 & 대댓글
-- ------------------------------------------------------------
CREATE TABLE post_comments (
    comment_id   BIGSERIAL,
    post_id      BIGINT NOT NULL,
    account_id   BIGINT,
    parent_id    BIGINT,
    content_text TEXT NOT NULL,
    is_deleted   BOOLEAN NOT NULL DEFAULT FALSE,
    deleted_at   TIMESTAMPTZ,
    created_at   TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at   TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT post_comments_pkey PRIMARY KEY (comment_id),
    CONSTRAINT uk_comments_id_post UNIQUE (comment_id, post_id),
    CONSTRAINT fk_comments_account FOREIGN KEY (account_id) REFERENCES accounts(account_id) ON DELETE SET NULL,
    CONSTRAINT fk_comments_parent_same_post FOREIGN KEY (parent_id, post_id) REFERENCES post_comments(comment_id, post_id) ON DELETE CASCADE,
    CONSTRAINT fk_comments_post FOREIGN KEY (post_id) REFERENCES posts(post_id) ON DELETE CASCADE
);
CREATE INDEX idx_comments_post_id ON post_comments (post_id) WHERE (is_deleted = false);

-- ------------------------------------------------------------
-- attachments : 첨부파일
-- ------------------------------------------------------------
CREATE TABLE attachments (
    attachment_id BIGSERIAL,
    post_id       BIGINT NOT NULL,
    original_name VARCHAR(255) NOT NULL,
    stored_name   VARCHAR(255) NOT NULL,
    file_path     VARCHAR(500) NOT NULL,
    file_size     BIGINT NOT NULL,
    file_type     VARCHAR(100),
    created_at    TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT attachments_pkey PRIMARY KEY (attachment_id),
    CONSTRAINT fk_attachments_post FOREIGN KEY (post_id) REFERENCES posts(post_id) ON DELETE CASCADE
);
CREATE INDEX idx_attachments_post_id ON attachments (post_id);

-- ------------------------------------------------------------
-- notifications : 계정·업체 알림 (과적 경고 / 공지 / 시스템)
-- ------------------------------------------------------------
CREATE TABLE notifications (
    notification_id   BIGSERIAL,
    account_id        BIGINT,                                 -- 수신 계정 (account_id·company_id 중 하나 이상 필수)
    company_id        BIGINT,                                 -- 수신 업체
    title             VARCHAR(150) NOT NULL,
    message           TEXT NOT NULL,
    notification_type VARCHAR(50) NOT NULL,
    reference_id      BIGINT,                                 -- 관련 대상 ID. FK 없음 - notification_type에 따라 백엔드에서 해석
    is_read           BOOLEAN NOT NULL DEFAULT FALSE,
    read_at           TIMESTAMPTZ,
    created_at        TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT notifications_pkey PRIMARY KEY (notification_id),
    CONSTRAINT chk_notification_target CHECK (((account_id IS NOT NULL) OR (company_id IS NOT NULL))),
    CONSTRAINT chk_notification_type CHECK ((notification_type IN ('OVERLOAD_WARNING', 'NOTICE', 'SYSTEM'))),
    CONSTRAINT fk_notifications_account FOREIGN KEY (account_id) REFERENCES accounts(account_id) ON DELETE CASCADE,
    CONSTRAINT fk_notifications_company FOREIGN KEY (company_id) REFERENCES companies(company_id) ON DELETE CASCADE
);
CREATE INDEX idx_notifications_account_read ON notifications (account_id, is_read, created_at DESC);
CREATE INDEX idx_notifications_company_read ON notifications (company_id, is_read, created_at DESC);

-- ============================================================
-- 2. 야드 · 차량 · 컨테이너
-- ============================================================

-- ------------------------------------------------------------
-- yards : 상위 야드 관리
-- ------------------------------------------------------------
CREATE TABLE yards (
    yard_id      BIGSERIAL,
    yard_name    VARCHAR(50) NOT NULL,
    yard_type    VARCHAR(30) NOT NULL DEFAULT 'GENERAL',
    latitude     NUMERIC(9,6),
    longitude    NUMERIC(9,6),
    status       VARCHAR(20) DEFAULT 'AVAILABLE',
    is_available BOOLEAN DEFAULT TRUE,
    created_at   TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT yards_pkey PRIMARY KEY (yard_id)
);

-- ------------------------------------------------------------
-- loading_locations : 야드 하위 세부 위치
-- ------------------------------------------------------------
CREATE TABLE loading_locations (
    location_id  BIGSERIAL,
    yard_id      BIGINT NOT NULL,
    sector       VARCHAR(50) NOT NULL,
    latitude     NUMERIC(9,6),
    longitude    NUMERIC(9,6),
    status       VARCHAR(20) DEFAULT 'AVAILABLE',
    is_available BOOLEAN DEFAULT TRUE,
    created_at   TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT loading_locations_pkey PRIMARY KEY (location_id),
    CONSTRAINT fk_locations_yard FOREIGN KEY (yard_id) REFERENCES yards(yard_id) ON DELETE CASCADE
);
CREATE INDEX idx_locations_yard_id ON loading_locations (yard_id);

-- ------------------------------------------------------------
-- trucks : 물류 화물차 정보 관리
-- ------------------------------------------------------------
CREATE TABLE trucks (
    vehicle_no          VARCHAR(20) NOT NULL,                    -- 차량번호 (PK)
    company_id          BIGINT NOT NULL,                         -- 소속 운송사
    is_semi_trailer     BOOLEAN NOT NULL DEFAULT FALSE,          -- 세미트레일러(트랙터+트레일러) 여부. CONTAINER_TRUCK이면 TRUE 필수
    trailer_no          VARCHAR(20),                             -- 트레일러 번호 (세미트레일러면 필수)
    truck_type          VARCHAR(30),
    max_load_ton        NUMERIC(8,2),                            -- 최대 적재량(톤). vehicle_specs.max_payload_kg는 kg 단위
    planned_route       JSONB,
    status              VARCHAR(20) DEFAULT 'OUTSIDE',           -- OUTSIDE 야드 밖 / INSIDE 야드 안 / IN_TRANSIT 운송 중
    created_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    vehicle_category    VARCHAR(30),                             -- 운송 구간용 차량 구분 (CONTAINER_TRUCK=항만→허브 / GENERAL_TRUCK=허브→센터)
    assigned_account_id BIGINT,                                  -- 배정 기사 계정
    entry_approval      VARCHAR(20) NOT NULL DEFAULT 'PENDING',  -- 출입 승인 상태 PENDING / APPROVED / REJECTED

    CONSTRAINT trucks_pkey PRIMARY KEY (vehicle_no),
    CONSTRAINT chk_truck_status CHECK ((status IN ('OUTSIDE', 'INSIDE', 'IN_TRANSIT'))),
    CONSTRAINT chk_trucks_container_is_semi CHECK (((vehicle_category IS DISTINCT FROM 'CONTAINER_TRUCK') OR is_semi_trailer)),
    CONSTRAINT chk_trucks_semi_trailer_no CHECK (((NOT is_semi_trailer) OR (trailer_no IS NOT NULL))),
    CONSTRAINT chk_trucks_vehicle_category CHECK (((vehicle_category IS NULL) OR (vehicle_category IN ('CONTAINER_TRUCK', 'GENERAL_TRUCK')))),
    CONSTRAINT trucks_entry_approval_check CHECK ((entry_approval IN ('PENDING', 'APPROVED', 'REJECTED'))),
    CONSTRAINT fk_trucks_company FOREIGN KEY (company_id) REFERENCES companies(company_id),
    CONSTRAINT trucks_assigned_account_id_fkey FOREIGN KEY (assigned_account_id) REFERENCES accounts(account_id) ON DELETE SET NULL
);
CREATE INDEX idx_trucks_company_id ON trucks (company_id);
COMMENT ON COLUMN trucks.max_load_ton IS '최대 적재량(톤). vehicle_specs.max_payload_kg는 kg 단위';
COMMENT ON COLUMN trucks.vehicle_category IS '운송 구간용 차량 구분 (CONTAINER_TRUCK=항만→허브 / GENERAL_TRUCK=허브→센터)';

-- ------------------------------------------------------------
-- vehicle_specs : 차량 제원 (자동차등록증 기준, trucks와 1:1)
-- ------------------------------------------------------------
CREATE TABLE vehicle_specs (
    vehicle_no            VARCHAR(20) NOT NULL,
    registration_category VARCHAR(20),                            -- 차량 등록 구분 (1. 승용 / 3. 화물 / 4. 특수)
    usage_purpose         VARCHAR(50),                            -- 목적 (일반형-카고, 견인형 등)
    empty_weight_kg       INTEGER,                                -- 공차중량
    gross_weight_kg       INTEGER,                                -- 총중량
    max_payload_kg        INTEGER,                                -- 최대적재량
    seat_count            INTEGER,
    tire_count            INTEGER,
    axle_count            INTEGER,                                -- 차축수
    chassis_height_m      NUMERIC(4,2),                           -- 적재면(섀시) 높이 → 적재 높이 계산용
    updated_at            TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT vehicle_specs_pkey PRIMARY KEY (vehicle_no),
    CONSTRAINT fk_vehicle_specs_truck FOREIGN KEY (vehicle_no) REFERENCES trucks(vehicle_no) ON DELETE CASCADE
);
COMMENT ON COLUMN vehicle_specs.registration_category IS '차량 등록 구분 (1. 승용 / 3. 화물 / 4. 특수)';

-- ------------------------------------------------------------
-- tire_management : 타이어 관리
-- ------------------------------------------------------------
CREATE TABLE tire_management (
    tire_id              BIGSERIAL,
    vehicle_no           VARCHAR(20) NOT NULL,
    tire_position        VARCHAR(50) NOT NULL,
    status               VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',
    installation_date    TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    installation_mileage INTEGER NOT NULL,
    memo                 TEXT,
    created_at           TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at           TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT tire_management_pkey PRIMARY KEY (tire_id),
    CONSTRAINT chk_tire_status CHECK ((status IN ('ACTIVE', 'REPLACED', 'DISCARDED'))),
    CONSTRAINT fk_tire_truck FOREIGN KEY (vehicle_no) REFERENCES trucks(vehicle_no) ON DELETE CASCADE
);
CREATE INDEX idx_tire_management_status ON tire_management (status);
CREATE INDEX idx_tire_management_vehicle ON tire_management (vehicle_no);

-- ------------------------------------------------------------
-- vehicle_daily_logs : 일일 운행 기록
-- ------------------------------------------------------------
CREATE TABLE vehicle_daily_logs (
    log_id              BIGSERIAL,
    vehicle_no          VARCHAR(20) NOT NULL,
    driving_date        DATE NOT NULL,
    daily_distance      INTEGER NOT NULL,
    accumulated_mileage INTEGER NOT NULL,
    memo                VARCHAR(255),
    created_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT vehicle_daily_logs_pkey PRIMARY KEY (log_id),
    CONSTRAINT uk_vehicle_daily_logs_day UNIQUE (vehicle_no, driving_date),
    CONSTRAINT fk_log_truck FOREIGN KEY (vehicle_no) REFERENCES trucks(vehicle_no) ON DELETE CASCADE
);
CREATE INDEX idx_vehicle_daily_logs_date ON vehicle_daily_logs (vehicle_no, driving_date DESC);
CREATE INDEX idx_vehicle_daily_logs_vehicle ON vehicle_daily_logs (vehicle_no);

-- ------------------------------------------------------------
-- vehicle_checklist : 운수종사자 일상점검표 - 법정 서식 반영
-- ------------------------------------------------------------
CREATE TABLE vehicle_checklist (
    inspection_id            BIGSERIAL,
    vehicle_no               VARCHAR(20) NOT NULL,
    inspector_account_id     BIGINT,
    inspection_date          DATE NOT NULL DEFAULT CURRENT_DATE,
    light_status             VARCHAR(20) NOT NULL DEFAULT 'PASS',
    brake_status             VARCHAR(20) NOT NULL DEFAULT 'PASS',
    air_brake_status         VARCHAR(20) NOT NULL DEFAULT 'PASS',
    tire_status              VARCHAR(20) NOT NULL DEFAULT 'PASS',
    steering_status          VARCHAR(20) NOT NULL DEFAULT 'PASS',
    cargo_securement_status  VARCHAR(20) NOT NULL DEFAULT 'PASS',
    engine_oil_status        VARCHAR(20) NOT NULL DEFAULT 'PASS',
    seatbelt_status          VARCHAR(20) NOT NULL DEFAULT 'PASS',
    fire_extinguisher_status VARCHAR(20) NOT NULL DEFAULT 'PASS',
    safety_triangle_status   VARCHAR(20) NOT NULL DEFAULT 'PASS',
    overall_status           VARCHAR(20) NOT NULL DEFAULT 'PASS',
    memo                     TEXT,
    created_at               TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at               TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT vehicle_checklist_pkey PRIMARY KEY (inspection_id),
    CONSTRAINT chk_air_brake_status CHECK ((air_brake_status IN ('PASS', 'FAIL', 'NA'))),
    CONSTRAINT chk_brake_chk_status CHECK ((brake_status IN ('PASS', 'FAIL', 'NA'))),
    CONSTRAINT chk_cargo_chk_status CHECK ((cargo_securement_status IN ('PASS', 'FAIL', 'NA'))),
    CONSTRAINT chk_fire_ext_status CHECK ((fire_extinguisher_status IN ('PASS', 'FAIL', 'NA'))),
    CONSTRAINT chk_light_chk_status CHECK ((light_status IN ('PASS', 'FAIL', 'NA'))),
    CONSTRAINT chk_oil_chk_status CHECK ((engine_oil_status IN ('PASS', 'FAIL', 'NA'))),
    CONSTRAINT chk_overall_status CHECK ((overall_status IN ('PASS', 'FAIL'))),
    CONSTRAINT chk_safety_tri_status CHECK ((safety_triangle_status IN ('PASS', 'FAIL', 'NA'))),
    CONSTRAINT chk_seatbelt_status CHECK ((seatbelt_status IN ('PASS', 'FAIL', 'NA'))),
    CONSTRAINT chk_steering_status CHECK ((steering_status IN ('PASS', 'FAIL', 'NA'))),
    CONSTRAINT chk_tire_chk_status CHECK ((tire_status IN ('PASS', 'FAIL', 'NA'))),
    CONSTRAINT fk_inspection_account FOREIGN KEY (inspector_account_id) REFERENCES accounts(account_id) ON DELETE SET NULL,
    CONSTRAINT fk_inspection_truck FOREIGN KEY (vehicle_no) REFERENCES trucks(vehicle_no) ON DELETE CASCADE
);
CREATE INDEX idx_vehicle_checklist_account ON vehicle_checklist (inspector_account_id);
CREATE INDEX idx_vehicle_checklist_date ON vehicle_checklist (vehicle_no, inspection_date DESC);
CREATE INDEX idx_vehicle_checklist_vehicle ON vehicle_checklist (vehicle_no);

-- ------------------------------------------------------------
-- containers : 컨테이너 정보
-- ------------------------------------------------------------
CREATE TABLE containers (
    container_no        VARCHAR(11) NOT NULL,
    company_id          BIGINT,
    iso_size_type_code  VARCHAR(4) NOT NULL,                    -- ISO 6346 크기·타입 코드 (예: 22G1, 45G1)
    container_type      VARCHAR(20) NOT NULL,                   -- 일반 / 냉동
    is_high_cube        BOOLEAN NOT NULL DEFAULT FALSE,
    max_gross_kg        NUMERIC(8,1) NOT NULL,
    tare_kg             NUMERIC(8,1) NOT NULL,
    max_payload_kg      NUMERIC(8,1) NOT NULL,                  -- 컨테이너 최대 화물 적재 가능 중량(kg) = max_gross_kg - tare_kg. 실제 적재 화물 중량이 아님
    cubic_capacity_cbm  NUMERIC(6,2) NOT NULL,
    csc_approval_no     VARCHAR(30),
    reserved_cargo_info JSONB,
    loading_location_id BIGINT,                                 -- 컨테이너 현재 야드 위치의 기준 값. loading_records.location_id는 적재 작업 시점의 기록
    created_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT containers_pkey PRIMARY KEY (container_no),
    CONSTRAINT chk_containers_no_format CHECK ((container_no ~ '^[A-Z]{4}[0-9]{7}$')),
    CONSTRAINT chk_containers_payload CHECK ((max_payload_kg = (max_gross_kg - tare_kg))),
    CONSTRAINT chk_containers_type CHECK ((container_type IN ('일반', '냉동'))),
    CONSTRAINT fk_containers_company FOREIGN KEY (company_id) REFERENCES companies(company_id) ON DELETE SET NULL,
    CONSTRAINT fk_containers_location FOREIGN KEY (loading_location_id) REFERENCES loading_locations(location_id) ON DELETE SET NULL
);
CREATE INDEX idx_containers_company_id ON containers (company_id);
CREATE INDEX idx_containers_location_id ON containers (loading_location_id);
COMMENT ON COLUMN containers.max_payload_kg IS '컨테이너 최대 화물 적재 가능 중량(kg) = max_gross_kg - tare_kg. 실제 적재 화물 중량이 아님';
COMMENT ON COLUMN containers.loading_location_id IS '컨테이너 현재 야드 위치의 기준 값. loading_records.location_id는 적재 작업 시점의 기록';

-- ============================================================
-- 3. 물류 거점 · 운송 건 · 물품
-- ============================================================

-- ------------------------------------------------------------
-- delivery_regions : 출고 권역: 인천 · 서울 · 경기
-- ------------------------------------------------------------
CREATE TABLE delivery_regions (
    region_id   BIGSERIAL,                              -- 권역 고유 ID
    region_code VARCHAR(10) NOT NULL,                   -- 권역 코드 (ICN, SEL, GGI)
    region_name VARCHAR(30) NOT NULL,                   -- 권역 이름 (인천, 서울, 경기)
    color_hex   VARCHAR(7),                             -- 3D/화면 표시 색
    created_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT delivery_regions_pkey PRIMARY KEY (region_id),
    CONSTRAINT delivery_regions_region_code_key UNIQUE (region_code)
);

-- ------------------------------------------------------------
-- logistics_centers : 항만·HUB·물류센터 등 물류 거점의 기본 정보와 위치 관리
-- ------------------------------------------------------------
CREATE TABLE logistics_centers (
    center_id   BIGSERIAL,                                       -- 물류 거점 고유 ID (PK)
    center_code VARCHAR(50) NOT NULL,                            -- 거점 식별 코드 (중복 불가)
    center_name VARCHAR(100) NOT NULL,                           -- 거점 이름
    center_type VARCHAR(20) NOT NULL,                            -- 거점 종류: PORT=항만, HUB=허브, DISTRIBUTION_CENTER=물류센터
    address     TEXT,                                            -- 거점 주소
    latitude    NUMERIC(10,7),                                   -- 거점 위도
    longitude   NUMERIC(10,7),                                   -- 거점 경도
    is_active   BOOLEAN NOT NULL DEFAULT TRUE,                   -- 거점 운영 여부
    created_at  TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 거점 등록 일시
    region_id   BIGINT,                                          -- 물류센터 담당 출고 권역 (DISTRIBUTION_CENTER만, 권역당 1곳)

    CONSTRAINT logistics_centers_pkey PRIMARY KEY (center_id),
    CONSTRAINT logistics_centers_center_code_key UNIQUE (center_code),
    CONSTRAINT chk_center_region_scope CHECK (((region_id IS NULL) OR (center_type = 'DISTRIBUTION_CENTER'))),
    CONSTRAINT logistics_centers_center_type_check CHECK ((center_type IN ('PORT', 'HUB', 'DISTRIBUTION_CENTER'))),
    CONSTRAINT fk_logistics_centers_region FOREIGN KEY (region_id) REFERENCES delivery_regions(region_id)
);
CREATE INDEX idx_logistics_centers_region ON logistics_centers (region_id);
CREATE UNIQUE INDEX uk_center_region ON logistics_centers (region_id) WHERE ((center_type = 'DISTRIBUTION_CENTER') AND (region_id IS NOT NULL));

-- ------------------------------------------------------------
-- terminals : 항만 컨테이너 터미널 (앱 오더 출발지)
-- ------------------------------------------------------------
CREATE TABLE terminals (
    terminal_id      BIGSERIAL,
    terminal_code    VARCHAR(20) NOT NULL,                   -- 예: 터미널 약어 코드
    terminal_name    VARCHAR(100) NOT NULL,                  -- 터미널명
    port_name        VARCHAR(50) NOT NULL DEFAULT '인천항',     -- 항만명 (기본값 인천항)
    operator         VARCHAR(100),                           -- 운영사
    address          TEXT,
    latitude         NUMERIC(10,7),
    longitude        NUMERIC(10,7),
    coord_verified   BOOLEAN NOT NULL DEFAULT FALSE,         -- 좌표 확인 여부 (FALSE = 대략 좌표)
    gate_height_note TEXT,                                   -- 게이트 높이 확인 상태 메모 (값은 height_restrictions)
    is_active        BOOLEAN NOT NULL DEFAULT TRUE,
    created_at       TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    center_id        BIGINT,                                 -- 터미널이 속한 항만 거점 (PORT 거점만 허용)

    CONSTRAINT terminals_pkey PRIMARY KEY (terminal_id),
    CONSTRAINT terminals_terminal_code_key UNIQUE (terminal_code),
    CONSTRAINT terminals_center_id_fkey FOREIGN KEY (center_id) REFERENCES logistics_centers(center_id)
);
CREATE INDEX idx_terminals_center ON terminals (center_id);
COMMENT ON COLUMN terminals.port_name IS '항만명 (기본값 인천항)';

-- ------------------------------------------------------------
-- shipments : 항만부터 최종 물류센터까지 하나의 운송 건을 추적
-- ------------------------------------------------------------
CREATE TABLE shipments (
    shipment_id           BIGSERIAL,                                       -- 운송 건 고유 ID (PK)
    shipment_code         VARCHAR(60) NOT NULL,                            -- 운송 건 식별 코드
    container_no          VARCHAR(11),                                     -- 관련 컨테이너 번호 (FK, 선택)
    origin_center_id      BIGINT NOT NULL,                                 -- 최초 출발 거점 ID (FK)
    destination_center_id BIGINT,                                          -- 운송 건 단위 최종 목적지. 한 컨테이너에 목적지가 여러 곳이면 NULL이며, 물품별 목적지는 items.destination_center_id
    status                VARCHAR(30) NOT NULL DEFAULT 'READY',            -- 운송 상태: 준비/허브 이동/허브 도착/분류/센터 이동/부분 배송/완료/보류/취소
    started_at            TIMESTAMPTZ,                                     -- 운송 시작 일시
    completed_at          TIMESTAMPTZ,                                     -- 운송 완료 일시
    created_at            TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 등록 일시
    updated_at            TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 최종 수정 일시

    CONSTRAINT shipments_pkey PRIMARY KEY (shipment_id),
    CONSTRAINT shipments_shipment_code_key UNIQUE (shipment_code),
    CONSTRAINT uk_shipments_id_container UNIQUE (shipment_id, container_no),
    CONSTRAINT shipments_status_check CHECK ((status IN ('READY', 'TO_HUB', 'AT_HUB', 'SORTING', 'TO_CENTER', 'PARTIALLY_DELIVERED', 'DELIVERED', 'ON_HOLD', 'CANCELED'))),
    CONSTRAINT shipments_container_no_fkey FOREIGN KEY (container_no) REFERENCES containers(container_no) ON DELETE SET NULL,
    CONSTRAINT shipments_destination_center_id_fkey FOREIGN KEY (destination_center_id) REFERENCES logistics_centers(center_id),
    CONSTRAINT shipments_origin_center_id_fkey FOREIGN KEY (origin_center_id) REFERENCES logistics_centers(center_id)
);
CREATE INDEX idx_shipments_container ON shipments (container_no);
CREATE INDEX idx_shipments_status ON shipments (status);
COMMENT ON COLUMN shipments.destination_center_id IS '운송 건 단위 최종 목적지. 한 컨테이너에 목적지가 여러 곳이면 NULL이며, 물품별 목적지는 items.destination_center_id';

-- ------------------------------------------------------------
-- shipment_legs : 운송 건의 구간별 차량 배정, 출발·도착 일시 및 운송 상태 관리 (shipment_legs = 운송 구간)
-- ------------------------------------------------------------
CREATE TABLE shipment_legs (
    assignment_id  BIGSERIAL,                                       -- 운송 구간 고유 ID (PK)
    shipment_id    BIGINT,                                          -- 항만→허브는 원본 운송 건 필수. 허브→센터는 여러 원본 운송 건을 함께 운송하므로 NULL. 원본 연결은 물품/작업 이력에서 유지.
    vehicle_no     VARCHAR(20) NOT NULL,                            -- 배정 차량 (PORT_TO_HUB=CONTAINER_TRUCK, HUB_TO_CENTER=GENERAL_TRUCK). 진행 중 구간은 차량당 1건
    from_center_id BIGINT NOT NULL,                                 -- 해당 구간 출발 거점 ID (FK)
    to_center_id   BIGINT NOT NULL,                                 -- 해당 구간 도착 거점 ID (FK)
    leg_sequence   INTEGER NOT NULL DEFAULT 1,                      -- 운송 건 내 구간 순서(항만→허브 구간 기준). 허브→센터 구간은 shipment_id가 NULL이므로 순서 유일성 대상이 아님
    leg_type       VARCHAR(30) NOT NULL,                            -- PORT_TO_HUB 항만→허브 / HUB_TO_CENTER 허브→물류센터 / OTHER 기타
    status         VARCHAR(20) NOT NULL DEFAULT 'ASSIGNED',         -- ASSIGNED 배정 / IN_TRANSIT 운송 중 / ARRIVED 도착 / CANCELED 취소
    departed_at    TIMESTAMPTZ,                                     -- 구간 출발 일시
    arrived_at     TIMESTAMPTZ,                                     -- 구간 도착 일시
    created_at     TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 배정 등록 일시

    CONSTRAINT shipment_legs_pkey PRIMARY KEY (assignment_id),
    CONSTRAINT shipment_legs_assignment_id_shipment_id_key UNIQUE (assignment_id, shipment_id),
    CONSTRAINT chk_leg_shipment_scope CHECK ((((leg_type = 'PORT_TO_HUB') AND (shipment_id IS NOT NULL)) OR ((leg_type = 'HUB_TO_CENTER') AND (shipment_id IS NULL)) OR (leg_type = 'OTHER'))),
    CONSTRAINT shipment_legs_check CHECK ((from_center_id <> to_center_id)),
    CONSTRAINT shipment_legs_leg_sequence_check CHECK ((leg_sequence > 0)),
    CONSTRAINT shipment_legs_leg_type_check CHECK ((leg_type IN ('PORT_TO_HUB', 'HUB_TO_CENTER', 'OTHER'))),
    CONSTRAINT shipment_legs_status_check CHECK ((status IN ('ASSIGNED', 'IN_TRANSIT', 'ARRIVED', 'CANCELED'))),
    CONSTRAINT shipment_legs_from_center_id_fkey FOREIGN KEY (from_center_id) REFERENCES logistics_centers(center_id),
    CONSTRAINT shipment_legs_shipment_id_fkey FOREIGN KEY (shipment_id) REFERENCES shipments(shipment_id),
    CONSTRAINT shipment_legs_to_center_id_fkey FOREIGN KEY (to_center_id) REFERENCES logistics_centers(center_id),
    CONSTRAINT shipment_legs_vehicle_no_fkey FOREIGN KEY (vehicle_no) REFERENCES trucks(vehicle_no)
);
CREATE INDEX idx_assignments_shipment ON shipment_legs (shipment_id);
CREATE INDEX idx_assignments_vehicle ON shipment_legs (vehicle_no, status);
CREATE UNIQUE INDEX uk_legs_active_container_trip ON shipment_legs (shipment_id) WHERE ((leg_type = 'PORT_TO_HUB') AND (status IN ('ASSIGNED', 'IN_TRANSIT')));
CREATE UNIQUE INDEX uk_legs_active_vehicle ON shipment_legs (vehicle_no) WHERE (status IN ('ASSIGNED', 'IN_TRANSIT'));
CREATE UNIQUE INDEX uk_legs_shipment_seq ON shipment_legs (shipment_id, leg_sequence) WHERE (shipment_id IS NOT NULL);
COMMENT ON COLUMN shipment_legs.shipment_id IS '항만→허브는 원본 운송 건 필수. 허브→센터는 여러 원본 운송 건을 함께 운송하므로 NULL. 원본 연결은 물품/작업 이력에서 유지.';
COMMENT ON COLUMN shipment_legs.leg_sequence IS '운송 건 내 구간 순서(항만→허브 구간 기준). 허브→센터 구간은 shipment_id가 NULL이므로 순서 유일성 대상이 아님';

-- ------------------------------------------------------------
-- items : 컨테이너에 담긴 개별 물품의 바코드·목적지·배송 상태 관리
-- ------------------------------------------------------------
CREATE TABLE items (
    item_id               BIGSERIAL,                                       -- 물품 고유 ID (PK)
    tracking_no           VARCHAR(80) NOT NULL,                            -- 운송장 번호 (중복 불가)
    barcode               VARCHAR(100) NOT NULL,                           -- HUB 하역/분류/적재 시 스캔하는 현장용 물품 바코드 (중복 불가)
    shipment_id           BIGINT NOT NULL,                                 -- 원본 운송 건 ID (FK)
    container_no          VARCHAR(11),                                     -- 원본 컨테이너 번호(FK, 선택). 값이 있으면 원본 운송 건 컨테이너와 지연 FK로 일치 검사
    item_name             VARCHAR(150) NOT NULL,                           -- 물품명
    quantity              INTEGER NOT NULL DEFAULT 1,                      -- 물품 수량 (1 이상)
    weight_kg             NUMERIC(10,3) NOT NULL,                          -- 물품 중량 (kg)
    destination_center_id BIGINT NOT NULL,                                 -- 물품 최종 목적지 (DISTRIBUTION_CENTER 거점만 허용)
    status                VARCHAR(30) NOT NULL DEFAULT 'IN_CONTAINER',     -- 물품 현재 상태 (컨테이너/허브/분류/적재/운송/배송 등)
    delivered_at          TIMESTAMPTZ,                                     -- 배송 완료 일시
    created_at            TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 물품 등록 일시

    CONSTRAINT items_pkey PRIMARY KEY (item_id),
    CONSTRAINT items_barcode_key UNIQUE (barcode),
    CONSTRAINT items_item_id_shipment_id_key UNIQUE (item_id, shipment_id),
    CONSTRAINT items_tracking_no_key UNIQUE (tracking_no),
    CONSTRAINT items_quantity_check CHECK ((quantity > 0)),
    CONSTRAINT items_status_check CHECK ((status IN ('IN_CONTAINER', 'AT_HUB', 'SORTING', 'LOADED', 'IN_TRANSIT', 'DELIVERED', 'ON_HOLD', 'LOST', 'DAMAGED'))),
    CONSTRAINT items_weight_kg_check CHECK ((weight_kg >= 0)),
    CONSTRAINT fk_items_shipment_container FOREIGN KEY (shipment_id, container_no) REFERENCES shipments(shipment_id, container_no) DEFERRABLE INITIALLY DEFERRED,
    CONSTRAINT items_container_no_fkey FOREIGN KEY (container_no) REFERENCES containers(container_no) ON DELETE SET NULL,
    CONSTRAINT items_destination_center_id_fkey FOREIGN KEY (destination_center_id) REFERENCES logistics_centers(center_id),
    CONSTRAINT items_shipment_id_fkey FOREIGN KEY (shipment_id) REFERENCES shipments(shipment_id)
);
CREATE INDEX idx_items_destination ON items (destination_center_id, status);
CREATE INDEX idx_items_shipment ON items (shipment_id);

-- ------------------------------------------------------------
-- item_vehicle_assignments : 개별 물품과 운송 구간의 차량 배정을 연결해 환적·분할 배송 추적
-- ------------------------------------------------------------
CREATE TABLE item_vehicle_assignments (
    item_assignment_id BIGSERIAL,                                -- 물품-차량 배정 고유 ID (PK)
    item_id            BIGINT NOT NULL,                          -- 배정 물품 ID (복합 FK 구성)
    shipment_id        BIGINT NOT NULL,                          -- 물품의 원본 운송 건 ID. (item_id, shipment_id)로 소속 일치 확인
    assignment_id      BIGINT NOT NULL,                          -- 운송 구간 (FK). 항만→허브 구간이면 원본 운송 건 일치 검사, 허브→센터 구간이면 물품 목적지 일치 검사
    loaded_at          TIMESTAMPTZ,                              -- 차량에 물품 적재한 일시
    unloaded_at        TIMESTAMPTZ,                              -- 차량에서 물품 하역한 일시
    status             VARCHAR(20) NOT NULL DEFAULT 'ASSIGNED',  -- 물품 배정 상태: 배정/적재/하역/취소

    CONSTRAINT item_vehicle_assignments_pkey PRIMARY KEY (item_assignment_id),
    CONSTRAINT item_vehicle_assignments_item_id_assignment_id_key UNIQUE (item_id, assignment_id),
    CONSTRAINT item_vehicle_assignments_status_check CHECK ((status IN ('ASSIGNED', 'LOADED', 'UNLOADED', 'CANCELED'))),
    CONSTRAINT fk_item_assignment_leg FOREIGN KEY (assignment_id) REFERENCES shipment_legs(assignment_id),
    CONSTRAINT item_vehicle_assignments_item_id_shipment_id_fkey FOREIGN KEY (item_id, shipment_id) REFERENCES items(item_id, shipment_id)
);
CREATE INDEX idx_item_vehicle_assignment ON item_vehicle_assignments (assignment_id);

-- ------------------------------------------------------------
-- item_history : 컨테이너 하역(CONTAINER_UNLOAD)·물품 하역(UNLOAD)·분류(SORT)·적재(LOAD) 작업 이력
-- ------------------------------------------------------------
CREATE TABLE item_history (
    handling_id   BIGSERIAL,                                       -- 물품 작업 이력 고유 ID (PK)
    shipment_id   BIGINT NOT NULL,                                 -- 관련 운송 건 ID (FK)
    item_id       BIGINT,                                          -- 작업 대상 물품 ID (FK, 선택)
    center_id     BIGINT NOT NULL,                                 -- 작업 수행 물류 거점 ID (FK)
    assignment_id BIGINT,                                          -- 운송 구간 (FK, 선택). 항만→허브 구간이면 원본 운송 건 일치 검사
    handling_type VARCHAR(25) NOT NULL,                            -- 작업 유형: 컨테이너 하역/일반 하역/분류/적재
    status        VARCHAR(20) NOT NULL DEFAULT 'PENDING',          -- 작업 상태: 대기/진행/완료/취소
    started_at    TIMESTAMPTZ,                                     -- 작업 시작 일시
    completed_at  TIMESTAMPTZ,                                     -- 작업 완료 일시
    created_at    TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 작업 기록 생성 일시

    CONSTRAINT item_history_pkey PRIMARY KEY (handling_id),
    CONSTRAINT item_history_handling_type_check CHECK ((handling_type IN ('CONTAINER_UNLOAD', 'UNLOAD', 'SORT', 'LOAD'))),
    CONSTRAINT item_history_status_check CHECK ((status IN ('PENDING', 'IN_PROGRESS', 'COMPLETED', 'CANCELED'))),
    CONSTRAINT fk_item_history_leg FOREIGN KEY (assignment_id) REFERENCES shipment_legs(assignment_id),
    CONSTRAINT item_history_center_id_fkey FOREIGN KEY (center_id) REFERENCES logistics_centers(center_id),
    CONSTRAINT item_history_item_id_shipment_id_fkey FOREIGN KEY (item_id, shipment_id) REFERENCES items(item_id, shipment_id),
    CONSTRAINT item_history_shipment_id_fkey FOREIGN KEY (shipment_id) REFERENCES shipments(shipment_id)
);
CREATE INDEX idx_handling_shipment ON item_history (shipment_id, center_id);
CREATE INDEX idx_item_history_assignment ON item_history (assignment_id);
COMMENT ON TABLE item_history IS '물류 거점의 컨테이너 하역·물품 하역·분류·적재 작업 이력';

-- ------------------------------------------------------------
-- loading_records : 컨테이너 적재 이력
-- ------------------------------------------------------------
CREATE TABLE loading_records (
    record_id    BIGSERIAL,
    vehicle_no   VARCHAR(20) NOT NULL,                    -- 적재 차량 (세미트레일러 차량만 허용)
    container_no VARCHAR(11) NOT NULL,                    -- 적재 컨테이너 (진행 중 작업은 컨테이너당 1건)
    location_id  BIGINT NOT NULL,                         -- 적재 작업 위치
    status       VARCHAR(20) NOT NULL DEFAULT 'PENDING',  -- PENDING 대기 / IN_PROGRESS 작업 중 / COMPLETED 완료 / CANCELED 취소
    loaded_at    TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    shipment_id  BIGINT,                                  -- 항만 적재 기록에 연결된 운송 건 ID (FK)

    CONSTRAINT loading_records_pkey PRIMARY KEY (record_id),
    CONSTRAINT chk_loading_status CHECK ((status IN ('PENDING', 'IN_PROGRESS', 'COMPLETED', 'CANCELED'))),
    CONSTRAINT fk_records_container FOREIGN KEY (container_no) REFERENCES containers(container_no),
    CONSTRAINT fk_records_location FOREIGN KEY (location_id) REFERENCES loading_locations(location_id),
    CONSTRAINT fk_records_truck FOREIGN KEY (vehicle_no) REFERENCES trucks(vehicle_no),
    CONSTRAINT loading_records_shipment_id_fkey FOREIGN KEY (shipment_id) REFERENCES shipments(shipment_id)
);
CREATE INDEX idx_loading_records_container ON loading_records (container_no);
CREATE INDEX idx_loading_records_container_status ON loading_records (container_no, status);
CREATE INDEX idx_loading_records_location ON loading_records (location_id);
CREATE INDEX idx_loading_records_status ON loading_records (status);
CREATE INDEX idx_loading_records_vehicle ON loading_records (vehicle_no);
CREATE UNIQUE INDEX uk_loading_active_container ON loading_records (container_no) WHERE (status IN ('PENDING', 'IN_PROGRESS'));

-- ============================================================
-- 4. 게이트 · 과적검사 · 톨게이트
-- ============================================================

-- ------------------------------------------------------------
-- gates : 게이트 마스터 (항만 터미널·물류 거점 출입문)
-- ------------------------------------------------------------
CREATE TABLE gates (
    gate_id              SERIAL,
    gate_code            VARCHAR(50) NOT NULL,                   -- 게이트 관리용 코드 (예: 'GATE_IN_01')
    gate_name            VARCHAR(100) NOT NULL,                  -- 게이트 이름 (예: '제1 물류문(정문)')
    gate_type            VARCHAR(10) NOT NULL,                   -- IN 입문 / OUT 출문 / BOTH 출입 겸용
    latitude             NUMERIC(10,7),                          -- 위도 (GPS 좌표)
    longitude            NUMERIC(10,7),                          -- 경도 (GPS 좌표)
    location_description TEXT,                                   -- 상세 위치 설명
    is_active            BOOLEAN NOT NULL DEFAULT TRUE,          -- 사용 여부 (활성/비활성)
    created_at           TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    center_id            BIGINT,                                 -- 게이트가 속한 물류 거점 ID (FK)
    terminal_id          BIGINT,                                 -- 소속 항만 터미널 (선택)

    CONSTRAINT gates_pkey PRIMARY KEY (gate_id),
    CONSTRAINT gates_gate_code_key UNIQUE (gate_code),
    CONSTRAINT chk_master_gate_type CHECK ((gate_type IN ('IN', 'OUT', 'BOTH'))),
    CONSTRAINT gates_center_id_fkey FOREIGN KEY (center_id) REFERENCES logistics_centers(center_id),
    CONSTRAINT gates_terminal_id_fkey FOREIGN KEY (terminal_id) REFERENCES terminals(terminal_id) ON DELETE SET NULL
);
CREATE INDEX idx_gates_center ON gates (center_id);
COMMENT ON COLUMN gates.gate_type IS 'IN 입문 / OUT 출문 / BOTH 출입 겸용';

-- ------------------------------------------------------------
-- gate_logs : OCR 게이트 통과 이력
-- ------------------------------------------------------------
CREATE TABLE gate_logs (
    gate_log_id           BIGSERIAL,
    gate_id               INTEGER NOT NULL,                        -- gates 테이블 참조 외래 키
    recognized_plate_no   VARCHAR(20),
    recognized_trailer_no VARCHAR(20),
    actual_vehicle_no     VARCHAR(20),                             -- 확정 차량번호 (OCR 실패 시 NULL). 운송 구간이 있으면 구간 차량과 일치해야 함

-- ------------------------------------------------------------
-- overload_checks : 과적/VGM 검사 결과
-- ------------------------------------------------------------
CREATE TABLE overload_checks (
    check_id           BIGSERIAL,
    vehicle_no         VARCHAR(20) NOT NULL,                   -- 검사 차량. 운송 구간이 있으면 구간 차량과 일치해야 함
    container_no       VARCHAR(11),                            -- 검사 컨테이너 (선택). 운송 건이 있으면 운송 건 컨테이너와 일치해야 함
    usage_purpose      VARCHAR(50),                            -- 검사 당시 차량 용도 (제원 스냅샷)
    empty_weight_kg    INTEGER,                                -- 차량 공차중량(kg)
    total_weight_kg    INTEGER,                                -- 계근 총중량(kg) = 차량 공차 + 컨테이너(VGM). 축중 합계와 같아야 함
    max_payload_kg     INTEGER,                                -- 차량 최대적재량(kg)
    vgm_kg             INTEGER,                                -- 컨테이너 총중량(VGM, kg): 컨테이너 자체 중량 + 적재 화물
    tire_count         INTEGER,                                -- 검사 당시 타이어 수 (제원 스냅샷)
    vehicle_axle_count INTEGER,                                -- 차량 제원상(등록) 차축 수
    axle_count         INTEGER,                                -- 이번 검사에서 계측된 차축 수
    axle1_weight       INTEGER DEFAULT 0,                      -- 축별 중량(kg) = 좌·우 바퀴 중량 합. 1~8축 합계 = total_weight_kg
    axle2_weight       INTEGER DEFAULT 0,
    axle3_weight       INTEGER DEFAULT 0,
    axle4_weight       INTEGER DEFAULT 0,
    axle5_weight       INTEGER DEFAULT 0,
    axle6_weight       INTEGER DEFAULT 0,
    axle7_weight       INTEGER DEFAULT 0,
    axle8_weight       INTEGER DEFAULT 0,
    axle1_wheel_left   INTEGER DEFAULT 0,                      -- 축별 좌·우 바퀴 중량(kg)
    axle1_wheel_right  INTEGER DEFAULT 0,
    axle2_wheel_left   INTEGER DEFAULT 0,
    axle2_wheel_right  INTEGER DEFAULT 0,
    axle3_wheel_left   INTEGER DEFAULT 0,
    axle3_wheel_right  INTEGER DEFAULT 0,
    axle4_wheel_left   INTEGER DEFAULT 0,
    axle4_wheel_right  INTEGER DEFAULT 0,
    axle5_wheel_left   INTEGER DEFAULT 0,
    axle5_wheel_right  INTEGER DEFAULT 0,
    axle6_wheel_left   INTEGER DEFAULT 0,
    axle6_wheel_right  INTEGER DEFAULT 0,
    axle7_wheel_left   INTEGER DEFAULT 0,
    axle7_wheel_right  INTEGER DEFAULT 0,
    axle8_wheel_left   INTEGER DEFAULT 0,
    axle8_wheel_right  INTEGER DEFAULT 0,
    is_violation       BOOLEAN NOT NULL,                       -- 과적 위반 여부
    violation_reason   TEXT,
    retry_count        INTEGER NOT NULL DEFAULT 0,             -- 재검사 횟수
    is_passed          BOOLEAN NOT NULL,                       -- 최종 통과 여부
    checked_at         TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    shipment_id        BIGINT,                                 -- 검사 대상 운송 건 ID (FK)
    assignment_id      BIGINT,                                 -- 검사한 운송 구간 (FK, 선택). 항만→허브 구간이면 shipment_id 자동 채움·일치 검사
    station_id         BIGINT,                                 -- 검사를 수행한 과적 검사소 ID (FK)
    check_type         VARCHAR(20),                            -- 검사 구분: 고속도로 / 기타
    detection_status   VARCHAR(20),                            -- 감지 상태: 정상 / 의심 / 확정

    CONSTRAINT overload_checks_pkey PRIMARY KEY (check_id),
    CONSTRAINT chk_overload_axle_count CHECK ((((axle_count IS NULL) OR ((axle_count >= 0) AND (axle_count <= 8))) AND ((vehicle_axle_count IS NULL) OR ((vehicle_axle_count >= 1) AND (vehicle_axle_count <= 8))))),
    CONSTRAINT chk_overload_axle_sum CHECK (((total_weight_kg IS NULL) OR (total_weight_kg = (((((((axle1_weight + axle2_weight) + axle3_weight) + axle4_weight) + axle5_weight) + axle6_weight) + axle7_weight) + axle8_weight)))),
    CONSTRAINT chk_overload_check_type CHECK (((check_type IS NULL) OR (check_type IN ('HIGHWAY', 'OTHER')))),
    CONSTRAINT chk_overload_detection_status CHECK (((detection_status IS NULL) OR (detection_status IN ('NORMAL', 'SUSPECTED', 'CONFIRMED')))),
    CONSTRAINT chk_overload_wheel_sum CHECK (((axle1_weight = (axle1_wheel_left + axle1_wheel_right)) AND (axle2_weight = (axle2_wheel_left + axle2_wheel_right)) AND (axle3_weight = (axle3_wheel_left + axle3_wheel_right)) AND (axle4_weight = (axle4_wheel_left + axle4_wheel_right)) AND (axle5_weight = (axle5_wheel_left + axle5_wheel_right)) AND (axle6_weight = (axle6_wheel_left + axle6_wheel_right)) AND (axle7_weight = (axle7_wheel_left + axle7_wheel_right)) AND (axle8_weight = (axle8_wheel_left + axle8_wheel_right)))),
    CONSTRAINT fk_checks_container FOREIGN KEY (container_no) REFERENCES containers(container_no),
    CONSTRAINT fk_checks_truck FOREIGN KEY (vehicle_no) REFERENCES trucks(vehicle_no),
    CONSTRAINT fk_overload_leg FOREIGN KEY (assignment_id) REFERENCES shipment_legs(assignment_id),
    CONSTRAINT fk_overload_shipment FOREIGN KEY (shipment_id) REFERENCES shipments(shipment_id),
    CONSTRAINT fk_overload_station FOREIGN KEY (station_id) REFERENCES weigh_stations(station_id)
);
CREATE INDEX idx_overload_assignment_time ON overload_checks (assignment_id, checked_at DESC);
CREATE INDEX idx_overload_checks_container ON overload_checks (container_no);
CREATE INDEX idx_overload_checks_vehicle ON overload_checks (vehicle_no);
CREATE INDEX idx_overload_station ON overload_checks (station_id);
COMMENT ON COLUMN overload_checks.empty_weight_kg IS '차량 공차중량(kg)';
COMMENT ON COLUMN overload_checks.total_weight_kg IS '계근 총중량(kg) = 차량 공차 + 컨테이너(VGM). 축중 합계와 같아야 함';
COMMENT ON COLUMN overload_checks.max_payload_kg IS '차량 최대적재량(kg)';
COMMENT ON COLUMN overload_checks.vgm_kg IS '컨테이너 총중량(VGM, kg): 컨테이너 자체 중량 + 적재 화물';
COMMENT ON COLUMN overload_checks.vehicle_axle_count IS '차량 제원상(등록) 차축 수';
COMMENT ON COLUMN overload_checks.axle_count IS '이번 검사에서 계측된 차축 수';

-- ------------------------------------------------------------
-- toll_passages : OCR 결과, 등록 여부, 과적검사 및 통과 결과 관리
-- ------------------------------------------------------------
CREATE TABLE toll_passages (
    passage_id          BIGSERIAL,                                       -- 통과 시도 고유 ID
    tollgate_id         BIGINT NOT NULL,                                 -- 대상 톨게이트
    recognized_plate_no VARCHAR(20),                                     -- OCR 인식 차량번호
    vehicle_no          VARCHAR(20),                                     -- 등록 차량번호 (미등록 시 NULL). 운송 구간이 있으면 구간 차량과 일치해야 함
    assignment_id       BIGINT,                                          -- 운송 구간
    check_id            BIGINT,                                          -- 과적검사 결과 (미검사 시 NULL)
    recognition_status  VARCHAR(20) NOT NULL,                            -- OCR 인식 결과 SUCCESS / FAILED
    passage_status      VARCHAR(20) NOT NULL DEFAULT 'PENDING',          -- PENDING 판정 대기 / PASSED 통과 / BLOCKED 차단
    block_reason        VARCHAR(30),                                     -- 차단 사유 OCR_FAILED / UNREGISTERED / OVERLOAD / OTHER (BLOCKED일 때만)
    detected_at         TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 차량 감지 시각
    passed_at           TIMESTAMPTZ,                                     -- 실제 통과 시각 (차단 시 NULL)
    created_at          TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT toll_passages_pkey PRIMARY KEY (passage_id),
    CONSTRAINT chk_toll_passage_block_reason CHECK ((((block_reason IS DISTINCT FROM 'OCR_FAILED') OR (recognition_status = 'FAILED')) AND ((block_reason IS DISTINCT FROM 'UNREGISTERED') OR (vehicle_no IS NULL)) AND ((block_reason IS DISTINCT FROM 'OVERLOAD') OR (check_id IS NOT NULL)))),
    CONSTRAINT chk_toll_passage_passed_valid CHECK (((passage_status <> 'PASSED') OR ((recognition_status = 'SUCCESS') AND (vehicle_no IS NOT NULL)))),
    CONSTRAINT chk_toll_passage_result CHECK ((((passage_status = 'PASSED') AND (passed_at IS NOT NULL) AND (block_reason IS NULL)) OR ((passage_status = 'BLOCKED') AND (passed_at IS NULL) AND (block_reason IS NOT NULL)) OR ((passage_status = 'PENDING') AND (passed_at IS NULL) AND (block_reason IS NULL)))),
    CONSTRAINT toll_passages_block_reason_check CHECK ((block_reason IN ('OCR_FAILED', 'UNREGISTERED', 'OVERLOAD', 'OTHER'))),
    CONSTRAINT toll_passages_passage_status_check CHECK ((passage_status IN ('PENDING', 'PASSED', 'BLOCKED'))),
    CONSTRAINT toll_passages_recognition_status_check CHECK ((recognition_status IN ('SUCCESS', 'FAILED'))),
    CONSTRAINT toll_passages_assignment_id_fkey FOREIGN KEY (assignment_id) REFERENCES shipment_legs(assignment_id),
    CONSTRAINT toll_passages_check_id_fkey FOREIGN KEY (check_id) REFERENCES overload_checks(check_id),
    CONSTRAINT toll_passages_tollgate_id_fkey FOREIGN KEY (tollgate_id) REFERENCES tollgates(tollgate_id),
    CONSTRAINT toll_passages_vehicle_no_fkey FOREIGN KEY (vehicle_no) REFERENCES trucks(vehicle_no)
);
CREATE INDEX idx_toll_passages_tollgate_time ON toll_passages (tollgate_id, detected_at DESC);
CREATE INDEX idx_toll_passages_vehicle_time ON toll_passages (vehicle_no, detected_at DESC);

    passed_at        TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 통과 시각

    CONSTRAINT toll_payments_payment_method_check CHECK ((payment_method IN ('HI_PASS', 'CARD', 'CASH', 'OTHER'))),
    CONSTRAINT toll_payments_payment_status_check CHECK ((payment_status IN ('PENDING', 'PAID', 'FAILED', 'CANCELED', 'REFUNDED'))),
    CONSTRAINT toll_payments_toll_system_check CHECK ((toll_system IN ('OPEN', 'CLOSED'))),
    CONSTRAINT toll_payments_assignment_id_fkey FOREIGN KEY (assignment_id) REFERENCES shipment_legs(assignment_id),
    CONSTRAINT toll_payments_entry_passage_id_fkey FOREIGN KEY (entry_passage_id) REFERENCES toll_passages(passage_id),
    CONSTRAINT toll_payments_passage_id_fkey FOREIGN KEY (passage_id) REFERENCES toll_passages(passage_id),

    CONSTRAINT uk_zones_hub_tag UNIQUE (hub_id, zone_tag),
    CONSTRAINT uk_zones_id_hub UNIQUE (zone_id, hub_id),
    CONSTRAINT chk_zones_rack_row CHECK (((rack_row >= 0) AND (rack_row <= 3))),
    CONSTRAINT fk_zones_hub FOREIGN KEY (hub_id) REFERENCES hubs(hub_id) ON DELETE CASCADE,
    CONSTRAINT fk_zones_region FOREIGN KEY (region_id) REFERENCES delivery_regions(region_id)
);

-- ------------------------------------------------------------
-- hub_rack_slots : 랙 칸: 0단 = AMR 바닥 칸, 1~3단 = 무인 지게차
-- ------------------------------------------------------------
CREATE TABLE hub_rack_slots (
    slot_id    BIGSERIAL,                              -- 랙 칸 고유 ID
    hub_id     BIGINT NOT NULL,                        -- 복합 FK로 구역 및 박스와 허브 일치 보장
    zone_id    BIGINT NOT NULL,                        -- 소속 구역
    side_no    INTEGER NOT NULL,                       -- 랙 면 (0: 북쪽, 1: 남쪽)
    slot_index INTEGER NOT NULL,                       -- 칸 번호 (0~8)
    level_no   INTEGER NOT NULL DEFAULT 0,             -- 단 (0~3)
    slot_label VARCHAR(20) NOT NULL,                   -- 표시용 라벨 (예: A-03-0면-0단)
    status     VARCHAR(20) NOT NULL DEFAULT 'EMPTY',   -- 칸 상태
    pos_x      NUMERIC(7,2),                           -- 평면도 X (1000 x 640 기준)
    pos_y      NUMERIC(7,2),                           -- 평면도 Y
    updated_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT hub_rack_slots_pkey PRIMARY KEY (slot_id),
    CONSTRAINT uk_slots_id_hub UNIQUE (slot_id, hub_id),
    CONSTRAINT uk_slots_position UNIQUE (zone_id, side_no, slot_index, level_no),

    CONSTRAINT chk_slots_level CHECK (((level_no >= 0) AND (level_no <= 3))),
    CONSTRAINT chk_slots_side CHECK ((side_no IN (0, 1))),
    CONSTRAINT chk_slots_status CHECK ((status IN ('EMPTY', 'OCCUPIED', 'RESERVED_IN', 'RESERVED_OUT', 'DISABLED'))),
    CONSTRAINT fk_slots_zone FOREIGN KEY (zone_id, hub_id) REFERENCES hub_zones(zone_id, hub_id) ON DELETE CASCADE
);
CREATE INDEX idx_slots_zone_status ON hub_rack_slots (zone_id, status);
COMMENT ON COLUMN hub_rack_slots.slot_label IS '표시용 라벨 (예: A-03-0면-0단)';

-- ------------------------------------------------------------
-- parcels : 물품별 허브 박스 정보; 운송장 번호는 items에서 관리
-- ------------------------------------------------------------
CREATE TABLE parcels (
    parcel_no   VARCHAR(20) NOT NULL,                     -- 내부 박스 ID (운송장은 items.tracking_no)
    item_id     BIGINT NOT NULL,                          -- 박스에 물품 연결 필수. 물품당 박스 정보는 최대 1개
    hub_id      BIGINT NOT NULL,                          -- 현재 허브
    receipt_id  BIGINT,                                   -- 입고 접수 (어느 화물차로 왔는지)
    region_id   BIGINT,                                   -- 목적지 권역 (OCR 결과)
    category    VARCHAR(30),                              -- 품목 (전자제품, 식품, 의류 ...)
    size_class  CHAR(1),                                  -- 크기 S / M / L
    handling    VARCHAR(20) NOT NULL DEFAULT '일반',        -- 취급 표시
    status      VARCHAR(20) NOT NULL DEFAULT 'IN_TRUCK',  -- 화물 상태
    slot_id     BIGINT,                                   -- 현재 보관 랙 칸
    dispatch_id BIGINT,                                   -- 출고된 택배차
    inbound_at  TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    stored_at   TIMESTAMPTZ,                              -- 랙 보관 시각
    outbound_at TIMESTAMPTZ,                              -- 출고 시각

    CONSTRAINT parcels_pkey PRIMARY KEY (parcel_no),
    CONSTRAINT parcels_item_id_key UNIQUE (item_id),
    CONSTRAINT chk_parcels_handling CHECK ((handling IN ('일반', '파손주의', '취급주의', '냉장', '중량물'))),
    CONSTRAINT chk_parcels_size CHECK ((size_class IN ('S', 'M', 'L'))),
    CONSTRAINT chk_parcels_status CHECK ((status IN ('IN_TRUCK', 'MOVING', 'STORED', 'OUTBOUND_READY', 'SHIPPED', 'SORT_FAILED'))),
    CONSTRAINT fk_parcels_dispatch FOREIGN KEY (dispatch_id, hub_id) REFERENCES hub_dispatches(dispatch_id, hub_id),
    CONSTRAINT fk_parcels_hub FOREIGN KEY (hub_id) REFERENCES hubs(hub_id),
    CONSTRAINT fk_parcels_item FOREIGN KEY (item_id) REFERENCES items(item_id),
    CONSTRAINT fk_parcels_receipt FOREIGN KEY (receipt_id, hub_id) REFERENCES hub_inbound_receipts(receipt_id, hub_id),
    CONSTRAINT fk_parcels_region FOREIGN KEY (region_id) REFERENCES delivery_regions(region_id),
    CONSTRAINT fk_parcels_slot FOREIGN KEY (slot_id, hub_id) REFERENCES hub_rack_slots(slot_id, hub_id)
);
CREATE INDEX idx_parcels_receipt ON parcels (receipt_id);
CREATE INDEX idx_parcels_region ON parcels (region_id);
CREATE INDEX idx_parcels_slot_hub ON parcels (slot_id, hub_id);
CREATE INDEX idx_parcels_status ON parcels (hub_id, status);

    CONSTRAINT amr_robots_robot_code_key UNIQUE (robot_code),

    overload_check_id   BIGINT,                                  -- 출발 전 과적 자가점검 결과 (overload_checks)

    vgm_kg              INTEGER,                                 -- 실제 컨테이너 총중량(빈 컨테이너 자체 중량 + 적재 화물 중량), 신고값
    gross_weight_kg     INTEGER,                                 -- 차량 공차중량 + 컨테이너 VGM. 차량 조합/섀시 포함 여부는 vehicle_specs의 공차중량 정의에 따름
    precheck            JSONB,                                   -- 출발 전 점검표 결과
    planned_route       JSONB,                                   -- 외부 경로 API 응답(좌표열, 경유 영업소)

    assigned_at         TIMESTAMPTZ,                             -- 배차
    loaded_at           TIMESTAMPTZ,                             -- 적재 등록
    gate_out_at         TIMESTAMPTZ,                             -- 게이트아웃
    departed_at         TIMESTAMPTZ,                             -- 운행 시작
    arrived_at          TIMESTAMPTZ,                             -- HUB 도착
    handed_over_at      TIMESTAMPTZ,                             -- 인계 완료
    created_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at          TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    shipment_id         BIGINT,                                  -- 원본 운송 건 (배차 이후 필수)
    assignment_id       BIGINT,                                  -- 항만→허브 운송 구간 (배차 이후 필수, 차량·컨테이너·거점 일치 검사)

    CONSTRAINT transport_orders_pkey PRIMARY KEY (order_id),
    CONSTRAINT transport_orders_gate_out_log_id_key UNIQUE (gate_out_log_id),
    CONSTRAINT transport_orders_order_no_key UNIQUE (order_no),
    CONSTRAINT chk_order_assignment_required CHECK (((status IN ('CREATED', 'CANCELED')) OR ((vehicle_no IS NOT NULL) AND (shipment_id IS NOT NULL) AND (assignment_id IS NOT NULL)))),
    CONSTRAINT chk_order_status CHECK ((status IN ('CREATED', 'ASSIGNED', 'LOADED', 'GATE_OUT', 'IN_TRANSIT', 'ARRIVED', 'HANDED_OVER', 'CANCELED'))),
    CONSTRAINT fk_order_company FOREIGN KEY (company_id) REFERENCES companies(company_id),
    CONSTRAINT fk_order_container FOREIGN KEY (container_no) REFERENCES containers(container_no),
    CONSTRAINT fk_order_dest FOREIGN KEY (dest_hub_id) REFERENCES hubs(hub_id),
    CONSTRAINT fk_order_driver FOREIGN KEY (driver_account_id) REFERENCES accounts(account_id) ON DELETE SET NULL,
    CONSTRAINT fk_order_gate_out FOREIGN KEY (gate_out_log_id) REFERENCES gate_logs(gate_log_id) ON DELETE SET NULL,
    CONSTRAINT fk_order_origin FOREIGN KEY (origin_terminal_id) REFERENCES terminals(terminal_id),
    CONSTRAINT fk_order_overload FOREIGN KEY (overload_check_id) REFERENCES overload_checks(check_id) ON DELETE SET NULL,
    CONSTRAINT fk_order_truck FOREIGN KEY (vehicle_no) REFERENCES trucks(vehicle_no) ON DELETE RESTRICT,
    CONSTRAINT fk_orders_leg_same_shipment FOREIGN KEY (assignment_id, shipment_id) REFERENCES shipment_legs(assignment_id, shipment_id),
    CONSTRAINT transport_orders_shipment_id_fkey FOREIGN KEY (shipment_id) REFERENCES shipments(shipment_id)
);
CREATE INDEX idx_order_driver ON transport_orders (driver_account_id, status);
CREATE INDEX idx_order_shipment ON transport_orders (shipment_id);
CREATE INDEX idx_order_vehicle_status ON transport_orders (vehicle_no, status);
CREATE UNIQUE INDEX uk_order_active_container ON transport_orders (container_no) WHERE (status NOT IN ('HANDED_OVER', 'CANCELED'));
CREATE UNIQUE INDEX uk_orders_active_vehicle ON transport_orders (vehicle_no) WHERE ((vehicle_no IS NOT NULL) AND (status NOT IN ('HANDED_OVER', 'CANCELED')));
CREATE UNIQUE INDEX uk_orders_leg ON transport_orders (assignment_id) WHERE (assignment_id IS NOT NULL);
CREATE UNIQUE INDEX uk_transport_orders_id_shipment ON transport_orders (order_id, shipment_id);
COMMENT ON COLUMN transport_orders.vgm_kg IS '실제 컨테이너 총중량(빈 컨테이너 자체 중량 + 적재 화물 중량), 신고값';
COMMENT ON COLUMN transport_orders.gross_weight_kg IS '차량 공차중량 + 컨테이너 VGM. 차량 조합/섀시 포함 여부는 vehicle_specs의 공차중량 정의에 따름';

-- ------------------------------------------------------------
-- container_seals : 원본 운송 건별 컨테이너 씰 장착/종료 및 교체 연결. 같은 컨테이너의 다른 운송 건 이력과 구분

CREATE TABLE container_seals (

    container_no           VARCHAR(11) NOT NULL,
    seal_no                VARCHAR(30) NOT NULL,
    seal_position          VARCHAR(30) NOT NULL DEFAULT 'MAIN',             -- 봉인 위치 코드. 기본 MAIN. 다른 위치/별도 봉인은 별도 코드 사용
    seal_type              VARCHAR(20) NOT NULL DEFAULT 'CARRIER',
    status                 VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',           -- 물리적 사용 상태 ACTIVE/REMOVED/REPLACED/LOST. 훼손은 이벤트 observed_condition으로 별도 기록
    attached_at            TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    attached_by_account_id BIGINT,
    attached_by_name       VARCHAR(100),
    removed_at             TIMESTAMPTZ,                                     -- 개봉/교체/분실로 활성 상태가 끝난 시각. 분실이면 확인 시각
    previous_seal_id       BIGINT,                                          -- 교체 전 씰. 같은 원본 운송 건·컨테이너·봉인 위치만 연결 가능
    memo                   TEXT,
    created_at             TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT uk_seals_id_shipment_number UNIQUE (seal_id, shipment_id, seal_no),
    CONSTRAINT uk_seals_id_trip_position UNIQUE (seal_id, shipment_id, container_no, seal_position),
    CONSTRAINT uk_seals_trip_number UNIQUE (shipment_id, seal_no),
    CONSTRAINT chk_seals_actor CHECK (((attached_by_account_id IS NOT NULL) OR (NULLIF(btrim(attached_by_name), '') IS NOT NULL))),
    CONSTRAINT chk_seals_not_self CHECK (((previous_seal_id IS NULL) OR (previous_seal_id <> seal_id))),
    CONSTRAINT chk_seals_state_time CHECK ((((status = 'ACTIVE') AND (removed_at IS NULL)) OR ((status IN ('REMOVED', 'REPLACED', 'LOST')) AND (removed_at IS NOT NULL) AND (removed_at >= attached_at)))),
    CONSTRAINT container_seals_seal_no_check CHECK (((seal_no = btrim(seal_no)) AND (length(seal_no) > 0))),
    CONSTRAINT container_seals_seal_position_check CHECK (((seal_position = btrim(seal_position)) AND (length(seal_position) > 0))),
    CONSTRAINT container_seals_seal_type_check CHECK ((seal_type IN ('CARRIER', 'CUSTOMS', 'OTHER'))),
    CONSTRAINT container_seals_status_check CHECK ((status IN ('ACTIVE', 'REMOVED', 'REPLACED', 'LOST'))),
    CONSTRAINT container_seals_attached_by_account_id_fkey FOREIGN KEY (attached_by_account_id) REFERENCES accounts(account_id) ON DELETE RESTRICT,
    CONSTRAINT fk_seals_previous_same_trip FOREIGN KEY (previous_seal_id, shipment_id, container_no, seal_position) REFERENCES container_seals(seal_id, shipment_id, container_no, seal_position),
    CONSTRAINT fk_seals_shipment_container FOREIGN KEY (shipment_id, container_no) REFERENCES shipments(shipment_id, container_no)
);

CREATE INDEX idx_seals_shipment_time ON container_seals (shipment_id, attached_at);

CREATE UNIQUE INDEX uk_seals_previous_once ON container_seals (previous_seal_id) WHERE (previous_seal_id IS NOT NULL);
COMMENT ON TABLE container_seals IS '원본 운송 건별 컨테이너 씰 장착/종료 및 교체 연결. 같은 컨테이너의 다른 운송 건 이력과 구분';
COMMENT ON COLUMN container_seals.seal_position IS '봉인 위치 코드. 기본 MAIN. 다른 위치/별도 봉인은 별도 코드 사용';
COMMENT ON COLUMN container_seals.status IS '물리적 사용 상태 ACTIVE/REMOVED/REPLACED/LOST. 훼손은 이벤트 observed_condition으로 별도 기록';
COMMENT ON COLUMN container_seals.removed_at IS '개봉/교체/분실로 활성 상태가 끝난 시각. 분실이면 확인 시각';
COMMENT ON COLUMN container_seals.previous_seal_id IS '교체 전 씰. 같은 원본 운송 건·컨테이너·봉인 위치만 연결 가능';

-- ------------------------------------------------------------
-- seal_events : 씰 장착·반복 확인·개봉·교체·분실·훼손 이력. 담당자·위치·시각·번호 관측값을 기록
-- ------------------------------------------------------------
CREATE TABLE seal_events (
    event_id           BIGSERIAL,
    seal_id            BIGINT NOT NULL,
    shipment_id        BIGINT NOT NULL,
    order_id           BIGINT,
    event_type         VARCHAR(20) NOT NULL,
    expected_seal_no   VARCHAR(30) NOT NULL,                                  -- 해당 씰의 등록 번호를 복사. 복합 FK로 부모 번호와 일치 보장
    observed_seal_no   VARCHAR(30),                                           -- 현장에서 확인한 번호. 분실/판독 불가이면 NULL
    observed_condition VARCHAR(20) NOT NULL DEFAULT 'UNKNOWN',                -- 물리 상태 INTACT/DAMAGED/MISSING/UNKNOWN. 번호 불일치는 number_matches=FALSE로 별도 표현
    number_matches     BOOLEAN GENERATED ALWAYS AS ( CASE WHEN (observed_seal_no IS NULL) THEN NULL::boolean ELSE ((observed_seal_no)::text = (expected_seal_no)::text) END) STORED,-- 관측 번호와 등록 번호의 자동 비교. 관측값 NULL이면 NULL(미판정)
    actor_account_id   BIGINT,
    actor_name         VARCHAR(100),
    center_id          BIGINT,
    occurred_at        TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    reason             TEXT,
    photo_urls         JSONB NOT NULL DEFAULT '[]'::jsonb,
    memo               TEXT,
    created_at         TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT seal_events_pkey PRIMARY KEY (event_id),
    CONSTRAINT chk_seal_events_actor CHECK (((actor_account_id IS NOT NULL) OR (NULLIF(btrim(actor_name), '') IS NOT NULL))),
    CONSTRAINT chk_seal_events_damaged CHECK (((event_type <> 'DAMAGED') OR (observed_condition = 'DAMAGED'))),
    CONSTRAINT chk_seal_events_lost CHECK (((event_type <> 'LOST') OR (observed_condition = 'MISSING'))),
    CONSTRAINT chk_seal_events_missing CHECK (((observed_condition <> 'MISSING') OR (observed_seal_no IS NULL))),
    CONSTRAINT chk_seal_events_replacement_reason CHECK (((event_type <> 'REPLACED') OR (NULLIF(btrim(reason), '') IS NOT NULL))),
    CONSTRAINT seal_events_event_type_check CHECK ((event_type IN ('ATTACHED', 'INSPECTED', 'OPENED', 'REPLACED', 'LOST', 'DAMAGED'))),
    CONSTRAINT seal_events_observed_condition_check CHECK ((observed_condition IN ('INTACT', 'DAMAGED', 'MISSING', 'UNKNOWN'))),
    CONSTRAINT seal_events_observed_seal_no_check CHECK (((observed_seal_no IS NULL) OR ((observed_seal_no = btrim(observed_seal_no)) AND (length(observed_seal_no) > 0)))),
    CONSTRAINT seal_events_photo_urls_check CHECK ((jsonb_typeof(photo_urls) = 'array')),
    CONSTRAINT fk_seal_events_order FOREIGN KEY (order_id, shipment_id) REFERENCES transport_orders(order_id, shipment_id),
    CONSTRAINT fk_seal_events_source FOREIGN KEY (seal_id, shipment_id, expected_seal_no) REFERENCES container_seals(seal_id, shipment_id, seal_no),
    CONSTRAINT seal_events_actor_account_id_fkey FOREIGN KEY (actor_account_id) REFERENCES accounts(account_id) ON DELETE RESTRICT,
    CONSTRAINT seal_events_center_id_fkey FOREIGN KEY (center_id) REFERENCES logistics_centers(center_id) ON DELETE RESTRICT
);
CREATE INDEX idx_seal_events_order ON seal_events (order_id);
CREATE INDEX idx_seal_events_seal_time ON seal_events (seal_id, occurred_at, event_id);
CREATE INDEX idx_seal_events_shipment_time ON seal_events (shipment_id, occurred_at);
COMMENT ON TABLE seal_events IS '씰 장착·반복 확인·개봉·교체·분실·훼손 이력. 담당자·위치·시각·번호 관측값을 기록';
COMMENT ON COLUMN seal_events.expected_seal_no IS '해당 씰의 등록 번호를 복사. 복합 FK로 부모 번호와 일치 보장';
COMMENT ON COLUMN seal_events.observed_seal_no IS '현장에서 확인한 번호. 분실/판독 불가이면 NULL';
COMMENT ON COLUMN seal_events.observed_condition IS '물리 상태 INTACT/DAMAGED/MISSING/UNKNOWN. 번호 불일치는 number_matches=FALSE로 별도 표현';
COMMENT ON COLUMN seal_events.number_matches IS '관측 번호와 등록 번호의 자동 비교. 관측값 NULL이면 NULL(미판정)';

-- ------------------------------------------------------------
-- handovers : HUB 도착 후 컨테이너 인계 확인 (오더당 1건)
-- ------------------------------------------------------------
CREATE TABLE handovers (
    handover_id            BIGSERIAL,
    order_id               BIGINT NOT NULL,                        -- 인계 오더 (오더당 1건)
    container_no_confirmed BOOLEAN NOT NULL DEFAULT FALSE,         -- 컨테이너 번호 확인 여부
    seal_no                VARCHAR(30),                            -- 인계 당시 실제 관측한 씰 번호 스냅샷. 불일치가 있어도 그대로 보존
    seal_ok                BOOLEAN NOT NULL DEFAULT FALSE,         -- 인계 당시 씰 종합 정상 여부. 번호 대조/물리 상태에 따른 판정은 백엔드에서 계산
    damage_free            BOOLEAN NOT NULL DEFAULT FALSE,         -- 외관 손상 없음 확인 여부
    photo_urls             JSONB,                                  -- 컨테이너 사진 URL 목록
    receiver_name          VARCHAR(50),                            -- 인수자 이름
    signature_url          TEXT,                                   -- 인수자 서명 이미지
    completed_at           TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    seal_id                BIGINT,                                 -- 인계 시 확인 대상 씰. seal_shipment_id도 함께 입력
    seal_shipment_id       BIGINT,                                 -- 인계 오더 및 씰의 원본 운송 건 일치 검증용. seal_id와 함께 NULL 또는 함께 입력

    CONSTRAINT handovers_pkey PRIMARY KEY (handover_id),
    CONSTRAINT handovers_order_id_key UNIQUE (order_id),

COMMENT ON COLUMN handovers.seal_no IS '인계 당시 실제 관측한 씰 번호 스냅샷. 불일치가 있어도 그대로 보존';

-- ------------------------------------------------------------
-- height_restrictions : 확인된 높이 제한 지점 (터미널 게이트 / HUB 입구 / 도로)
-- ------------------------------------------------------------
CREATE TABLE height_restrictions (
    restriction_id   BIGSERIAL,
    target_type      VARCHAR(20) NOT NULL,                   -- TERMINAL_GATE / HUB_ENTRANCE / ROAD
    terminal_id      BIGINT,                                 -- 대상 터미널 (TERMINAL_GATE일 때 필수)
    gate_id          BIGINT,                                 -- 대상 터미널 게이트 (TERMINAL_GATE 전용, 선택)
    hub_id           BIGINT,                                 -- 대상 허브 (HUB_ENTRANCE일 때 필수)
    restriction_name VARCHAR(150) NOT NULL,                  -- 예: ○○지하차도
    limit_height_m   NUMERIC(4,2) NOT NULL,                  -- 제한 높이 (m)
    latitude         NUMERIC(10,7),
    longitude        NUMERIC(10,7),
    alert_radius_m   INTEGER NOT NULL DEFAULT 1000,          -- 사전 경고 반경 (m)
    source           VARCHAR(100),                           -- 출처 (현장 조사, 공공데이터 등)
    verified_at      DATE,                                   -- 확인 일자
    created_at       TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT height_restrictions_pkey PRIMARY KEY (restriction_id),
    CONSTRAINT chk_height_gate_scope CHECK (((gate_id IS NULL) OR (target_type = 'TERMINAL_GATE'))),
    CONSTRAINT chk_height_positive CHECK ((limit_height_m > 0)),
    CONSTRAINT chk_height_target_ref CHECK ((((target_type = 'TERMINAL_GATE') AND (terminal_id IS NOT NULL)) OR ((target_type = 'HUB_ENTRANCE') AND (hub_id IS NOT NULL)) OR ((target_type = 'ROAD') AND (latitude IS NOT NULL) AND (longitude IS NOT NULL)))),
    CONSTRAINT chk_height_target_type CHECK ((target_type IN ('TERMINAL_GATE', 'HUB_ENTRANCE', 'ROAD'))),
    CONSTRAINT fk_height_gate FOREIGN KEY (gate_id) REFERENCES gates(gate_id) ON DELETE CASCADE,
    CONSTRAINT fk_height_hub FOREIGN KEY (hub_id) REFERENCES hubs(hub_id) ON DELETE CASCADE,
    CONSTRAINT fk_height_terminal FOREIGN KEY (terminal_id) REFERENCES terminals(terminal_id) ON DELETE CASCADE
);
CREATE INDEX idx_height_lat_lng ON height_restrictions (latitude, longitude);

-- v_order_autofill : 앱 오더 화면 자동 채움용 (컨테이너·거점·게이트·차량 제원·높이 제한)
CREATE VIEW v_order_autofill AS
SELECT o.order_id,
    o.order_no,
    o.status,
    o.vehicle_no,
    o.container_no,
    c.iso_size_type_code,
    c.is_high_cube,
    c.max_gross_kg,
    t.terminal_name AS origin_name,
    t.latitude AS origin_lat,
    t.longitude AS origin_lng,
    dest.center_name AS dest_name,
    dest.address AS dest_address,
    dest.latitude AS dest_lat,
    dest.longitude AS dest_lng,
    gl.pass_at AS gate_out_at,
    g.gate_name AS gate_out_name,
    vs.empty_weight_kg,
    vs.max_payload_kg,
    vs.axle_count,
    vs.chassis_height_m,
    ( SELECT min(hr.limit_height_m) AS min
           FROM height_restrictions hr
          WHERE hr.target_type::text = 'HUB_ENTRANCE'::text AND hr.hub_id = h.hub_id) AS dest_height_limit_m,
    ( SELECT min(hr.limit_height_m) AS min
           FROM height_restrictions hr
          WHERE hr.target_type::text = 'TERMINAL_GATE'::text AND hr.terminal_id = t.terminal_id) AS origin_height_limit_m
   FROM transport_orders o
     JOIN containers c ON c.container_no::text = o.container_no::text
     JOIN terminals t ON t.terminal_id = o.origin_terminal_id
     JOIN hubs h ON h.hub_id = o.dest_hub_id
     JOIN logistics_centers dest ON dest.center_id = h.center_id
     LEFT JOIN gate_logs gl ON gl.gate_log_id = o.gate_out_log_id
     LEFT JOIN gates g ON g.gate_id = gl.gate_id
     LEFT JOIN vehicle_specs vs ON vs.vehicle_no::text = o.vehicle_no::text;

CREATE OR REPLACE FUNCTION distance_m(lat1 numeric, lng1 numeric, lat2 numeric, lng2 numeric)
 RETURNS double precision
 LANGUAGE sql
 IMMUTABLE
AS $function$
    SELECT 6371000 * 2 * asin(sqrt(
        power(sin(radians((lat2 - lat1)::float8) / 2), 2) +
        cos(radians(lat1::float8)) * cos(radians(lat2::float8)) *
        power(sin(radians((lng2 - lng1)::float8) / 2), 2)))
$function$;

CREATE OR REPLACE FUNCTION scargo_check_port_leg_shipment()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE leg RECORD;
BEGIN
 IF NEW.assignment_id IS NULL THEN RETURN NEW; END IF;
 SELECT shipment_id,leg_type INTO leg FROM shipment_legs
 WHERE assignment_id=NEW.assignment_id FOR SHARE;
 IF NOT FOUND THEN
  RAISE EXCEPTION '존재하지 않는 운송 구간: %',NEW.assignment_id USING ERRCODE='23503';
 END IF;
 IF leg.leg_type='PORT_TO_HUB' THEN
  IF NEW.shipment_id IS NULL THEN
   NEW.shipment_id := leg.shipment_id;
  ELSIF NEW.shipment_id IS DISTINCT FROM leg.shipment_id THEN
   RAISE EXCEPTION '항만→허브 운송 건 불일치: table=%, assignment_id=%, 입력 shipment_id=%, 구간 shipment_id=%',
    TG_TABLE_NAME,NEW.assignment_id,NEW.shipment_id,leg.shipment_id USING ERRCODE='23514';
  END IF;
 END IF;
 RETURN NEW;
END $function$;

CREATE OR REPLACE FUNCTION scargo_freeze_paid_passage()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
 IF TG_OP = 'UPDATE' THEN
  IF NEW IS NOT DISTINCT FROM OLD THEN RETURN NEW; END IF;
 END IF;
 IF EXISTS (SELECT 1 FROM toll_payments
            WHERE passage_id = OLD.passage_id OR entry_passage_id = OLD.passage_id) THEN
  RAISE EXCEPTION '결제 기록에 연결된 통과 기록은 수정/삭제할 수 없습니다 (passage_id=%)', OLD.passage_id
   USING ERRCODE = '23514';
 END IF;
 IF TG_OP = 'DELETE' THEN RETURN OLD; END IF;
 RETURN NEW;
END $function$;

CREATE OR REPLACE FUNCTION scargo_guard_check_passages()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
 IF EXISTS (SELECT 1 FROM toll_passages p WHERE p.check_id=OLD.check_id AND
    (p.vehicle_no IS DISTINCT FROM NEW.vehicle_no
     OR p.assignment_id IS DISTINCT FROM NEW.assignment_id
     OR (p.passage_status='PASSED' AND (NEW.is_violation OR NOT NEW.is_passed))
     OR (p.block_reason='OVERLOAD' AND NOT NEW.is_violation))) THEN
  RAISE EXCEPTION '검사 변경이 기존 통과 기록과 모순됩니다: %',OLD.check_id USING ERRCODE='23514';
 END IF;
 RETURN NEW;
END $function$;

CREATE OR REPLACE FUNCTION scargo_guard_port_leg_change()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE t TEXT; bad BOOLEAN;
BEGIN
 IF NEW.leg_type <> 'PORT_TO_HUB' THEN RETURN NEW; END IF;
 FOREACH t IN ARRAY ARRAY['item_vehicle_assignments','item_history','gate_logs','overload_checks'] LOOP
  EXECUTE format('SELECT EXISTS(SELECT 1 FROM %I WHERE assignment_id=$1 AND shipment_id IS DISTINCT FROM $2)',t)
   INTO bad USING OLD.assignment_id,NEW.shipment_id;
  IF bad THEN
   RAISE EXCEPTION '구간 변경 불가: %의 기존 연결과 운송 건이 다릅니다 (assignment_id=%)',t,OLD.assignment_id
    USING ERRCODE='23514';
  END IF;
 END LOOP;
 RETURN NEW;
END $function$;

CREATE OR REPLACE FUNCTION scargo_lock_payment_passages()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
 IF TG_OP='DELETE' THEN
  RAISE EXCEPTION '결제 이력은 삭제하지 않고 취소/환불 상태로 보존하세요' USING ERRCODE='23514';
 END IF;
 IF TG_OP='UPDATE' AND
   (NEW.passage_id IS DISTINCT FROM OLD.passage_id OR NEW.entry_passage_id IS DISTINCT FROM OLD.entry_passage_id) THEN
  RAISE EXCEPTION '결제에 연결된 진입/진출 통과 기록은 변경할 수 없습니다' USING ERRCODE='23514';
 END IF;
 PERFORM passage_id FROM toll_passages
 WHERE passage_id IN (NEW.passage_id,NEW.entry_passage_id)
 ORDER BY passage_id FOR UPDATE;
 RETURN NEW;
END $function$;

CREATE OR REPLACE FUNCTION scargo_validate_amr_task_hub()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
 IF NEW.hub_id IS NULL THEN RAISE EXCEPTION 'AMR 작업 hub_id 필수'; END IF;
 IF NEW.robot_id IS NOT NULL AND NOT EXISTS
    (SELECT 1 FROM amr_robots WHERE robot_id=NEW.robot_id AND hub_id=NEW.hub_id) THEN
   RAISE EXCEPTION 'AMR 로봇 HUB 불일치'; END IF;
 IF NEW.parcel_no IS NOT NULL AND NOT EXISTS
    (SELECT 1 FROM parcels WHERE parcel_no=NEW.parcel_no AND hub_id=NEW.hub_id) THEN
   RAISE EXCEPTION 'AMR 물품 HUB 불일치'; END IF;
 IF NEW.from_dock_id IS NOT NULL AND NOT EXISTS
    (SELECT 1 FROM hub_docks WHERE dock_id=NEW.from_dock_id AND hub_id=NEW.hub_id) THEN
   RAISE EXCEPTION 'AMR 출발 도크 HUB 불일치'; END IF;
 IF NEW.to_dock_id IS NOT NULL AND NOT EXISTS
    (SELECT 1 FROM hub_docks WHERE dock_id=NEW.to_dock_id AND hub_id=NEW.hub_id) THEN
   RAISE EXCEPTION 'AMR 도착 도크 HUB 불일치'; END IF;
 IF NEW.from_slot_id IS NOT NULL AND NOT EXISTS
    (SELECT 1 FROM hub_rack_slots WHERE slot_id=NEW.from_slot_id AND hub_id=NEW.hub_id) THEN
   RAISE EXCEPTION 'AMR 출발 랙 HUB 불일치'; END IF;
 IF NEW.to_slot_id IS NOT NULL AND NOT EXISTS
    (SELECT 1 FROM hub_rack_slots WHERE slot_id=NEW.to_slot_id AND hub_id=NEW.hub_id) THEN
   RAISE EXCEPTION 'AMR 도착 랙 HUB 불일치'; END IF;
 RETURN NEW;
END $function$;

CREATE OR REPLACE FUNCTION scargo_validate_hub_transport()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE x RECORD;
BEGIN
 SELECT l.vehicle_no, l.from_center_id, l.to_center_id, l.leg_type, h.center_id
 INTO x FROM shipment_legs l JOIN hubs h ON h.hub_id=NEW.hub_id
 WHERE l.assignment_id=NEW.assignment_id;
 IF NOT FOUND THEN RAISE EXCEPTION '운송 구간 또는 HUB가 없음'; END IF;
 IF TG_TABLE_NAME='hub_inbound_receipts' THEN
   IF x.leg_type <> 'PORT_TO_HUB' OR x.to_center_id <> x.center_id
      OR NEW.vehicle_no IS DISTINCT FROM x.vehicle_no THEN
     RAISE EXCEPTION 'HUB 입고 차량/도착 HUB/운송 구간 불일치';
   END IF;
 ELSE
   IF x.leg_type <> 'HUB_TO_CENTER' OR x.from_center_id <> x.center_id
      OR NEW.delivery_vehicle_no IS DISTINCT FROM x.vehicle_no THEN
     RAISE EXCEPTION 'HUB 출고 차량/출발 HUB/운송 구간 불일치';
   END IF;
   IF NOT EXISTS (SELECT 1 FROM logistics_centers c
       WHERE c.center_id=x.to_center_id AND c.region_id=NEW.region_id) THEN
     RAISE EXCEPTION '출고 권역과 운송 구간 목적지 물류센터 불일치';
   END IF;
 END IF;
 RETURN NEW;
END $function$;

CREATE OR REPLACE FUNCTION scargo_validate_item_assignment_destination()

   NEW.item_id, NEW.assignment_id USING ERRCODE = '23514';
 END IF;
 RETURN NEW;
END $function$;

CREATE OR REPLACE FUNCTION scargo_validate_order_leg()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE x RECORD;
BEGIN
 IF NEW.assignment_id IS NULL THEN
   IF NEW.status NOT IN ('CREATED','CANCELED') AND NEW.shipment_id IS NOT NULL THEN
     RAISE EXCEPTION '배차 이후 앱 오더는 운송 구간이 필요합니다: %', NEW.order_no;
   END IF;
   RETURN NEW;
 END IF;
 IF NEW.shipment_id IS NULL THEN RAISE EXCEPTION 'assignment_id 사용 시 shipment_id 필수'; END IF;
 SELECT l.vehicle_no, l.from_center_id, l.to_center_id, l.leg_type,
        s.container_no, t.center_id AS terminal_center_id, h.center_id AS hub_center_id
 INTO x FROM shipment_legs l JOIN shipments s ON s.shipment_id=l.shipment_id
 JOIN terminals t ON t.terminal_id=NEW.origin_terminal_id
 JOIN hubs h ON h.hub_id=NEW.dest_hub_id WHERE l.assignment_id=NEW.assignment_id;
 IF NOT FOUND OR x.leg_type <> 'PORT_TO_HUB' OR NEW.vehicle_no IS DISTINCT FROM x.vehicle_no
    OR NEW.container_no IS DISTINCT FROM x.container_no
    OR x.from_center_id IS DISTINCT FROM x.terminal_center_id
    OR x.to_center_id IS DISTINCT FROM x.hub_center_id THEN
   RAISE EXCEPTION '앱 오더와 운송 구간/차량/컨테이너/거점 불일치: %', NEW.order_no;
 END IF;
 RETURN NEW;
END $function$;

CREATE OR REPLACE FUNCTION scargo_validate_parcel_links()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE rec_ship BIGINT; item_ship BIGINT; dest_region BIGINT; disp_region BIGINT;
BEGIN
 IF NEW.receipt_id IS NOT NULL THEN
  SELECT l.shipment_id INTO rec_ship
  FROM hub_inbound_receipts r JOIN shipment_legs l ON l.assignment_id = r.assignment_id
  WHERE r.receipt_id = NEW.receipt_id;
  SELECT shipment_id INTO item_ship FROM items WHERE item_id = NEW.item_id;
  IF rec_ship IS DISTINCT FROM item_ship THEN
   RAISE EXCEPTION '박스의 입고 접수 운송 건(%)과 물품의 운송 건(%)이 다릅니다 (item_id=%)',
    rec_ship, item_ship, NEW.item_id USING ERRCODE = '23514';
  END IF;
 END IF;

 SELECT c.region_id INTO dest_region
 FROM items i JOIN logistics_centers c ON c.center_id = i.destination_center_id
 WHERE i.item_id = NEW.item_id;

 IF NEW.region_id IS NOT NULL AND NEW.status <> 'SORT_FAILED'
    AND NEW.region_id IS DISTINCT FROM dest_region THEN
  RAISE EXCEPTION '박스 권역(%)이 물품 목적지 센터의 권역(%)과 다릅니다 (item_id=%)',
   NEW.region_id, dest_region, NEW.item_id USING ERRCODE = '23514';
 END IF;

 IF NEW.dispatch_id IS NOT NULL THEN
  SELECT region_id INTO disp_region FROM hub_dispatches WHERE dispatch_id = NEW.dispatch_id;
  IF disp_region IS DISTINCT FROM dest_region THEN
   RAISE EXCEPTION '출고 권역(%)과 물품 목적지 권역(%)이 다릅니다 (item_id=%)',
    disp_region, dest_region, NEW.item_id USING ERRCODE = '23514';
  END IF;
 END IF;
 RETURN NEW;
END $function$;

CREATE OR REPLACE FUNCTION scargo_validate_passage_check()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE c RECORD;
BEGIN
 IF NEW.check_id IS NULL THEN RETURN NEW; END IF;
 SELECT vehicle_no,assignment_id,is_violation,is_passed INTO c
 FROM overload_checks WHERE check_id=NEW.check_id FOR SHARE;
 IF NOT FOUND THEN RAISE EXCEPTION '존재하지 않는 검사: %',NEW.check_id USING ERRCODE='23503'; END IF;
 IF NEW.vehicle_no IS DISTINCT FROM c.vehicle_no OR NEW.assignment_id IS DISTINCT FROM c.assignment_id THEN
  RAISE EXCEPTION '통과 기록과 검사 차량/운송 구간 불일치' USING ERRCODE='23514';
 END IF;
 IF NEW.passage_status='PASSED' AND (c.is_violation OR NOT c.is_passed) THEN
  RAISE EXCEPTION '위반 또는 불합격 검사로 통과 처리할 수 없습니다' USING ERRCODE='23514';
 END IF;
 IF NEW.block_reason='OVERLOAD' AND NOT c.is_violation THEN
  RAISE EXCEPTION '과적 차단은 위반 검사 결과가 필요합니다' USING ERRCODE='23514';
 END IF;
 RETURN NEW;
END $function$;

CREATE OR REPLACE FUNCTION scargo_validate_receipt_container()

   USING ERRCODE = '23514';

CREATE OR REPLACE FUNCTION scargo_validate_toll_payment()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE p RECORD; e RECORD;
BEGIN
 IF NEW.passage_id IS NULL THEN RAISE EXCEPTION '결제에는 통과 이벤트 passage_id가 필요'; END IF;
 SELECT * INTO p FROM toll_passages WHERE passage_id=NEW.passage_id;
 IF NOT FOUND OR p.vehicle_no IS DISTINCT FROM NEW.vehicle_no OR
    p.tollgate_id IS DISTINCT FROM NEW.tollgate_id OR p.passage_status <> 'PASSED' THEN
   RAISE EXCEPTION '결제와 진출/개방식 통과 이벤트 불일치'; END IF;
 IF NEW.assignment_id IS DISTINCT FROM p.assignment_id THEN
   RAISE EXCEPTION '결제와 통과 이벤트의 운송 구간 불일치'; END IF;
 IF NEW.toll_system='CLOSED' THEN
   SELECT * INTO e FROM toll_passages WHERE passage_id=NEW.entry_passage_id;
   IF NOT FOUND OR e.vehicle_no IS DISTINCT FROM NEW.vehicle_no
      OR e.passage_status <> 'PASSED' OR e.detected_at > p.detected_at THEN
     RAISE EXCEPTION '폐쇄식 진입 이벤트와 진출 이벤트 불일치'; END IF;
 END IF;
 RETURN NEW;
END $function$;

-- 11차: 참조 행 FOR SHARE 잠금, ERRCODE 23514 지정 (검증 로직은 10차와 동일)
CREATE OR REPLACE FUNCTION trg_check_shipment_leg()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE
    v_cat  VARCHAR;
    f_type VARCHAR;
    t_type VARCHAR;
BEGIN
    SELECT vehicle_category INTO v_cat  FROM trucks WHERE vehicle_no = NEW.vehicle_no FOR SHARE;
    SELECT center_type      INTO f_type FROM logistics_centers WHERE center_id = NEW.from_center_id FOR SHARE;
    SELECT center_type      INTO t_type FROM logistics_centers WHERE center_id = NEW.to_center_id FOR SHARE;

    IF NEW.leg_type = 'PORT_TO_HUB'
       AND (v_cat IS DISTINCT FROM 'CONTAINER_TRUCK' OR f_type <> 'PORT' OR t_type <> 'HUB') THEN
        RAISE EXCEPTION 'PORT_TO_HUB 구간은 컨테이너 특수화물차(CONTAINER_TRUCK), 항만→허브여야 합니다 (vehicle_no=%)', NEW.vehicle_no
         USING ERRCODE = '23514';
    ELSIF NEW.leg_type = 'HUB_TO_CENTER'
       AND (v_cat IS DISTINCT FROM 'GENERAL_TRUCK' OR f_type <> 'HUB' OR t_type <> 'DISTRIBUTION_CENTER') THEN
        RAISE EXCEPTION 'HUB_TO_CENTER 구간은 일반화물차(GENERAL_TRUCK), 허브→물류센터여야 합니다 (vehicle_no=%)', NEW.vehicle_no
         USING ERRCODE = '23514';
    END IF;
    RETURN NEW;
END $function$;

-- 11차 보강: 중복 저장 값 불일치 차단 (10차 검토 🔴 항목)
--   R1. 운송 구간(shipment_legs)/운송 건(shipments) 변경 시 연결된 오더·입고·출고·물품 배정 재검증
--   R2. 과적검사·게이트 OCR·톨게이트 통과 기록의 차량이 운송 구간 차량과 같은지 검증
--   R3. 운행 중인 차량의 차량 구분(vehicle_category)/세미트레일러 여부 변경 차단
--   R4. 사용 중인 물류 거점의 유형(center_type) 변경 차단, 터미널 거점은 PORT만 허용
--   R5. 다른 테이블을 읽는 기존 검증 트리거에 FOR SHARE 잠금 추가 (동시 수정으로 검증 우회 방지)
-- 모든 오류는 SQLSTATE 23514(check_violation)로 발생합니다.
--
-- [운영 규칙] 구간 차량 교체 등 연결 값 변경은 한 트랜잭션에서 "구간 먼저 → 연결 테이블" 순서로 수정합니다.
--   BEGIN;
--   UPDATE shipment_legs    SET vehicle_no = '새차량' WHERE assignment_id = 1;
--   UPDATE transport_orders SET vehicle_no = '새차량' WHERE assignment_id = 1;
--   COMMIT;   -- R1 재검증은 COMMIT 시점(지연 트리거)에 수행
-- ============================================================

-- ------------------------------------------------------------
-- R1-1. 운송 구간 변경 → 연결 테이블 재검증 (COMMIT 시점)
-- ------------------------------------------------------------
CREATE OR REPLACE FUNCTION scargo_recheck_leg_links()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE l RECORD; bad TEXT;
BEGIN
 -- 같은 트랜잭션에서 여러 번 수정될 수 있으므로 COMMIT 시점의 현재 값으로 검사
 SELECT * INTO l FROM shipment_legs WHERE assignment_id = NEW.assignment_id;
 IF NOT FOUND THEN RETURN NULL; END IF;

 SELECT o.order_no INTO bad
 FROM transport_orders o
 JOIN terminals t ON t.terminal_id = o.origin_terminal_id
 JOIN hubs h ON h.hub_id = o.dest_hub_id
 LEFT JOIN shipments s ON s.shipment_id = l.shipment_id
 WHERE o.assignment_id = l.assignment_id
   AND (l.leg_type <> 'PORT_TO_HUB'
        OR o.vehicle_no IS DISTINCT FROM l.vehicle_no
        OR o.container_no IS DISTINCT FROM s.container_no
        OR t.center_id IS DISTINCT FROM l.from_center_id
        OR h.center_id IS DISTINCT FROM l.to_center_id)
 LIMIT 1;
 IF FOUND THEN
  RAISE EXCEPTION '운송 구간 변경이 앱 오더와 맞지 않습니다 (assignment_id=%, order_no=%). 오더도 같은 트랜잭션에서 수정하세요',
   l.assignment_id, bad USING ERRCODE = '23514';
 END IF;

 SELECT r.receipt_id::text INTO bad
 FROM hub_inbound_receipts r
 JOIN hubs h ON h.hub_id = r.hub_id
 LEFT JOIN shipments s ON s.shipment_id = l.shipment_id
 WHERE r.assignment_id = l.assignment_id
   AND (l.leg_type <> 'PORT_TO_HUB'
        OR r.vehicle_no IS DISTINCT FROM l.vehicle_no
        OR h.center_id IS DISTINCT FROM l.to_center_id
        OR (r.container_no IS NOT NULL AND r.container_no IS DISTINCT FROM s.container_no))
 LIMIT 1;
 IF FOUND THEN
  RAISE EXCEPTION '운송 구간 변경이 허브 입고 접수와 맞지 않습니다 (assignment_id=%, receipt_id=%)',
   l.assignment_id, bad USING ERRCODE = '23514';
 END IF;

 SELECT d.dispatch_id::text INTO bad
 FROM hub_dispatches d
 JOIN hubs h ON h.hub_id = d.hub_id
 LEFT JOIN logistics_centers c ON c.center_id = l.to_center_id
 WHERE d.assignment_id = l.assignment_id
   AND (l.leg_type <> 'HUB_TO_CENTER'
        OR d.delivery_vehicle_no IS DISTINCT FROM l.vehicle_no
        OR h.center_id IS DISTINCT FROM l.from_center_id
        OR c.region_id IS DISTINCT FROM d.region_id)
 LIMIT 1;
 IF FOUND THEN
  RAISE EXCEPTION '운송 구간 변경이 허브 출고와 맞지 않습니다 (assignment_id=%, dispatch_id=%)',
   l.assignment_id, bad USING ERRCODE = '23514';
 END IF;

 IF l.leg_type = 'HUB_TO_CENTER' THEN
  SELECT a.item_id::text INTO bad
  FROM item_vehicle_assignments a JOIN items i ON i.item_id = a.item_id
  WHERE a.assignment_id = l.assignment_id AND a.status <> 'CANCELED'
    AND i.destination_center_id IS DISTINCT FROM l.to_center_id
  LIMIT 1;
  IF FOUND THEN
   RAISE EXCEPTION '운송 구간 도착 센터가 배정된 물품 목적지와 다릅니다 (assignment_id=%, item_id=%)',
    l.assignment_id, bad USING ERRCODE = '23514';
  END IF;
 END IF;
 RETURN NULL;
END $function$;

CREATE CONSTRAINT TRIGGER trg_recheck_leg_links
 AFTER UPDATE OF vehicle_no, shipment_id, from_center_id, to_center_id, leg_type ON shipment_legs
 DEFERRABLE INITIALLY DEFERRED
 FOR EACH ROW EXECUTE FUNCTION scargo_recheck_leg_links();

-- ------------------------------------------------------------
-- R1-2. 운송 건 컨테이너 변경 → 앱 오더·입고 접수 재검증 (COMMIT 시점)
-- ------------------------------------------------------------
CREATE OR REPLACE FUNCTION scargo_recheck_shipment_container()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE c VARCHAR(11); bad TEXT;
BEGIN
 SELECT container_no INTO c FROM shipments WHERE shipment_id = NEW.shipment_id;
 IF NOT FOUND THEN RETURN NULL; END IF;

 SELECT o.order_no INTO bad FROM transport_orders o
 WHERE o.shipment_id = NEW.shipment_id AND o.container_no IS DISTINCT FROM c LIMIT 1;
 IF FOUND THEN
  RAISE EXCEPTION '운송 건 컨테이너(%)가 앱 오더(%)의 컨테이너와 다릅니다', c, bad USING ERRCODE = '23514';
 END IF;

 SELECT r.receipt_id::text INTO bad
 FROM hub_inbound_receipts r JOIN shipment_legs l ON l.assignment_id = r.assignment_id
 WHERE l.shipment_id = NEW.shipment_id AND r.container_no IS NOT NULL AND r.container_no IS DISTINCT FROM c
 LIMIT 1;
 IF FOUND THEN
  RAISE EXCEPTION '운송 건 컨테이너(%)가 허브 입고 접수(receipt_id=%)의 컨테이너와 다릅니다', c, bad USING ERRCODE = '23514';

END $function$;

-- ------------------------------------------------------------
-- R2. 기록 차량 = 운송 구간 차량 (입력·수정 시점 검사)
--     과거 기록은 차량 교체 후에도 당시 차량을 그대로 보존합니다.
--     트리거 이름이 *_port_shipment 보다 뒤에 정렬되어 shipment_id 자동 채움 이후 실행됩니다.
-- ------------------------------------------------------------
CREATE OR REPLACE FUNCTION scargo_validate_record_vehicle()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE v TEXT; leg_vehicle TEXT; ship_container TEXT;
BEGIN
 IF NEW.assignment_id IS NULL THEN RETURN NEW; END IF;
 v := CASE TG_TABLE_NAME WHEN 'gate_logs' THEN to_jsonb(NEW)->>'actual_vehicle_no'
                         ELSE to_jsonb(NEW)->>'vehicle_no' END;
 SELECT vehicle_no INTO leg_vehicle FROM shipment_legs WHERE assignment_id = NEW.assignment_id FOR SHARE;
 IF v IS NOT NULL AND v IS DISTINCT FROM leg_vehicle THEN
  RAISE EXCEPTION '%: 기록 차량(%)과 운송 구간 차량(%)이 다릅니다 (assignment_id=%)',
   TG_TABLE_NAME, v, leg_vehicle, NEW.assignment_id USING ERRCODE = '23514';
 END IF;

 IF TG_TABLE_NAME = 'overload_checks' AND (to_jsonb(NEW)->>'container_no') IS NOT NULL
    AND (to_jsonb(NEW)->>'shipment_id') IS NOT NULL THEN
  SELECT container_no INTO ship_container FROM shipments
  WHERE shipment_id = (to_jsonb(NEW)->>'shipment_id')::bigint;
  IF ship_container IS DISTINCT FROM to_jsonb(NEW)->>'container_no' THEN
   RAISE EXCEPTION 'overload_checks: 검사 컨테이너(%)와 운송 건 컨테이너(%)가 다릅니다',
    to_jsonb(NEW)->>'container_no', ship_container USING ERRCODE = '23514';
  END IF;
 END IF;
 RETURN NEW;
END $function$;

 BEFORE INSERT OR UPDATE OF vehicle_no, assignment_id, shipment_id, container_no ON overload_checks
 FOR EACH ROW EXECUTE FUNCTION scargo_validate_record_vehicle();

 BEFORE INSERT OR UPDATE OF actual_vehicle_no, assignment_id ON gate_logs

CREATE TRIGGER trg_passage_vehicle_match
 BEFORE INSERT OR UPDATE OF vehicle_no, assignment_id ON toll_passages

-- ------------------------------------------------------------
-- R3. 운행 중인 차량의 구분/세미트레일러 여부 변경 차단
--     완료·취소된 과거 구간만 있는 차량은 변경 가능
-- ------------------------------------------------------------
CREATE OR REPLACE FUNCTION scargo_guard_truck_change()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
 IF NEW.vehicle_category IS DISTINCT FROM OLD.vehicle_category AND EXISTS (
     SELECT 1 FROM shipment_legs
     WHERE vehicle_no = NEW.vehicle_no AND status IN ('ASSIGNED','IN_TRANSIT')
       AND ((leg_type = 'PORT_TO_HUB'   AND NEW.vehicle_category IS DISTINCT FROM 'CONTAINER_TRUCK')
         OR (leg_type = 'HUB_TO_CENTER' AND NEW.vehicle_category IS DISTINCT FROM 'GENERAL_TRUCK'))) THEN
  RAISE EXCEPTION '진행 중인 운송 구간이 있어 차량 구분을 %로 변경할 수 없습니다 (vehicle_no=%)',
   NEW.vehicle_category, NEW.vehicle_no USING ERRCODE = '23514';
 END IF;
 IF OLD.is_semi_trailer AND NOT NEW.is_semi_trailer AND EXISTS (
     SELECT 1 FROM loading_records
     WHERE vehicle_no = NEW.vehicle_no AND status IN ('PENDING','IN_PROGRESS')) THEN
  RAISE EXCEPTION '진행 중인 컨테이너 적재 기록이 있어 세미트레일러 여부를 해제할 수 없습니다 (vehicle_no=%)',
   NEW.vehicle_no USING ERRCODE = '23514';
 END IF;
 RETURN NEW;
END $function$;

CREATE TRIGGER trg_guard_truck_change
 BEFORE UPDATE OF vehicle_category, is_semi_trailer ON trucks
 FOR EACH ROW EXECUTE FUNCTION scargo_guard_truck_change();

-- ------------------------------------------------------------
-- R4-1. 사용 중인 물류 거점의 유형 변경 차단
-- ------------------------------------------------------------
CREATE OR REPLACE FUNCTION scargo_guard_center_type_change()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
DECLARE t TEXT;
BEGIN
 IF NEW.center_type IS NOT DISTINCT FROM OLD.center_type THEN RETURN NEW; END IF;
 SELECT x.tbl INTO t FROM (
   SELECT 'hubs' tbl WHERE EXISTS (SELECT 1 FROM hubs WHERE center_id = OLD.center_id)
   UNION ALL SELECT 'terminals' WHERE EXISTS (SELECT 1 FROM terminals WHERE center_id = OLD.center_id)
   UNION ALL SELECT 'gates' WHERE EXISTS (SELECT 1 FROM gates WHERE center_id = OLD.center_id)
   UNION ALL SELECT 'shipments' WHERE EXISTS (SELECT 1 FROM shipments
             WHERE OLD.center_id IN (origin_center_id, destination_center_id))
   UNION ALL SELECT 'shipment_legs' WHERE EXISTS (SELECT 1 FROM shipment_legs
             WHERE OLD.center_id IN (from_center_id, to_center_id))
   UNION ALL SELECT 'items' WHERE EXISTS (SELECT 1 FROM items WHERE destination_center_id = OLD.center_id)
   UNION ALL SELECT 'item_history' WHERE EXISTS (SELECT 1 FROM item_history WHERE center_id = OLD.center_id)
   UNION ALL SELECT 'seal_events' WHERE EXISTS (SELECT 1 FROM seal_events WHERE center_id = OLD.center_id)
 ) x LIMIT 1;
 IF t IS NOT NULL THEN
  RAISE EXCEPTION '%에서 사용 중인 거점은 유형을 % → %로 변경할 수 없습니다 (center_id=%)',
   t, OLD.center_type, NEW.center_type, OLD.center_id USING ERRCODE = '23514';
 END IF;
 RETURN NEW;
END $function$;

CREATE TRIGGER trg_guard_center_type_change
 BEFORE UPDATE OF center_type ON logistics_centers
 FOR EACH ROW EXECUTE FUNCTION scargo_guard_center_type_change();

-- ------------------------------------------------------------
-- R4-2. terminals.center_id는 PORT 거점만 허용
-- ------------------------------------------------------------
CREATE OR REPLACE FUNCTION scargo_check_terminal_center()
 RETURNS trigger
 LANGUAGE plpgsql
AS $function$
BEGIN
 IF NEW.center_id IS NOT NULL AND NOT EXISTS (
     SELECT 1 FROM logistics_centers WHERE center_id = NEW.center_id AND center_type = 'PORT' FOR SHARE) THEN
  RAISE EXCEPTION 'terminals.center_id는 PORT 거점이어야 합니다 (center_id=%)', NEW.center_id
   USING ERRCODE = '23514';
 END IF;
 RETURN NEW;
END $function$;

CREATE TRIGGER trg_terminals_center_check
 BEFORE INSERT OR UPDATE OF center_id ON terminals
 FOR EACH ROW EXECUTE FUNCTION scargo_check_terminal_center();

COMMIT;
