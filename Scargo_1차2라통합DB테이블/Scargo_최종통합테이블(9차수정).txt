--   - 허브는 HUB-GONJIAM-01 한 곳만 생성하고, 앱 오더 3건도 이 허브로 연결합니다.
--   - 출발 항만은 인천항이며 B경로 과적검문소 5곳(38 → 18 → 36 → 10 → 34)을 포함합니다.
--     좌표는 HTML/JSON 제공값을 대조했고, NUMERIC(10,7)에 맞춰 소수 7자리로 저장됩니다.
--   - 전국 허브/영업소 원본 시드는 포함하지 않습니다.
--   - shipment_routes, shipment_events는 두지 않고 shipment_legs에 leg_sequence를 둡니다.
-- 변경 이력
--   9차: 오래된 주석 정리, 차량 FK를 RESTRICT로 변경 (fk_order_truck, fk_receipts_truck, fk_dispatches_vehicle)
--   8차: 배차 필수값 CHECK 추가 (chk_order_assignment_required)
--   7차: 민간계량소 기능 제거 (최종 50개 테이블)
--   3~6차: 중복 테이블 통합, 컬럼 개명, 제약/트리거 보강, 더미 데이터 정리

-- 1. companies 테이블 (업체 정보)

CREATE TABLE companies (
    company_id BIGSERIAL PRIMARY KEY,
    company_name VARCHAR(100) NOT NULL,
    address TEXT NOT NULL,
    business_no VARCHAR(20) UNIQUE,
    industry_type VARCHAR(50),
    representative_name VARCHAR(50),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

-- ==========================================
-- 2. accounts 테이블 (개인회원 GENERAL / 관리자 ADMIN)
-- ==========================================
CREATE TABLE accounts (
    account_id BIGSERIAL PRIMARY KEY,
    user_name VARCHAR(30),
    user_id VARCHAR(50) NOT NULL UNIQUE,
    user_pw VARCHAR(255) NOT NULL,
    user_type VARCHAR(30) NOT NULL DEFAULT 'GENERAL',
    company_id BIGINT,
    phone_num VARCHAR(20),

        REFERENCES companies(company_id) ON DELETE SET NULL,
    CONSTRAINT chk_user_type CHECK (user_type IN ('GENERAL', 'ADMIN'))

    is_deleted BOOLEAN NOT NULL DEFAULT FALSE,

    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_posts_account FOREIGN KEY (account_id)
        REFERENCES accounts(account_id) ON DELETE SET NULL,
    CONSTRAINT chk_post_category CHECK (category IN ('NOTICE', 'FREE', 'QNA', 'FAQ'))

-- 4. trucks 테이블 (물류 화물차 정보 관리)
-- ==========================================
CREATE TABLE trucks (
    vehicle_no VARCHAR(20) PRIMARY KEY,
    company_id BIGINT NOT NULL,
    is_semi_trailer BOOLEAN NOT NULL DEFAULT FALSE,
    trailer_no VARCHAR(20),
    truck_type VARCHAR(30),
    max_load_weight NUMERIC(8, 2),
    planned_route JSONB,
    status VARCHAR(20) DEFAULT 'OUTSIDE',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_trucks_company FOREIGN KEY (company_id)
        REFERENCES companies(company_id),
    CONSTRAINT chk_truck_status CHECK (status IN ('OUTSIDE', 'INSIDE', 'IN_TRANSIT'))

-- ==========================================
-- 5. containers 테이블 (컨테이너 정보)
-- ==========================================
CREATE TABLE containers (
    container_no VARCHAR(11) PRIMARY KEY,
    company_id BIGINT,
    iso_size_type_code VARCHAR(4) NOT NULL,
    container_type VARCHAR(20) NOT NULL,
    is_high_cube BOOLEAN NOT NULL DEFAULT FALSE,
    max_gross_kg NUMERIC(8, 1) NOT NULL,
    tare_kg NUMERIC(8, 1) NOT NULL,
    net_kg NUMERIC(8, 1) NOT NULL,  -- 최대 적재 가능 중량(max_gross-tare). 실제 화물 중량 아님 (3차 수정 구간에서 max_payload_kg로 개명)
    cubic_capacity_cbm NUMERIC(6, 2) NOT NULL,
    csc_approval_no VARCHAR(30),
    reserved_cargo_info JSONB,
    loading_location_id BIGINT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_containers_company FOREIGN KEY (company_id)
        REFERENCES companies(company_id) ON DELETE SET NULL,
    CONSTRAINT fk_containers_location FOREIGN KEY (loading_location_id)
        REFERENCES loading_locations(location_id) ON DELETE SET NULL,
    CONSTRAINT chk_containers_no_format CHECK (container_no ~ '^[A-Z]{4}[0-9]{7}$'),
    CONSTRAINT chk_containers_weight CHECK (max_gross_kg >= tare_kg + net_kg)
);

-- ==========================================
-- 6. loading_records 테이블 (컨테이너 적재 이력)
-- ==========================================
CREATE TABLE loading_records (
    record_id BIGSERIAL PRIMARY KEY,
    vehicle_no VARCHAR(20) NOT NULL,
    container_no VARCHAR(11) NOT NULL,
    location_id BIGINT NOT NULL,
    -- 첫 게이트 OCR 인식 전 상하차 작업 대기 상태
    status VARCHAR(20) NOT NULL DEFAULT 'PENDING',
    loaded_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_records_truck FOREIGN KEY (vehicle_no)
        REFERENCES trucks(vehicle_no),
    CONSTRAINT fk_records_container FOREIGN KEY (container_no)
        REFERENCES containers(container_no),
    CONSTRAINT fk_records_location FOREIGN KEY (location_id)
        REFERENCES loading_locations(location_id),
    CONSTRAINT chk_loading_status CHECK (status IN ('PENDING', 'IN_PROGRESS', 'COMPLETED', 'CANCELED'))
);

CREATE TABLE gates (

    CONSTRAINT chk_seatbelt_status CHECK (seatbelt_status IN ('PASS', 'FAIL', 'NA')),
    CONSTRAINT chk_fire_ext_status CHECK (fire_extinguisher_status IN ('PASS', 'FAIL', 'NA')),

-- 거점: 항만 / HUB / 물류센터
-- [테이블 역할] 항만·HUB·물류센터 등 물류 거점의 기본 정보와 위치 관리
CREATE TABLE IF NOT EXISTS logistics_centers (
    center_id BIGSERIAL PRIMARY KEY,  -- 물류 거점 고유 ID (PK)
    center_code VARCHAR(50) NOT NULL UNIQUE,  -- 거점 식별 코드 (중복 불가)
    center_name VARCHAR(100) NOT NULL,  -- 거점 이름
    center_type VARCHAR(20) NOT NULL CHECK (center_type IN ('PORT','HUB','DISTRIBUTION_CENTER')),  -- 거점 종류: PORT=항만, HUB=허브, DISTRIBUTION_CENTER=물류센터
    address TEXT,  -- 거점 주소
    latitude NUMERIC(10,7),  -- 거점 위도
    longitude NUMERIC(10,7),  -- 거점 경도
    is_active BOOLEAN NOT NULL DEFAULT TRUE,  -- 거점 운영 여부
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP  -- 거점 등록 일시
);

-- 운송 건: 최초 항만에서 시작하여 최종 물류센터까지 추적
-- [테이블 역할] 항만부터 최종 물류센터까지 하나의 운송 건을 추적
CREATE TABLE IF NOT EXISTS shipments (
    shipment_id BIGSERIAL PRIMARY KEY,  -- 운송 건 고유 ID (PK)
    shipment_code VARCHAR(60) NOT NULL UNIQUE,  -- 운송 건 식별 코드
    container_no VARCHAR(11) REFERENCES containers(container_no) ON DELETE SET NULL,  -- 관련 컨테이너 번호 (FK, 선택)
    origin_center_id BIGINT NOT NULL REFERENCES logistics_centers(center_id),  -- 최초 출발 거점 ID (FK)
    destination_center_id BIGINT REFERENCES logistics_centers(center_id),  -- 운송 건 단위 최종 목적지 (FK, 선택). 목적지가 여러 곳이면 NULL, 물품별 목적지는 items.destination_center_id
    status VARCHAR(30) NOT NULL DEFAULT 'READY' CHECK (status IN  -- 운송 상태: 준비/허브 이동/허브 도착/분류/센터 이동/부분 배송/완료/보류/취소
      ('READY','TO_HUB','AT_HUB','SORTING','TO_CENTER','PARTIALLY_DELIVERED','DELIVERED','ON_HOLD','CANCELED')),
    started_at TIMESTAMPTZ,  -- 운송 시작 일시
    completed_at TIMESTAMPTZ,  -- 운송 완료 일시
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 등록 일시
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP  -- 최종 수정 일시
);

-- 실제 운송 차량은 구간별로 배정 관리 (항만→HUB: 컨테이너 차량, HUB→물류센터: 일반 화물차)
-- HUB에서 차량이 변경되면 새로운 운송 구간 배정 기록을 생성
-- 물류센터 이후 추가 운송이 없다면 별도 구간 배정 기록은 생성하지 않음
-- [테이블 역할] 운송 건의 구간별 차량 배정, 출발·도착 일시 및 운송 상태 관리 (shipment_legs = 운송 구간)
CREATE TABLE IF NOT EXISTS shipment_legs (
    assignment_id BIGSERIAL PRIMARY KEY,  -- 운송 구간 고유 ID (PK)
    shipment_id BIGINT NOT NULL REFERENCES shipments(shipment_id),  -- 초기 정의. 최종은 항만→허브 필수, 허브→센터 NULL 허용
    vehicle_no VARCHAR(20) NOT NULL REFERENCES trucks(vehicle_no),  -- 배정 차량번호 (FK)
    from_center_id BIGINT NOT NULL REFERENCES logistics_centers(center_id),  -- 해당 구간 출발 거점 ID (FK)
    to_center_id BIGINT NOT NULL REFERENCES logistics_centers(center_id),  -- 해당 구간 도착 거점 ID (FK)
    leg_sequence INT NOT NULL DEFAULT 1 CHECK (leg_sequence > 0),  -- 항만 구간의 원본 운송 건 내 순서(1부터). 허브 출고는 shipment_id NULL로 순서 유일성 대상에서 제외
    leg_type VARCHAR(30) NOT NULL CHECK (leg_type IN ('PORT_TO_HUB','HUB_TO_CENTER','OTHER')),  -- 구간 종류: 항만→HUB / HUB→센터 / 기타
    status VARCHAR(20) NOT NULL DEFAULT 'ASSIGNED' CHECK (status IN ('ASSIGNED','IN_TRANSIT','ARRIVED','CANCELED')),  -- 차량 배정 및 운송 상태
    departed_at TIMESTAMPTZ,  -- 구간 출발 일시
    arrived_at TIMESTAMPTZ,  -- 구간 도착 일시
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 배정 등록 일시
    CHECK (from_center_id <> to_center_id),
    UNIQUE (assignment_id, shipment_id)
);

-- 실제 택배 물품 단위: 컨테이너에서 분류되어 개별 목적지로 이동
-- [테이블 역할] 컨테이너에 담긴 개별 물품의 바코드·목적지·배송 상태 관리
CREATE TABLE IF NOT EXISTS items (
    item_id BIGSERIAL PRIMARY KEY,  -- 물품 고유 ID (PK)
    tracking_no VARCHAR(80) NOT NULL UNIQUE,  -- 운송장 번호 (중복 불가)
    barcode VARCHAR(100) NOT NULL UNIQUE, -- HUB 하역/분류/적재 시 스캔하는 현장용 물품 바코드 (중복 불가)
    shipment_id BIGINT NOT NULL REFERENCES shipments(shipment_id),  -- 원본 운송 건 ID (FK)
    container_no VARCHAR(11) REFERENCES containers(container_no) ON DELETE SET NULL,  -- 원본 컨테이너 번호(FK, 선택). 값이 있으면 원본 운송 건 컨테이너와 지연 FK로 일치 검사
    item_name VARCHAR(150) NOT NULL,  -- 물품명
    quantity INT NOT NULL DEFAULT 1 CHECK (quantity > 0),  -- 물품 수량 (1 이상)
    weight_kg NUMERIC(10,3) NOT NULL CHECK (weight_kg >= 0),  -- 물품 중량 (kg)
    destination_center_id BIGINT NOT NULL REFERENCES logistics_centers(center_id),  -- 물품 최종 목적지 거점 ID (FK)
    status VARCHAR(30) NOT NULL DEFAULT 'IN_CONTAINER' CHECK (status IN  -- 물품 현재 상태 (컨테이너/허브/분류/적재/운송/배송 등)
      ('IN_CONTAINER','AT_HUB','SORTING','LOADED','IN_TRANSIT','DELIVERED','ON_HOLD','LOST','DAMAGED')),
    delivered_at TIMESTAMPTZ,  -- 배송 완료 일시
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 물품 등록 일시
    UNIQUE (item_id, shipment_id)
);

-- 택배 물품이 실제로 어떤 차량 구간에 배정되었는지 연결 (분할 배송 가능)
-- [테이블 역할] 개별 물품과 운송 구간의 차량 배정을 연결해 환적·분할 배송 추적
CREATE TABLE IF NOT EXISTS item_vehicle_assignments (
    item_assignment_id BIGSERIAL PRIMARY KEY,  -- 물품-차량 배정 고유 ID (PK)
    item_id BIGINT NOT NULL,  -- 배정 물품 ID (복합 FK 구성)
    shipment_id BIGINT NOT NULL,  -- 물품의 원본 운송 건 ID. (item_id, shipment_id)로 소속 일치 확인
    assignment_id BIGINT NOT NULL,  -- 구간 ID. 최종은 단독 FK이며 항만 원본 운송 건 일치는 트리거로 검사
    loaded_at TIMESTAMPTZ,  -- 차량에 물품 적재한 일시
    unloaded_at TIMESTAMPTZ,  -- 차량에서 물품 하역한 일시
    status VARCHAR(20) NOT NULL DEFAULT 'ASSIGNED' CHECK (status IN ('ASSIGNED','LOADED','UNLOADED','CANCELED')),  -- 물품 배정 상태: 배정/적재/하역/취소
    FOREIGN KEY (item_id, shipment_id) REFERENCES items(item_id, shipment_id),
    FOREIGN KEY (assignment_id, shipment_id) REFERENCES shipment_legs(assignment_id, shipment_id),
    UNIQUE (item_id, assignment_id)
);

-- HUB 하역/분류/재적재 이력 (항만 loading_records는 그대로 유지)
-- [테이블 역할] 컨테이너 하역(CONTAINER_UNLOAD)·물품 하역(UNLOAD)·분류(SORT)·적재(LOAD) 작업 이력
CREATE TABLE IF NOT EXISTS item_history (
    handling_id BIGSERIAL PRIMARY KEY,  -- 물품 작업 이력 고유 ID (PK)
    shipment_id BIGINT NOT NULL,  -- 관련 운송 건 ID (FK)
    item_id BIGINT,  -- 작업 대상 물품 ID (FK, 선택)
    center_id BIGINT NOT NULL REFERENCES logistics_centers(center_id),  -- 작업 수행 물류 거점 ID (FK)
    assignment_id BIGINT,  -- 구간 ID(선택). 최종은 단독 FK이며 항만 원본 운송 건 일치는 트리거로 검사
    handling_type VARCHAR(25) NOT NULL CHECK (handling_type IN ('CONTAINER_UNLOAD','UNLOAD','SORT','LOAD')),  -- 작업 유형: 컨테이너 하역/일반 하역/분류/적재
    status VARCHAR(20) NOT NULL DEFAULT 'PENDING' CHECK (status IN ('PENDING','IN_PROGRESS','COMPLETED','CANCELED')),  -- 작업 상태: 대기/진행/완료/취소
    started_at TIMESTAMPTZ,  -- 작업 시작 일시
    completed_at TIMESTAMPTZ,  -- 작업 완료 일시
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 작업 기록 생성 일시
    FOREIGN KEY (shipment_id) REFERENCES shipments(shipment_id),
    FOREIGN KEY (item_id, shipment_id) REFERENCES items(item_id, shipment_id),
    FOREIGN KEY (assignment_id, shipment_id) REFERENCES shipment_legs(assignment_id, shipment_id)
);

-- 톨게이트 기본 정보 및 시뮬레이션용 고정 통행료
-- [테이블 역할] 톨게이트 위치와 고정 요금 관리
CREATE TABLE IF NOT EXISTS tollgates (
    tollgate_id BIGSERIAL PRIMARY KEY,  -- 톨게이트 고유 ID
    tollgate_code VARCHAR(50) NOT NULL UNIQUE,  -- 톨게이트 코드
    tollgate_name VARCHAR(100) NOT NULL,  -- 톨게이트 이름
    road_name VARCHAR(100),  -- 소속 고속도로 이름
    latitude NUMERIC(10,7),  -- 위도
    longitude NUMERIC(10,7),  -- 경도
    toll_fee INT NOT NULL DEFAULT 0 CHECK (toll_fee >= 0),  -- 가상 고정 통행료(원)
    is_active BOOLEAN NOT NULL DEFAULT TRUE,  -- 운영 여부
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP  -- 등록 일시
);

-- 과적검사소 기본 정보
-- [테이블 역할] 톨게이트 연계 또는 독립 과적검사소 관리
CREATE TABLE IF NOT EXISTS weigh_stations (
    station_id BIGSERIAL PRIMARY KEY,  -- 검사소 고유 ID
    station_code VARCHAR(50) NOT NULL UNIQUE,  -- 검사소 코드
    station_name VARCHAR(100) NOT NULL,  -- 검사소 이름
    address TEXT,  -- 검사소 주소
    tollgate_id BIGINT REFERENCES tollgates(tollgate_id) ON DELETE SET NULL,  -- NULL이면 독립 검사소
    road_name VARCHAR(100),  -- 소속 도로
    latitude NUMERIC(10,7),  -- 위도
    longitude NUMERIC(10,7),  -- 경도
    is_active BOOLEAN NOT NULL DEFAULT TRUE  -- 운영 여부
);

-- 톨게이트 통행료 결제 기록
-- [테이블 역할] 정상 통과 차량의 가상 고정 통행료 결제 이력 관리
CREATE TABLE IF NOT EXISTS toll_payments (
    payment_id BIGSERIAL PRIMARY KEY,  -- 결제 기록 고유 ID
    tollgate_id BIGINT NOT NULL REFERENCES tollgates(tollgate_id),  -- 통과 톨게이트 ID
    entry_tollgate_id BIGINT REFERENCES tollgates(tollgate_id),  -- 진입 톨게이트 ID (선택)
    vehicle_no VARCHAR(20) NOT NULL REFERENCES trucks(vehicle_no),  -- 등록 차량번호
    assignment_id BIGINT REFERENCES shipment_legs(assignment_id),  -- 운송 구간 ID
    amount NUMERIC(12,2) NOT NULL CHECK (amount >= 0),  -- 결제 시점의 고정 통행료
    payment_method VARCHAR(20) NOT NULL CHECK (payment_method IN ('HI_PASS','CARD','CASH','OTHER')),  -- 결제 수단
    payment_status VARCHAR(20) NOT NULL DEFAULT 'PENDING'
        CHECK (payment_status IN ('PENDING','PAID','FAILED','CANCELED','REFUNDED')),  -- 결제 상태
    passed_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 통과 시각
    paid_at TIMESTAMPTZ,  -- 결제 완료 시각
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP  -- 기록 생성 시각
);

-- 톨게이트 차량 인식 및 통과 판정 이력
-- [테이블 역할] OCR 결과, 등록 여부, 과적검사 및 통과 결과 관리
CREATE TABLE IF NOT EXISTS toll_passages (
    passage_id BIGSERIAL PRIMARY KEY,  -- 통과 시도 고유 ID
    tollgate_id BIGINT NOT NULL REFERENCES tollgates(tollgate_id),  -- 대상 톨게이트
    recognized_plate_no VARCHAR(20),  -- OCR 인식 차량번호
    vehicle_no VARCHAR(20) REFERENCES trucks(vehicle_no),  -- 등록 차량번호 (미등록 시 NULL)
    assignment_id BIGINT REFERENCES shipment_legs(assignment_id),  -- 운송 구간
    check_id BIGINT REFERENCES overload_checks(check_id),  -- 과적검사 결과 (미검사 시 NULL)
    recognition_status VARCHAR(20) NOT NULL
        CHECK (recognition_status IN ('SUCCESS','FAILED')),  -- OCR 인식 결과
    passage_status VARCHAR(20) NOT NULL DEFAULT 'PENDING'
        CHECK (passage_status IN ('PENDING','PASSED','BLOCKED')),  -- 통과 대기/성공/차단
    block_reason VARCHAR(30)
        CHECK (block_reason IN ('OCR_FAILED','UNREGISTERED','OVERLOAD','OTHER')),  -- 차단 사유
    detected_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- 차량 감지 시각
    passed_at TIMESTAMPTZ,  -- 실제 통과 시각 (차단 시 NULL)
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT chk_toll_passage_result CHECK (
        (passage_status = 'PASSED' AND passed_at IS NOT NULL AND block_reason IS NULL)
        OR (passage_status = 'BLOCKED' AND passed_at IS NULL AND block_reason IS NOT NULL)
        OR (passage_status = 'PENDING' AND passed_at IS NULL AND block_reason IS NULL)
    )
);

ALTER TABLE trucks ADD COLUMN IF NOT EXISTS gross_vehicle_weight_kg NUMERIC(10,2); -- 초기 임시 차량 총중량(kg). 아래 단계에서 제원 테이블로 단일화

ALTER TABLE overload_checks ADD COLUMN IF NOT EXISTS station_id BIGINT; -- 검사를 수행한 과적 검사소 ID (FK)

    FOREIGN KEY (shipment_id) REFERENCES shipments(shipment_id);

ALTER TABLE overload_checks ADD CONSTRAINT chk_overload_detection_status
    CHECK (detection_status IS NULL OR detection_status IN ('NORMAL','SUSPECTED','CONFIRMED'));

ALTER TABLE gates ADD COLUMN IF NOT EXISTS center_id BIGINT REFERENCES logistics_centers(center_id); -- 게이트가 속한 물류 거점 ID (FK)

ALTER TABLE gate_logs ADD CONSTRAINT fk_gate_log_assignment

-- ==================================================
-- HUB · AMR 디지털 트윈 · 예지보전 테이블 15개
-- ==================================================
-- ==========================================
-- 13-1. hubs 테이블 (허브 마스터)
-- ==========================================
CREATE TABLE hubs (
    source_id VARCHAR(50) UNIQUE,
    hub_type VARCHAR(20) NOT NULL DEFAULT '물류창고',
    jibun_address TEXT,
    total_area_m2 NUMERIC(12,2),
    cold_storage_area_m2 NUMERIC(12,2),
    size_class VARCHAR(10) CHECK (size_class IN ('소형','중형','대형')),
    is_mega BOOLEAN NOT NULL DEFAULT FALSE,
    geofence_radius_m INTEGER NOT NULL DEFAULT 300,

    hub_id BIGSERIAL PRIMARY KEY,                              -- 허브 고유 ID
    center_id BIGINT NOT NULL UNIQUE REFERENCES logistics_centers(center_id), -- 물류 거점과 1:1 연결
    hub_code VARCHAR(30) NOT NULL UNIQUE,                      -- 허브 코드 (예: HUB-GONJIAM-01)
    hub_name VARCHAR(100) NOT NULL,                            -- 허브 이름
    company_id BIGINT,                                         -- 운영 업체 ID
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_hubs_company FOREIGN KEY (company_id)
        REFERENCES companies(company_id) ON DELETE SET NULL
);

-- ==========================================
-- 13-2. delivery_regions 테이블 (출고 권역: 인천 · 서울 · 경기)
-- ==========================================
CREATE TABLE delivery_regions (
    region_id BIGSERIAL PRIMARY KEY,                           -- 권역 고유 ID
    region_code VARCHAR(10) NOT NULL UNIQUE,                   -- 권역 코드 (ICN, SEL, GGI)
    region_name VARCHAR(30) NOT NULL,                          -- 권역 이름 (인천, 서울, 경기)
    color_hex VARCHAR(7),                                      -- 3D/화면 표시 색
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- ==========================================
-- 13-3. hub_zones 테이블 (허브 내부 보관 구역: A 인천 · B 서울 · C 경기 · D 냉장)
-- ==========================================
CREATE TABLE hub_zones (
    zone_id BIGSERIAL PRIMARY KEY,                             -- 구역 고유 ID
    hub_id BIGINT NOT NULL,                                    -- 소속 허브
    zone_tag VARCHAR(5) NOT NULL,                              -- 구역 기호 (A, B, C, D)
    zone_name VARCHAR(30) NOT NULL,                            -- 구역 이름
    region_id BIGINT,                                          -- 이 구역 화물의 출고 권역 (냉장은 NULL)
    is_cold BOOLEAN NOT NULL DEFAULT FALSE,                    -- 냉장 · 냉동 창고 여부
    rack_row INT NOT NULL,                                     -- 시뮬레이션 랙 줄 번호 (0~3)

    CONSTRAINT fk_zones_hub FOREIGN KEY (hub_id) REFERENCES hubs(hub_id) ON DELETE CASCADE,
    CONSTRAINT fk_zones_region FOREIGN KEY (region_id) REFERENCES delivery_regions(region_id),
    CONSTRAINT uk_zones_id_hub UNIQUE (zone_id, hub_id),
    CONSTRAINT uk_zones_hub_tag UNIQUE (hub_id, zone_tag)
);

-- ==========================================
-- 13-4. hub_rack_slots 테이블 (랙 칸: 0단 = AMR 바닥 칸, 1~3단 = 무인 지게차)
-- ==========================================
CREATE TABLE hub_rack_slots (
    slot_id BIGSERIAL PRIMARY KEY,                             -- 랙 칸 고유 ID
    hub_id BIGINT NOT NULL, -- 복합 FK로 구역 및 박스와 허브 일치 보장
    zone_id BIGINT NOT NULL,                                   -- 소속 구역
    side_no INT NOT NULL,                                      -- 랙 면 (0: 북쪽, 1: 남쪽)
    slot_index INT NOT NULL,                                   -- 칸 번호 (0~8)
    level_no INT NOT NULL DEFAULT 0,                           -- 단 (0~3)
    slot_label VARCHAR(20) NOT NULL,                           -- 표시용 라벨 (예: A-03-0면-0단)
    status VARCHAR(20) NOT NULL DEFAULT 'EMPTY',               -- 칸 상태
    pos_x NUMERIC(7, 2),                                       -- 평면도 X (1000 x 640 기준)
    pos_y NUMERIC(7, 2),                                       -- 평면도 Y
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_slots_zone FOREIGN KEY (zone_id, hub_id) REFERENCES hub_zones(zone_id, hub_id) ON DELETE CASCADE,
    CONSTRAINT uk_slots_id_hub UNIQUE (slot_id, hub_id),
    CONSTRAINT uk_slots_position UNIQUE (zone_id, side_no, slot_index, level_no),
    CONSTRAINT chk_slots_level CHECK (level_no BETWEEN 0 AND 3),
    CONSTRAINT chk_slots_status CHECK (status IN ('EMPTY', 'OCCUPIED', 'RESERVED_IN', 'RESERVED_OUT', 'DISABLED'))
);

-- ==========================================
-- 13-5. hub_docks 테이블 (입고 · 출고 도크)
-- ==========================================
CREATE TABLE hub_docks (
    dock_id BIGSERIAL PRIMARY KEY,                             -- 도크 고유 ID
    hub_id BIGINT NOT NULL,                                    -- 소속 허브
    dock_code VARCHAR(10) NOT NULL,                            -- 도크 코드 (D1~D6)
    dock_type VARCHAR(10) NOT NULL,                            -- 'IN' 입고 / 'OUT' 출고
    region_id BIGINT,                                          -- 출고 도크의 권역 (입고 도크는 NULL)
    status VARCHAR(20) NOT NULL DEFAULT 'EMPTY',               -- 도크 상태
    capacity INT NOT NULL DEFAULT 4,                           -- 택배차 1대 적재 건수
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_docks_hub FOREIGN KEY (hub_id) REFERENCES hubs(hub_id) ON DELETE CASCADE,
    CONSTRAINT fk_docks_region FOREIGN KEY (region_id) REFERENCES delivery_regions(region_id),
    CONSTRAINT uk_docks_id_hub UNIQUE (dock_id, hub_id),
    CONSTRAINT uk_docks_hub_code UNIQUE (hub_id, dock_code),
    CONSTRAINT chk_docks_type CHECK (dock_type IN ('IN', 'OUT')),
    CONSTRAINT chk_docks_status CHECK (status IN ('EMPTY', 'ARRIVING', 'DOCKED', 'LEAVING'))
);

-- ==========================================
-- 13-6. hub_inbound_receipts 테이블 (화물차 허브 도착 · 하차)
-- ==========================================
CREATE TABLE hub_inbound_receipts (
    receipt_id BIGSERIAL PRIMARY KEY,                          -- 입고 접수 ID
    assignment_id BIGINT NOT NULL REFERENCES shipment_legs(assignment_id), -- 입고/출고 운송 구간
    hub_id BIGINT NOT NULL,                                    -- 도착 허브
    dock_id BIGINT,                                            -- 하차 도크
    vehicle_no VARCHAR(20),                                    -- 화물차 번호판 (1차 trucks)
    container_no VARCHAR(11),                                  -- 컨테이너 번호 (1차 containers)
    parcel_count INT NOT NULL DEFAULT 0,                       -- 하차 화물 건수
    status VARCHAR(20) NOT NULL DEFAULT 'ARRIVED',             -- 접수 상태
    arrived_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    unloaded_at TIMESTAMP WITH TIME ZONE,                      -- 하차 완료 시각

    CONSTRAINT uk_hub_inbound_receipts_id_hub UNIQUE (receipt_id, hub_id),
    CONSTRAINT fk_receipts_hub FOREIGN KEY (hub_id) REFERENCES hubs(hub_id),
    CONSTRAINT fk_receipts_dock FOREIGN KEY (dock_id, hub_id) REFERENCES hub_docks(dock_id, hub_id),
    CONSTRAINT fk_receipts_truck FOREIGN KEY (vehicle_no) REFERENCES trucks(vehicle_no) ON DELETE RESTRICT,  -- 입고 접수가 참조하는 차량은 삭제 불가 (SET NULL이면 운송 구간 불일치 트리거 오류로 메시지가 혼동됨)
    CONSTRAINT fk_receipts_container FOREIGN KEY (container_no) REFERENCES containers(container_no) ON DELETE SET NULL,
    CONSTRAINT chk_receipts_status CHECK (status IN ('ARRIVED', 'UNLOADING', 'COMPLETED'))
);

-- ==========================================
-- 13-7. hub_dispatches 테이블 (권역행 택배차 출고)
-- ==========================================
CREATE TABLE hub_dispatches (
    dispatch_id BIGSERIAL PRIMARY KEY,                         -- 출고 ID
    assignment_id BIGINT NOT NULL REFERENCES shipment_legs(assignment_id), -- 입고/출고 운송 구간
    hub_id BIGINT NOT NULL,                                    -- 출발 허브
    dock_id BIGINT,                                            -- 출고 도크
    region_id BIGINT NOT NULL,                                 -- 도착 권역
    delivery_vehicle_no VARCHAR(20),                           -- 택배차 번호판
    parcel_count INT NOT NULL DEFAULT 0,                       -- 적재 건수
    total_weight_kg NUMERIC(8, 1) NOT NULL DEFAULT 0,          -- 총 중량 (kg)
    departed_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT uk_hub_dispatches_id_hub UNIQUE (dispatch_id, hub_id),
    CONSTRAINT fk_dispatches_hub FOREIGN KEY (hub_id) REFERENCES hubs(hub_id),
    CONSTRAINT fk_dispatches_dock FOREIGN KEY (dock_id, hub_id) REFERENCES hub_docks(dock_id, hub_id),
    CONSTRAINT fk_dispatches_region FOREIGN KEY (region_id) REFERENCES delivery_regions(region_id)
);

-- ==========================================
-- 13-8. parcels 테이블 (물품별 허브 박스 정보; 운송장 번호는 items에서 관리)
-- ==========================================
CREATE TABLE parcels (
    parcel_no VARCHAR(20) PRIMARY KEY, -- 내부 박스 ID (운송장은 items.tracking_no)
    item_id BIGINT NOT NULL UNIQUE, -- 박스에 물품 연결 필수. 물품당 박스 정보는 최대 1개
    hub_id BIGINT NOT NULL,                                    -- 현재 허브
    receipt_id BIGINT,                                         -- 입고 접수 (어느 화물차로 왔는지)
    region_id BIGINT,                                          -- 목적지 권역 (OCR 결과)
    category VARCHAR(30),                                      -- 품목 (전자제품, 식품, 의류 ...)
    size_class CHAR(1),                                        -- 크기 S / M / L
    handling VARCHAR(20) NOT NULL DEFAULT '일반',               -- 취급 표시
    status VARCHAR(20) NOT NULL DEFAULT 'IN_TRUCK',            -- 화물 상태
    slot_id BIGINT,                                            -- 현재 보관 랙 칸
    dispatch_id BIGINT,                                        -- 출고된 택배차
    inbound_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    stored_at TIMESTAMP WITH TIME ZONE,                        -- 랙 보관 시각
    outbound_at TIMESTAMP WITH TIME ZONE,                      -- 출고 시각

    CONSTRAINT fk_parcels_item FOREIGN KEY (item_id) REFERENCES items(item_id),
    CONSTRAINT fk_parcels_hub FOREIGN KEY (hub_id) REFERENCES hubs(hub_id),
    CONSTRAINT fk_parcels_receipt FOREIGN KEY (receipt_id, hub_id) REFERENCES hub_inbound_receipts(receipt_id, hub_id),
    CONSTRAINT fk_parcels_region FOREIGN KEY (region_id) REFERENCES delivery_regions(region_id),
    CONSTRAINT fk_parcels_slot FOREIGN KEY (slot_id, hub_id) REFERENCES hub_rack_slots(slot_id, hub_id),
    CONSTRAINT fk_parcels_dispatch FOREIGN KEY (dispatch_id, hub_id) REFERENCES hub_dispatches(dispatch_id, hub_id),
    CONSTRAINT chk_parcels_size CHECK (size_class IN ('S', 'M', 'L')),
    CONSTRAINT chk_parcels_handling CHECK (handling IN ('일반', '파손주의', '취급주의', '냉장', '중량물')),
    CONSTRAINT chk_parcels_status CHECK (status IN ('IN_TRUCK', 'MOVING', 'STORED', 'OUTBOUND_READY', 'SHIPPED'))
);

-- ==========================================
-- 13-9. waybill_scans 테이블 (입고 도크 운송장 OCR 기록)
-- ==========================================
CREATE TABLE waybill_scans (
    scan_id BIGSERIAL PRIMARY KEY,                             -- 스캔 ID
    parcel_no VARCHAR(20),                                     -- 매칭된 운송장
    dock_id BIGINT,                                            -- 스캔한 도크
    recognized_text VARCHAR(100),                              -- OCR 원문
    region_id BIGINT,                                          -- 인식된 목적지 권역
    confidence NUMERIC(5, 2),                                  -- 신뢰도 (0.00 ~ 100.00%)
    is_success BOOLEAN NOT NULL DEFAULT TRUE,                  -- 인식 성공 여부
    image_url TEXT,                                            -- 스캔 이미지
    scanned_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_scans_parcel FOREIGN KEY (parcel_no) REFERENCES parcels(parcel_no) ON DELETE SET NULL,
    CONSTRAINT fk_scans_dock FOREIGN KEY (dock_id) REFERENCES hub_docks(dock_id) ON DELETE SET NULL,
    CONSTRAINT fk_scans_region FOREIGN KEY (region_id) REFERENCES delivery_regions(region_id)
);

-- ==========================================
-- 13-10. amr_robots 테이블 (AMR · 무인 지게차 마스터 + 현재 상태)
-- ==========================================
CREATE TABLE amr_robots (
    robot_id BIGSERIAL PRIMARY KEY,                            -- 로봇 고유 ID
    hub_id BIGINT NOT NULL,                                    -- 배치 허브
    robot_code VARCHAR(20) NOT NULL UNIQUE,                    -- 코드 (AMR-01, FL-01)
    robot_type VARCHAR(20) NOT NULL DEFAULT 'AMR',             -- 'AMR' / 'FORKLIFT'
    robot_role VARCHAR(20) NOT NULL DEFAULT 'ACTIVE',          -- 'ACTIVE' 가동 / 'SPARE' 예비
    state VARCHAR(20) NOT NULL DEFAULT 'IDLE',                 -- 현재 작업 상태
    battery_pct NUMERIC(5, 2),                                 -- 배터리 (%)
    health_score NUMERIC(5, 2),                                -- 건강 점수 (0~100)
    cum_load_kg NUMERIC(10, 1) NOT NULL DEFAULT 0,             -- 정비 후 누적 운반 중량 (kg)
    home_x NUMERIC(7, 2),                                      -- 충전 · 대기 위치 X
    home_y NUMERIC(7, 2),                                      -- 충전 · 대기 위치 Y
    max_payload_kg NUMERIC(7, 1),                              -- 최대 적재 중량
    installed_at DATE,                                         -- 도입일
    last_maintained_at TIMESTAMP WITH TIME ZONE,               -- 마지막 정비
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_robots_hub FOREIGN KEY (hub_id) REFERENCES hubs(hub_id),
    CONSTRAINT chk_robots_type CHECK (robot_type IN ('AMR', 'FORKLIFT')),
    CONSTRAINT chk_robots_role CHECK (robot_role IN ('ACTIVE', 'SPARE')),
    CONSTRAINT chk_robots_state CHECK (state IN ('IDLE', 'TO_PICK', 'LOADING', 'TO_DROP', 'UNLOADING',
                                                 'TO_MAINT', 'MAINT', 'TO_HOME', 'TO_RACK', 'FORK_WORK', 'CHARGING'))
);

-- ==========================================
-- 13-11. amr_telemetry 테이블 (센서 시계열: 진동 · 온도 · 배터리 · 위치)
-- ==========================================
CREATE TABLE amr_telemetry (
    telemetry_id BIGSERIAL PRIMARY KEY,                        -- 측정 ID
    robot_id BIGINT NOT NULL,                                  -- 로봇
    vibration_mm_s NUMERIC(5, 2),                              -- 구동모터 진동 (mm/s, 정상 1.2~2.0)
    motor_temp_c NUMERIC(5, 2),                                -- 모터 온도 (℃)
    battery_pct NUMERIC(5, 2),                                 -- 배터리 (%)
    health_score NUMERIC(5, 2),                                -- 건강 점수
    cum_load_kg NUMERIC(10, 1),                                -- 누적 운반 중량
    speed_m_s NUMERIC(5, 2),                                   -- 속도
    pos_x NUMERIC(7, 2),                                       -- 평면도 X
    pos_y NUMERIC(7, 2),                                       -- 평면도 Y
    state VARCHAR(20),                                         -- 측정 시점 상태
    recorded_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_telemetry_robot FOREIGN KEY (robot_id) REFERENCES amr_robots(robot_id) ON DELETE CASCADE
);

-- ==========================================
-- 13-12. amr_tasks 테이블 (작업 지시: 입고 · 출고 · 보충 · 적치)
-- ==========================================
CREATE TABLE amr_tasks (
    task_id BIGSERIAL PRIMARY KEY,                             -- 작업 ID
    robot_id BIGINT,                                           -- 담당 로봇
    task_type VARCHAR(20) NOT NULL,                            -- 작업 종류
    parcel_no VARCHAR(20),                                     -- 대상 화물
    from_dock_id BIGINT,                                       -- 출발 도크 (입고)
    to_dock_id BIGINT,                                         -- 도착 도크 (출고)
    from_slot_id BIGINT,                                       -- 출발 랙 칸
    to_slot_id BIGINT,                                         -- 도착 랙 칸
    status VARCHAR(20) NOT NULL DEFAULT 'ASSIGNED',            -- 작업 상태
    assigned_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    started_at TIMESTAMP WITH TIME ZONE,
    completed_at TIMESTAMP WITH TIME ZONE,

    CONSTRAINT fk_tasks_robot FOREIGN KEY (robot_id) REFERENCES amr_robots(robot_id) ON DELETE SET NULL,
    CONSTRAINT fk_tasks_parcel FOREIGN KEY (parcel_no) REFERENCES parcels(parcel_no) ON DELETE SET NULL,
    CONSTRAINT fk_tasks_from_dock FOREIGN KEY (from_dock_id) REFERENCES hub_docks(dock_id) ON DELETE SET NULL,
    CONSTRAINT fk_tasks_to_dock FOREIGN KEY (to_dock_id) REFERENCES hub_docks(dock_id) ON DELETE SET NULL,
    CONSTRAINT fk_tasks_from_slot FOREIGN KEY (from_slot_id) REFERENCES hub_rack_slots(slot_id) ON DELETE SET NULL,
    CONSTRAINT fk_tasks_to_slot FOREIGN KEY (to_slot_id) REFERENCES hub_rack_slots(slot_id) ON DELETE SET NULL,
    CONSTRAINT chk_tasks_type CHECK (task_type IN ('INBOUND', 'OUTBOUND', 'REPLENISH', 'STOW', 'MAINT_MOVE')),
    CONSTRAINT chk_tasks_status CHECK (status IN ('ASSIGNED', 'IN_PROGRESS', 'COMPLETED', 'CANCELED'))
);

-- ==========================================
-- 13-13. amr_predictions 테이블 (AI 고장 예측 결과)
-- ==========================================
CREATE TABLE amr_predictions (
    prediction_id BIGSERIAL PRIMARY KEY,                       -- 예측 ID
    robot_id BIGINT NOT NULL,                                  -- 대상 로봇
    component VARCHAR(50) NOT NULL,                            -- 부품 (구동모터 베어링 등)
    risk_level VARCHAR(10) NOT NULL,                           -- 위험도
    predicted_rul_hours NUMERIC(7, 1),                         -- 예상 잔여 수명 (시간)
    vibration_slope NUMERIC(8, 4),                             -- 진동 증가 기울기
    cum_load_kg NUMERIC(10, 1),                                -- 예측 시점 누적 운반 중량
    reason TEXT,                                               -- 예측 근거
    model_version VARCHAR(30),                                 -- 모델 버전
    is_resolved BOOLEAN NOT NULL DEFAULT FALSE,                -- 정비로 해소 여부
    predicted_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_predictions_robot FOREIGN KEY (robot_id) REFERENCES amr_robots(robot_id) ON DELETE CASCADE,
    CONSTRAINT chk_predictions_risk CHECK (risk_level IN ('LOW', 'WARN', 'CRIT'))
);

-- ==========================================
-- 13-14. amr_maintenance 테이블 (정비 이력 · 예비기 교대)
-- ==========================================
CREATE TABLE amr_maintenance (
    maint_id BIGSERIAL PRIMARY KEY,                            -- 정비 ID
    robot_id BIGINT NOT NULL,                                  -- 정비 대상 로봇
    prediction_id BIGINT,                                      -- 근거가 된 예측 (계획 정비)
    spare_robot_id BIGINT,                                     -- 대신 투입된 예비기
    maint_type VARCHAR(20) NOT NULL DEFAULT 'PLANNED',         -- 'PLANNED' 계획 / 'CORRECTIVE' 고장 후
    part_name VARCHAR(50),                                     -- 교체 부품
    status VARCHAR(20) NOT NULL DEFAULT 'SCHEDULED',           -- 진행 상태
    worker_account_id BIGINT,                                  -- 담당 정비 인력 (accounts)
    memo TEXT,
    scheduled_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    started_at TIMESTAMP WITH TIME ZONE,
    completed_at TIMESTAMP WITH TIME ZONE,

    CONSTRAINT fk_maint_robot FOREIGN KEY (robot_id) REFERENCES amr_robots(robot_id) ON DELETE CASCADE,
    CONSTRAINT fk_maint_prediction FOREIGN KEY (prediction_id) REFERENCES amr_predictions(prediction_id) ON DELETE SET NULL,
    CONSTRAINT fk_maint_spare FOREIGN KEY (spare_robot_id) REFERENCES amr_robots(robot_id) ON DELETE SET NULL,
    CONSTRAINT fk_maint_worker FOREIGN KEY (worker_account_id) REFERENCES accounts(account_id) ON DELETE SET NULL,
    CONSTRAINT chk_maint_type CHECK (maint_type IN ('PLANNED', 'CORRECTIVE')),
    CONSTRAINT chk_maint_status CHECK (status IN ('SCHEDULED', 'IN_PROGRESS', 'DONE', 'CANCELED'))
);

-- ==========================================
-- 13-15. hub_events 테이블 (관제 이벤트 로그 · 화면 알림)
-- ==========================================
CREATE TABLE hub_events (
    event_id BIGSERIAL PRIMARY KEY,                            -- 이벤트 ID
    hub_id BIGINT NOT NULL,                                    -- 허브
    robot_id BIGINT,                                           -- 관련 로봇 (없으면 NULL)
    event_level VARCHAR(10) NOT NULL DEFAULT 'INFO',           -- 등급
    phase_no INT,                                              -- 시나리오 단계 (1~5)
    message TEXT NOT NULL,                                     -- 내용
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_events_hub FOREIGN KEY (hub_id) REFERENCES hubs(hub_id) ON DELETE CASCADE,
    CONSTRAINT fk_events_robot FOREIGN KEY (robot_id) REFERENCES amr_robots(robot_id) ON DELETE SET NULL,
    CONSTRAINT chk_events_level CHECK (event_level IN ('INFO', 'OK', 'WARN', 'CRIT', 'DISPATCH'))
);

CREATE OR REPLACE FUNCTION distance_m(lat1 NUMERIC, lng1 NUMERIC, lat2 NUMERIC, lng2 NUMERIC)
RETURNS DOUBLE PRECISION LANGUAGE sql IMMUTABLE AS $$
    SELECT 6371000 * 2 * asin(sqrt(
        power(sin(radians((lat2 - lat1)::float8) / 2), 2) +
        cos(radians(lat1::float8)) * cos(radians(lat2::float8)) *
        power(sin(radians((lng2 - lng1)::float8) / 2), 2)))
$$;

CREATE OR REPLACE VIEW v_order_autofill AS
SELECT o.order_id, o.order_no, o.status,
       o.vehicle_no, o.container_no,
       c.iso_size_type_code, c.is_high_cube, c.max_gross_kg,
       t.terminal_name  AS origin_name,
       t.latitude       AS origin_lat,  t.longitude AS origin_lng,
       dest.center_name AS dest_name,
       dest.address AS dest_address,
       dest.latitude AS dest_lat, dest.longitude AS dest_lng,
       gl.pass_at       AS gate_out_at,
       g.gate_name      AS gate_out_name,
       vs.empty_weight_kg, vs.max_payload_kg, vs.axle_count, vs.chassis_height_m,
       (SELECT MIN(hr.limit_height_m) FROM height_restrictions hr
         WHERE hr.target_type = 'HUB_ENTRANCE' AND hr.hub_id = h.hub_id) AS dest_height_limit_m,
       (SELECT MIN(hr.limit_height_m) FROM height_restrictions hr
         WHERE hr.target_type = 'TERMINAL_GATE' AND hr.terminal_id = t.terminal_id) AS origin_height_limit_m
FROM transport_orders o
JOIN containers c           ON c.container_no = o.container_no
JOIN terminals  t           ON t.terminal_id  = o.origin_terminal_id
JOIN hubs       h           ON h.hub_id       = o.dest_hub_id
JOIN logistics_centers dest ON dest.center_id = h.center_id
LEFT JOIN gate_logs gl      ON gl.gate_log_id = o.gate_out_log_id
LEFT JOIN gates g           ON g.gate_id      = gl.gate_id
LEFT JOIN vehicle_specs vs  ON vs.vehicle_no  = o.vehicle_no;

-- 구역: 랙 0줄 = D 냉장, 1줄 = A 인천, 2줄 = B 서울, 3줄 = C 경기 (시뮬레이션 ZONES 순서와 동일)
INSERT INTO hub_zones (hub_id, zone_tag, zone_name, region_id, is_cold, rack_row)
SELECT h.hub_id, z.tag, z.name, r.region_id, z.cold, z.row_no
FROM hubs h
CROSS JOIN (VALUES ('D', '냉장', NULL, TRUE, 0),
                   ('A', '인천', 'ICN', FALSE, 1),
                   ('B', '서울', 'SEL', FALSE, 2),
                   ('C', '경기', 'GGI', FALSE, 3)) AS z(tag, name, region_code, cold, row_no)
LEFT JOIN delivery_regions r ON r.region_code = z.region_code
WHERE h.hub_code = 'HUB-GONJIAM-01'
ON CONFLICT (hub_id, zone_tag) DO NOTHING;

-- 랙 칸: 구역마다 2면 x 9칸 x 4단 (냉장 창고는 북쪽 1면만) → 총 252칸
-- 좌표 = 시뮬레이션 SX(i) = 262 + i*49, AISLES = [62, 175, 295, 415, 540]
INSERT INTO hub_rack_slots (hub_id, zone_id, side_no, slot_index, level_no, slot_label, pos_x, pos_y)
SELECT z.hub_id, z.zone_id, s.side_no, i.idx, lv.level_no,
       z.zone_tag || '-' || LPAD((i.idx + 1)::text, 2, '0') || '-' || s.side_no || '면-' || lv.level_no || '단',
       262 + i.idx * 49,
       (ARRAY[62, 175, 295, 415, 540])[z.rack_row + s.side_no + 1]
FROM hub_zones z
CROSS JOIN (VALUES (0), (1)) AS s(side_no)
CROSS JOIN generate_series(0, 8) AS i(idx)
CROSS JOIN generate_series(0, 3) AS lv(level_no)
JOIN hubs h ON h.hub_id = z.hub_id AND h.hub_code = 'HUB-GONJIAM-01'
WHERE NOT (z.is_cold AND s.side_no = 1)
ON CONFLICT (zone_id, side_no, slot_index, level_no) DO NOTHING;

-- 1-7. 운송 구간 검증: 구간 종류 ↔ 차량 구분 ↔ 거점 종류
--      항만→허브 = 컨테이너 특수화물차 / 허브→물류센터 = 일반화물차
CREATE OR REPLACE FUNCTION trg_check_shipment_leg() RETURNS trigger AS $$
DECLARE
    v_cat  VARCHAR;
    f_type VARCHAR;
    t_type VARCHAR;
BEGIN
    SELECT vehicle_category INTO v_cat  FROM trucks WHERE vehicle_no = NEW.vehicle_no;
    SELECT center_type      INTO f_type FROM logistics_centers WHERE center_id = NEW.from_center_id;
    SELECT center_type      INTO t_type FROM logistics_centers WHERE center_id = NEW.to_center_id;

    IF NEW.leg_type = 'PORT_TO_HUB'
       AND (v_cat IS DISTINCT FROM 'CONTAINER_TRUCK' OR f_type <> 'PORT' OR t_type <> 'HUB') THEN
        RAISE EXCEPTION 'PORT_TO_HUB 구간은 컨테이너 특수화물차(CONTAINER_TRUCK), 항만→허브여야 합니다 (vehicle_no=%)', NEW.vehicle_no;
    ELSIF NEW.leg_type = 'HUB_TO_CENTER'
       AND (v_cat IS DISTINCT FROM 'GENERAL_TRUCK' OR f_type <> 'HUB' OR t_type <> 'DISTRIBUTION_CENTER') THEN
        RAISE EXCEPTION 'HUB_TO_CENTER 구간은 일반화물차(GENERAL_TRUCK), 허브→물류센터여야 합니다 (vehicle_no=%)', NEW.vehicle_no;
    END IF;
    RETURN NEW;
END $$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_shipment_leg_check ON shipment_legs;
CREATE TRIGGER trg_shipment_leg_check
    BEFORE INSERT OR UPDATE OF vehicle_no, from_center_id, to_center_id, leg_type ON shipment_legs
    FOR EACH ROW EXECUTE FUNCTION trg_check_shipment_leg();

-- 1-8. 거점 종류 검증: 허브는 HUB 거점, 물품 목적지는 물류센터
CREATE OR REPLACE FUNCTION trg_check_hub_center() RETURNS trigger AS $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM logistics_centers
                   WHERE center_id = NEW.center_id AND center_type = 'HUB') THEN
        RAISE EXCEPTION 'hubs.center_id는 HUB 타입 거점이어야 합니다 (center_id=%)', NEW.center_id;
    END IF;
    RETURN NEW;
END $$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_hubs_center_check ON hubs;
CREATE TRIGGER trg_hubs_center_check
    BEFORE INSERT OR UPDATE OF center_id ON hubs
    FOR EACH ROW EXECUTE FUNCTION trg_check_hub_center();

CREATE OR REPLACE FUNCTION trg_check_item_destination() RETURNS trigger AS $$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM logistics_centers
                   WHERE center_id = NEW.destination_center_id AND center_type = 'DISTRIBUTION_CENTER') THEN
        RAISE EXCEPTION 'items.destination_center_id는 물류센터(DISTRIBUTION_CENTER)여야 합니다 (center_id=%)', NEW.destination_center_id;
    END IF;
    RETURN NEW;
END $$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_items_destination_check ON items;
CREATE TRIGGER trg_items_destination_check
    BEFORE INSERT OR UPDATE OF destination_center_id ON items
    FOR EACH ROW EXECUTE FUNCTION trg_check_item_destination();

-- 운송 오더의 운송 건/차량/출발지/목적지와 실제 구간이 일치하는지 검증.
CREATE OR REPLACE FUNCTION scargo_validate_order_leg() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE x RECORD;
BEGIN
 IF NEW.assignment_id IS NULL THEN
   IF NEW.status NOT IN ('CREATED','CANCELLED') AND NEW.shipment_id IS NOT NULL THEN
     RAISE EXCEPTION '배차 이후 앱 오더는 운송 구간이 필요합니다: %', NEW.order_no;
   END IF;
   RETURN NEW;
 END IF;
 IF NEW.shipment_id IS NULL THEN RAISE EXCEPTION 'assignment_id 사용 시 shipment_id 필수'; END IF;
 SELECT l.vehicle_no, l.from_center_id, l.to_center_id, l.leg_type,
        s.container_no, t.center_id AS terminal_center_id, h.center_id AS hub_center_id
 INTO x FROM shipment_legs l JOIN shipments s ON s.shipment_id=l.shipment_id
 JOIN terminals t ON t.terminal_id=NEW.origin_terminal_id
 JOIN hubs h ON h.hub_id=NEW.dest_hub_id WHERE l.assignment_id=NEW.assignment_id;
 IF NOT FOUND OR x.leg_type <> 'PORT_TO_HUB' OR NEW.vehicle_no IS DISTINCT FROM x.vehicle_no
    OR NEW.container_no IS DISTINCT FROM x.container_no
    OR x.from_center_id IS DISTINCT FROM x.terminal_center_id
    OR x.to_center_id IS DISTINCT FROM x.hub_center_id THEN
   RAISE EXCEPTION '앱 오더와 운송 구간/차량/컨테이너/거점 불일치: %', NEW.order_no;
 END IF;
 RETURN NEW;
END $$;
CREATE TRIGGER trg_orders_validate_leg BEFORE INSERT OR UPDATE OF assignment_id, shipment_id,
 vehicle_no, container_no, origin_terminal_id, dest_hub_id, status ON transport_orders
 FOR EACH ROW EXECUTE FUNCTION scargo_validate_order_leg();

-- 입고/출고 기록은 운송 구간의 실제 차량, 도착/출발 HUB, 구간 유형을 따라야 함.
CREATE OR REPLACE FUNCTION scargo_validate_hub_transport() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE x RECORD;
BEGIN
 SELECT l.vehicle_no, l.from_center_id, l.to_center_id, l.leg_type, h.center_id
 INTO x FROM shipment_legs l JOIN hubs h ON h.hub_id=NEW.hub_id
 WHERE l.assignment_id=NEW.assignment_id;
 IF NOT FOUND THEN RAISE EXCEPTION '운송 구간 또는 HUB가 없음'; END IF;
 IF TG_TABLE_NAME='hub_inbound_receipts' THEN
   IF x.leg_type <> 'PORT_TO_HUB' OR x.to_center_id <> x.center_id
      OR NEW.vehicle_no IS DISTINCT FROM x.vehicle_no THEN
     RAISE EXCEPTION 'HUB 입고 차량/도착 HUB/운송 구간 불일치';
   END IF;
 ELSE
   IF x.leg_type <> 'HUB_TO_CENTER' OR x.from_center_id <> x.center_id
      OR NEW.delivery_vehicle_no IS DISTINCT FROM x.vehicle_no THEN
     RAISE EXCEPTION 'HUB 출고 차량/출발 HUB/운송 구간 불일치';
   END IF;
   IF NOT EXISTS (SELECT 1 FROM logistics_centers c
       WHERE c.center_id=x.to_center_id AND c.region_id=NEW.region_id) THEN
     RAISE EXCEPTION '출고 권역과 운송 구간 목적지 물류센터 불일치';
   END IF;
 END IF;
 RETURN NEW;
END $$;
CREATE TRIGGER trg_hub_receipt_transport BEFORE INSERT OR UPDATE OF assignment_id, hub_id, vehicle_no
 ON hub_inbound_receipts FOR EACH ROW EXECUTE FUNCTION scargo_validate_hub_transport();
CREATE TRIGGER trg_hub_dispatch_transport BEFORE INSERT OR UPDATE OF assignment_id, hub_id, delivery_vehicle_no, region_id
 ON hub_dispatches FOR EACH ROW EXECUTE FUNCTION scargo_validate_hub_transport();

ALTER TABLE amr_tasks ADD COLUMN hub_id BIGINT REFERENCES hubs(hub_id);
UPDATE amr_tasks t SET hub_id=COALESCE(

CREATE OR REPLACE FUNCTION scargo_validate_amr_task_hub() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
 IF NEW.hub_id IS NULL THEN RAISE EXCEPTION 'AMR 작업 hub_id 필수'; END IF;
 IF NEW.robot_id IS NOT NULL AND NOT EXISTS
    (SELECT 1 FROM amr_robots WHERE robot_id=NEW.robot_id AND hub_id=NEW.hub_id) THEN
   RAISE EXCEPTION 'AMR 로봇 HUB 불일치'; END IF;
 IF NEW.parcel_no IS NOT NULL AND NOT EXISTS
    (SELECT 1 FROM parcels WHERE parcel_no=NEW.parcel_no AND hub_id=NEW.hub_id) THEN
   RAISE EXCEPTION 'AMR 물품 HUB 불일치'; END IF;
 IF NEW.from_dock_id IS NOT NULL AND NOT EXISTS
    (SELECT 1 FROM hub_docks WHERE dock_id=NEW.from_dock_id AND hub_id=NEW.hub_id) THEN
   RAISE EXCEPTION 'AMR 출발 도크 HUB 불일치'; END IF;
 IF NEW.to_dock_id IS NOT NULL AND NOT EXISTS
    (SELECT 1 FROM hub_docks WHERE dock_id=NEW.to_dock_id AND hub_id=NEW.hub_id) THEN
   RAISE EXCEPTION 'AMR 도착 도크 HUB 불일치'; END IF;
 IF NEW.from_slot_id IS NOT NULL AND NOT EXISTS
    (SELECT 1 FROM hub_rack_slots WHERE slot_id=NEW.from_slot_id AND hub_id=NEW.hub_id) THEN
   RAISE EXCEPTION 'AMR 출발 랙 HUB 불일치'; END IF;
 IF NEW.to_slot_id IS NOT NULL AND NOT EXISTS
    (SELECT 1 FROM hub_rack_slots WHERE slot_id=NEW.to_slot_id AND hub_id=NEW.hub_id) THEN
   RAISE EXCEPTION 'AMR 도착 랙 HUB 불일치'; END IF;
 RETURN NEW;
END $$;

 from_dock_id, to_dock_id, from_slot_id, to_slot_id ON amr_tasks
 FOR EACH ROW EXECUTE FUNCTION scargo_validate_amr_task_hub();

-- 개방식: passage_id(결제 대상 통과 이벤트)만 연결, 해당 톨게이트 고정 요금.
-- 폐쇄식: entry_passage_id + passage_id(진출 이벤트)로 진입/진출 구간 식별.
-- 두 방식 모두 결제액은 당시 계산한 값으로 저장; 폐쇄식 요금표/거리 계산은 서비스에서 수행.
ALTER TABLE toll_payments ADD COLUMN passage_id BIGINT UNIQUE REFERENCES toll_passages(passage_id);
ALTER TABLE toll_payments ADD COLUMN entry_passage_id BIGINT REFERENCES toll_passages(passage_id);
ALTER TABLE toll_payments ADD COLUMN toll_system VARCHAR(10) NOT NULL DEFAULT 'OPEN'
    CHECK (toll_system IN ('OPEN','CLOSED'));
ALTER TABLE toll_payments ADD CONSTRAINT chk_toll_payment_passages CHECK (
 (toll_system='OPEN' AND entry_passage_id IS NULL)
 OR (toll_system='CLOSED' AND entry_passage_id IS NOT NULL AND passage_id IS NOT NULL
     AND entry_passage_id <> passage_id)
);
CREATE OR REPLACE FUNCTION scargo_validate_toll_payment() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE p RECORD; e RECORD;
BEGIN
 IF NEW.passage_id IS NULL THEN RAISE EXCEPTION '결제에는 통과 이벤트 passage_id가 필요'; END IF;
 SELECT * INTO p FROM toll_passages WHERE passage_id=NEW.passage_id;
 IF NOT FOUND OR p.vehicle_no IS DISTINCT FROM NEW.vehicle_no OR
    p.tollgate_id IS DISTINCT FROM NEW.tollgate_id OR p.passage_status <> 'PASSED' THEN
   RAISE EXCEPTION '결제와 진출/개방식 통과 이벤트 불일치'; END IF;
 IF NEW.assignment_id IS DISTINCT FROM p.assignment_id THEN
   RAISE EXCEPTION '결제와 통과 이벤트의 운송 구간 불일치'; END IF;
 IF NEW.toll_system='CLOSED' THEN
   SELECT * INTO e FROM toll_passages WHERE passage_id=NEW.entry_passage_id;
   IF NOT FOUND OR e.vehicle_no IS DISTINCT FROM NEW.vehicle_no
      OR e.tollgate_id IS DISTINCT FROM NEW.entry_tollgate_id
      OR e.passage_status <> 'PASSED' OR e.detected_at > p.detected_at THEN
     RAISE EXCEPTION '폐쇄식 진입 이벤트와 진출 이벤트 불일치'; END IF;
 ELSE
   IF NEW.entry_tollgate_id IS NOT NULL THEN
     RAISE EXCEPTION '개방식에서는 entry_tollgate_id를 비워야 함'; END IF;
 END IF;
 RETURN NEW;
END $$;
CREATE TRIGGER trg_toll_payment_validate BEFORE INSERT OR UPDATE OF
 passage_id, entry_passage_id, toll_system, vehicle_no, tollgate_id,
 entry_tollgate_id, assignment_id ON toll_payments
 FOR EACH ROW EXECUTE FUNCTION scargo_validate_toll_payment();

-- 포함 내용
--  A. 구간 존재 단독 FK 보완 (원본 운송 건이 NULL인 허브 출고 구간 포함)

--  H. 주석 정정
--  I. 최종 검증
--
-- 주의: 이 패치는 백엔드 코드 수정이 필요한 컬럼/테이블 변경을 포함합니다.

-- 컬럼 개명 도우미 (임시 함수, 세션 종료 시 사라짐). 이미 개명했으면 건너뜁니다.
CREATE OR REPLACE FUNCTION pg_temp.rename_col(t text, o text, n text) RETURNS void
LANGUAGE plpgsql AS $f$
BEGIN
  IF EXISTS (SELECT 1 FROM information_schema.columns
             WHERE table_schema = current_schema() AND table_name = t AND column_name = o)
     AND NOT EXISTS (SELECT 1 FROM information_schema.columns
             WHERE table_schema = current_schema() AND table_name = t AND column_name = n) THEN
    EXECUTE format('ALTER TABLE %I RENAME COLUMN %I TO %I', t, o, n);
  END IF;
END $f$;

-- ============================================================
-- D. CANCELLED → CANCELED 통일 (transport_orders)
--    순서: 트리거 함수 교체 → 제약 제거 → 데이터 변경 → 제약/인덱스 재생성
-- ============================================================
CREATE OR REPLACE FUNCTION scargo_validate_order_leg() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE x RECORD;
BEGIN
 IF NEW.assignment_id IS NULL THEN
   IF NEW.status NOT IN ('CREATED','CANCELED') AND NEW.shipment_id IS NOT NULL THEN
     RAISE EXCEPTION '배차 이후 앱 오더는 운송 구간이 필요합니다: %', NEW.order_no;
   END IF;
   RETURN NEW;
 END IF;
 IF NEW.shipment_id IS NULL THEN RAISE EXCEPTION 'assignment_id 사용 시 shipment_id 필수'; END IF;
 SELECT l.vehicle_no, l.from_center_id, l.to_center_id, l.leg_type,
        s.container_no, t.center_id AS terminal_center_id, h.center_id AS hub_center_id
 INTO x FROM shipment_legs l JOIN shipments s ON s.shipment_id=l.shipment_id
 JOIN terminals t ON t.terminal_id=NEW.origin_terminal_id
 JOIN hubs h ON h.hub_id=NEW.dest_hub_id WHERE l.assignment_id=NEW.assignment_id;
 IF NOT FOUND OR x.leg_type <> 'PORT_TO_HUB' OR NEW.vehicle_no IS DISTINCT FROM x.vehicle_no
    OR NEW.container_no IS DISTINCT FROM x.container_no
    OR x.from_center_id IS DISTINCT FROM x.terminal_center_id
    OR x.to_center_id IS DISTINCT FROM x.hub_center_id THEN
   RAISE EXCEPTION '앱 오더와 운송 구간/차량/컨테이너/거점 불일치: %', NEW.order_no;
 END IF;
 RETURN NEW;
END $$;

ALTER TABLE transport_orders ADD CONSTRAINT chk_order_status CHECK (status IN

CREATE OR REPLACE FUNCTION scargo_validate_toll_payment() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE p RECORD; e RECORD;
BEGIN
 IF NEW.passage_id IS NULL THEN RAISE EXCEPTION '결제에는 통과 이벤트 passage_id가 필요'; END IF;
 SELECT * INTO p FROM toll_passages WHERE passage_id=NEW.passage_id;
 IF NOT FOUND OR p.vehicle_no IS DISTINCT FROM NEW.vehicle_no OR
    p.tollgate_id IS DISTINCT FROM NEW.tollgate_id OR p.passage_status <> 'PASSED' THEN
   RAISE EXCEPTION '결제와 진출/개방식 통과 이벤트 불일치'; END IF;
 IF NEW.assignment_id IS DISTINCT FROM p.assignment_id THEN
   RAISE EXCEPTION '결제와 통과 이벤트의 운송 구간 불일치'; END IF;
 IF NEW.toll_system='CLOSED' THEN
   SELECT * INTO e FROM toll_passages WHERE passage_id=NEW.entry_passage_id;
   IF NOT FOUND OR e.vehicle_no IS DISTINCT FROM NEW.vehicle_no
      OR e.passage_status <> 'PASSED' OR e.detected_at > p.detected_at THEN
     RAISE EXCEPTION '폐쇄식 진입 이벤트와 진출 이벤트 불일치'; END IF;
 END IF;
 RETURN NEW;
END $$;

CREATE TRIGGER trg_toll_payment_validate BEFORE INSERT OR UPDATE OF
 passage_id, entry_passage_id, toll_system, vehicle_no, tollgate_id, assignment_id ON toll_payments
 FOR EACH ROW EXECUTE FUNCTION scargo_validate_toll_payment();

-- G. 더미 데이터 정합성 수정
-- ============================================================
-- G-1. 컨테이너 종류와 야드 종류 일치: 냉동→FROZEN 야드, 일반→GENERAL 야드
--      어긋난 컨테이너만 같은 종류 야드의 위치로 순환 배치합니다.
--      (loading_location_id가 NULL인 데모 컨테이너는 건드리지 않음)
CREATE TEMP TABLE _eligible_locations ON COMMIT DROP AS
SELECT CASE WHEN y.yard_type = 'FROZEN' THEN '냉동' ELSE '일반' END AS ctype,
       l.location_id,
       row_number() OVER (PARTITION BY (y.yard_type = 'FROZEN') ORDER BY l.location_id) AS rn,
       count(*)     OVER (PARTITION BY (y.yard_type = 'FROZEN')) AS cnt
FROM loading_locations l
JOIN yards y ON y.yard_id = l.yard_id
WHERE l.is_available;

CREATE TEMP TABLE _mismatched_containers ON COMMIT DROP AS
SELECT c.container_no, c.container_type,
       row_number() OVER (PARTITION BY c.container_type ORDER BY c.container_no) AS rn
FROM containers c
JOIN loading_locations l ON l.location_id = c.loading_location_id
JOIN yards y ON y.yard_id = l.yard_id
WHERE (c.container_type = '냉동' AND y.yard_type <> 'FROZEN')
   OR (c.container_type = '일반' AND y.yard_type =  'FROZEN');

UPDATE containers c
SET loading_location_id = e.location_id
FROM _mismatched_containers m
JOIN _eligible_locations e
  ON e.ctype = m.container_type AND e.rn = ((m.rn - 1) % e.cnt) + 1
WHERE c.container_no = m.container_no;

-- G-2. 컨테이너 위치의 기준은 containers.loading_location_id. 대기(PENDING) 적재 기록을 맞춥니다.
UPDATE loading_records lr
SET location_id = c.loading_location_id
FROM containers c
WHERE c.container_no = lr.container_no
  AND lr.status = 'PENDING'
  AND c.loading_location_id IS NOT NULL
  AND lr.location_id IS DISTINCT FROM c.loading_location_id;

COMMENT ON COLUMN containers.loading_location_id IS '컨테이너 현재 야드 위치의 기준 값. loading_records.location_id는 적재 작업 시점의 기록';

-- G-3. 일반 컨테이너에 냉동식품이 적재된 모순 해소 (화물명을 상온 식품으로 변경)
UPDATE containers
SET reserved_cargo_info = jsonb_set(reserved_cargo_info, '{cargo_name}', '"식음료"')
WHERE container_type = '일반'
  AND reserved_cargo_info->>'cargo_name' LIKE '냉동%';

-- G-4. overload_checks 더미: 총중량(축중 합계)과 공차+VGM이 크게 어긋난 행(15% 초과)만
--      VGM을 총중량-공차로 역산합니다. 컨테이너 자체중량~최대총중량 범위 안일 때만 적용.
--      축중/바퀴 합계는 그대로 유지되며, is_violation / is_passed 값은 변경하지 않습니다.
DO $$
DECLARE n INT;
BEGIN
  UPDATE overload_checks o
  SET vgm_kg = o.total_weight_kg - o.empty_weight_kg
  FROM containers c
  WHERE c.container_no = o.container_no
    AND o.total_weight_kg IS NOT NULL AND o.empty_weight_kg IS NOT NULL AND o.vgm_kg IS NOT NULL
    AND ABS(o.total_weight_kg - o.empty_weight_kg - o.vgm_kg) > 0.15 * o.total_weight_kg
    AND (o.total_weight_kg - o.empty_weight_kg) BETWEEN c.tare_kg AND c.max_gross_kg;
  GET DIAGNOSTICS n = ROW_COUNT;
  RAISE NOTICE 'overload_checks VGM 역산 행 수: %', n;
END $$;

-- 4차 수정 구간 (이 파일에 통합됨): 항만 구간 운송 건 일치 검증
-- 위의 3차 수정 구간 다음에 실행됩니다.
-- 이 검증 구간은 행을 삭제하지 않습니다. 전체 파일 자체는 재생성이며, 잘못된 기존 연결이 발견되면 전체 롤백합니다.

LOCK TABLE shipment_legs, item_vehicle_assignments, item_history, gate_logs, overload_checks
    IN SHARE ROW EXCLUSIVE MODE;

-- 기존 행도 검증합니다. 이후 입력만 검사하여 과거 오류를 방치하지 않습니다.
DO $$
DECLARE t TEXT; bad BIGINT;
BEGIN
 FOREACH t IN ARRAY ARRAY['item_vehicle_assignments','item_history','gate_logs','overload_checks'] LOOP
  EXECUTE format(
   'SELECT count(*) FROM %I c LEFT JOIN shipment_legs l ON l.assignment_id=c.assignment_id
    WHERE c.assignment_id IS NOT NULL AND
     (l.assignment_id IS NULL OR
      (l.leg_type=''PORT_TO_HUB'' AND c.shipment_id IS DISTINCT FROM l.shipment_id))',t)
   INTO bad;
  IF bad > 0 THEN
   RAISE EXCEPTION '%: 존재하지 않는 구간 또는 항만 구간과 운송 건이 다른 기존 행 %건. 데이터 확인 후 다시 실행하세요.',t,bad;
  END IF;
 END LOOP;
END $$;

CREATE OR REPLACE FUNCTION scargo_check_port_leg_shipment()
RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE leg RECORD;
BEGIN
 IF NEW.assignment_id IS NULL THEN RETURN NEW; END IF;
 -- 부모 구간의 변경과 이 검증이 동시에 진행되지 않도록 공유 잠금.
 SELECT shipment_id,leg_type INTO leg FROM shipment_legs
 WHERE assignment_id=NEW.assignment_id FOR SHARE;
 IF NOT FOUND THEN
  RAISE EXCEPTION '존재하지 않는 운송 구간: %',NEW.assignment_id USING ERRCODE='23503';
 END IF;
 IF leg.leg_type='PORT_TO_HUB' AND NEW.shipment_id IS DISTINCT FROM leg.shipment_id THEN
  RAISE EXCEPTION '항만→허브 운송 건 불일치: table=%, assignment_id=%, 입력 shipment_id=%, 구간 shipment_id=%',
   TG_TABLE_NAME,NEW.assignment_id,NEW.shipment_id,leg.shipment_id USING ERRCODE='23514';
 END IF;
 -- HUB_TO_CENTER의 여러 원본 운송 건 연결은 허용.
 RETURN NEW;
END $$;

DROP TRIGGER IF EXISTS trg_item_assignment_port_shipment ON item_vehicle_assignments;
CREATE TRIGGER trg_item_assignment_port_shipment
 BEFORE INSERT OR UPDATE OF assignment_id,shipment_id ON item_vehicle_assignments
 FOR EACH ROW EXECUTE FUNCTION scargo_check_port_leg_shipment();
DROP TRIGGER IF EXISTS trg_item_history_port_shipment ON item_history;
CREATE TRIGGER trg_item_history_port_shipment
 BEFORE INSERT OR UPDATE OF assignment_id,shipment_id ON item_history
 FOR EACH ROW EXECUTE FUNCTION scargo_check_port_leg_shipment();
DROP TRIGGER IF EXISTS trg_gate_log_port_shipment ON gate_logs;
CREATE TRIGGER trg_gate_log_port_shipment
 BEFORE INSERT OR UPDATE OF assignment_id,shipment_id ON gate_logs
 FOR EACH ROW EXECUTE FUNCTION scargo_check_port_leg_shipment();
DROP TRIGGER IF EXISTS trg_overload_port_shipment ON overload_checks;
CREATE TRIGGER trg_overload_port_shipment
 BEFORE INSERT OR UPDATE OF assignment_id,shipment_id ON overload_checks
 FOR EACH ROW EXECUTE FUNCTION scargo_check_port_leg_shipment();

-- 부모 구간의 운송 건/구간 종류를 변경하여 기존 연결을 깨뜨리는 경우도 차단.
CREATE OR REPLACE FUNCTION scargo_guard_port_leg_change()
RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE t TEXT; bad BOOLEAN;
BEGIN
 IF NEW.leg_type <> 'PORT_TO_HUB' THEN RETURN NEW; END IF;
 FOREACH t IN ARRAY ARRAY['item_vehicle_assignments','item_history','gate_logs','overload_checks'] LOOP
  EXECUTE format('SELECT EXISTS(SELECT 1 FROM %I WHERE assignment_id=$1 AND shipment_id IS DISTINCT FROM $2)',t)
   INTO bad USING OLD.assignment_id,NEW.shipment_id;
  IF bad THEN
   RAISE EXCEPTION '구간 변경 불가: %의 기존 연결과 운송 건이 다릅니다 (assignment_id=%)',t,OLD.assignment_id
    USING ERRCODE='23514';
  END IF;
 END LOOP;
 RETURN NEW;
END $$;
DROP TRIGGER IF EXISTS trg_guard_port_leg_change ON shipment_legs;
CREATE TRIGGER trg_guard_port_leg_change BEFORE UPDATE OF shipment_id,leg_type ON shipment_legs
 FOR EACH ROW EXECUTE FUNCTION scargo_guard_port_leg_change();


-- 5차 보완 구간 (이 파일에 통합됨). 데이터 보정은 개발/테스트용입니다.
-- VGM 오차 시나리오는 G-1 재계산에서 제외하고 신고값과 계근값의 차이를 유지합니다.
-- ============================================================
-- 5차 수정 패치 v2 (4차 수정 구간 다음에 실행)
-- 개발·테스트 DB 전용. 전체 파일이 단일 트랜잭션이므로 오류 시 전체 ROLLBACK.
-- 실행 방법은 파일 맨 위 안내를 따르세요.
--
-- v1 대비 달라진 점
--  - 데이터 정리(G)를 제약 추가(B~F)보다 먼저 실행: 정리한 결과가 새 제약에 걸리지 않도록
--  - 새 제약 적용 전 사전 점검: 걸리는 행이 있으면 테이블별 건수를 한 번에 안내
--  - EAN13 컨테이너의 VGM/총중량/축중 재계산 (실제 물품 합계 기준)
--  - G-1로 위반 처리된 검사 행을 참조하는 PASSED 통과 기록을 BLOCKED로 정리
--  - planned_route: origin/destination 게이트가 원래 달랐다면 보존
--  - 자체 테스트가 gate_logs / overload_checks / item_history / item_vehicle_assignments 4개 테이블 대상
--
-- 포함 내용
--  A. 4차 트리거 보완: 항만 구간 shipment_id 자동 채움 + 인덱스 2개
--  G. 더미 데이터 정리 (VGM, 축하중, 통행 기록, planned_route)
--  P. 사전 점검
--  B. 구간당 입고/출고 1건, 도크 점유 중복 방지 (랙 칸은 다중 박스 허용)
--  C. toll_passages 판정 모순 방지
--  D. notifications 수신자 필수, logistics_centers.region_id 범위 제한
--  E. 대댓글 부모가 같은 게시글인지 보장
--  F. items.container_no ↔ shipments.container_no 일치 보장
--  H. 최종 검증 + 트리거 자체 테스트
-- ============================================================

LOCK TABLE toll_passages, overload_checks IN SHARE ROW EXCLUSIVE MODE;
DROP TRIGGER IF EXISTS trg_guard_check_passages ON overload_checks;

-- ============================================================
-- A. 4차 트리거 보완
--    항만→허브 구간에서 shipment_id가 비어 있으면 구간의 값으로 채웁니다.
--    BEFORE ROW 트리거에서 채우므로 NOT NULL 컬럼(item_*)도 생략 가능합니다
--    (NOT NULL 검사는 BEFORE 트리거 이후에 실행됨). 맨 아래 자체 테스트가 4개 테이블 모두 확인합니다.
--    허브→센터 구간은 구간 shipment_id가 NULL이므로 item_* 에는 원본 shipment_id를 직접 넣어야 합니다.
-- ============================================================
CREATE OR REPLACE FUNCTION scargo_check_port_leg_shipment()
RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE leg RECORD;
BEGIN
 IF NEW.assignment_id IS NULL THEN RETURN NEW; END IF;
 SELECT shipment_id,leg_type INTO leg FROM shipment_legs
 WHERE assignment_id=NEW.assignment_id FOR SHARE;
 IF NOT FOUND THEN
  RAISE EXCEPTION '존재하지 않는 운송 구간: %',NEW.assignment_id USING ERRCODE='23503';
 END IF;
 IF leg.leg_type='PORT_TO_HUB' THEN
  IF NEW.shipment_id IS NULL THEN
   NEW.shipment_id := leg.shipment_id;
  ELSIF NEW.shipment_id IS DISTINCT FROM leg.shipment_id THEN
   RAISE EXCEPTION '항만→허브 운송 건 불일치: table=%, assignment_id=%, 입력 shipment_id=%, 구간 shipment_id=%',
    TG_TABLE_NAME,NEW.assignment_id,NEW.shipment_id,leg.shipment_id USING ERRCODE='23514';
  END IF;
 END IF;
 RETURN NEW;
END $$;

CREATE INDEX IF NOT EXISTS idx_gate_logs_assignment    ON gate_logs(assignment_id);
CREATE INDEX IF NOT EXISTS idx_item_history_assignment ON item_history(assignment_id);

-- ============================================================
-- G. 더미 데이터 정리 (제약 추가 전에 먼저 수행)
-- ============================================================

-- G-1. 컨테이너에 실제 실린 물품 합계 기준으로 VGM/총중량/축중 재계산
--      대상: items가 있는 컨테이너의 overload_checks 행
--        new_vgm   = 컨테이너 자체중량(tare) + 물품 중량 합계
--        new_total = 차량 공차 + new_vgm
--        축중/바퀴 = 기존 비율대로 축소, 반올림 오차는 2축에 반영 (바퀴 좌+우 = 축중)
--      제외: 축중량 위반 및 VGM 중량 오차 시나리오. 예외 시나리오의 신고값/계근값 차이를 보존합니다.
CREATE TEMP TABLE _vgm_targets (
    check_id BIGINT PRIMARY KEY, new_total INT NOT NULL, new_vgm INT NOT NULL
) ON COMMIT DROP;

INSERT INTO _vgm_targets (check_id, new_total, new_vgm)
SELECT o.check_id,
       o.empty_weight_kg + ROUND(c.tare_kg + s.cargo)::int,
       ROUND(c.tare_kg + s.cargo)::int
FROM overload_checks o
JOIN containers c ON c.container_no = o.container_no
JOIN (SELECT container_no, SUM(weight_kg) AS cargo
      FROM items WHERE container_no IS NOT NULL GROUP BY container_no) s
  ON s.container_no = o.container_no
WHERE o.empty_weight_kg IS NOT NULL
  AND o.total_weight_kg IS NOT NULL AND o.total_weight_kg > 0
  AND COALESCE(o.violation_reason, '') NOT LIKE '축중량%'
  AND o.violation_reason IS DISTINCT FROM 'VGM 중량 오차 범위 초과'
  AND o.vgm_kg IS DISTINCT FROM ROUND(c.tare_kg + s.cargo)::int
  AND ROUND(c.tare_kg + s.cargo) <= c.max_gross_kg;

UPDATE overload_checks o
SET axle1_weight = ROUND(o.axle1_weight::numeric * t.new_total / o.total_weight_kg)::int,
    axle2_weight = ROUND(o.axle2_weight::numeric * t.new_total / o.total_weight_kg)::int,
    axle3_weight = ROUND(o.axle3_weight::numeric * t.new_total / o.total_weight_kg)::int,
    axle4_weight = ROUND(o.axle4_weight::numeric * t.new_total / o.total_weight_kg)::int,
    axle5_weight = ROUND(o.axle5_weight::numeric * t.new_total / o.total_weight_kg)::int,
    axle6_weight = ROUND(o.axle6_weight::numeric * t.new_total / o.total_weight_kg)::int,
    axle7_weight = ROUND(o.axle7_weight::numeric * t.new_total / o.total_weight_kg)::int,
    axle8_weight = ROUND(o.axle8_weight::numeric * t.new_total / o.total_weight_kg)::int,
    total_weight_kg = t.new_total,
    vgm_kg = t.new_vgm
FROM _vgm_targets t
WHERE o.check_id = t.check_id;

UPDATE overload_checks o
SET axle2_weight = o.axle2_weight + (o.total_weight_kg -
      (o.axle1_weight + o.axle2_weight + o.axle3_weight + o.axle4_weight +
       o.axle5_weight + o.axle6_weight + o.axle7_weight + o.axle8_weight))
FROM _vgm_targets t
WHERE o.check_id = t.check_id;

UPDATE overload_checks o
SET axle1_wheel_left = o.axle1_weight / 2, axle1_wheel_right = o.axle1_weight - o.axle1_weight / 2,
    axle2_wheel_left = o.axle2_weight / 2, axle2_wheel_right = o.axle2_weight - o.axle2_weight / 2,
    axle3_wheel_left = o.axle3_weight / 2, axle3_wheel_right = o.axle3_weight - o.axle3_weight / 2,
    axle4_wheel_left = o.axle4_weight / 2, axle4_wheel_right = o.axle4_weight - o.axle4_weight / 2,
    axle5_wheel_left = o.axle5_weight / 2, axle5_wheel_right = o.axle5_weight - o.axle5_weight / 2,
    axle6_wheel_left = o.axle6_weight / 2, axle6_wheel_right = o.axle6_weight - o.axle6_weight / 2,
    axle7_wheel_left = o.axle7_weight / 2, axle7_wheel_right = o.axle7_weight - o.axle7_weight / 2,
    axle8_wheel_left = o.axle8_weight / 2, axle8_wheel_right = o.axle8_weight - o.axle8_weight / 2
FROM _vgm_targets t
WHERE o.check_id = t.check_id;

-- G-2. 'VGM 중량 오차 범위 초과' 사유 행: 신고 VGM을 계측값(총중량-공차)보다 12% 크게 설정
--      (컨테이너 최대총중량을 넘지 않는 범위에서만). 12%는 일반적인 ±5% 허용치를 넘기 위한 값입니다.
--      실제 적용 기준이 다르면 1.12를 조정하세요.
UPDATE overload_checks o
SET vgm_kg = ROUND((o.total_weight_kg - o.empty_weight_kg) * 1.12, -1)
FROM containers c
WHERE c.container_no = o.container_no
  AND o.violation_reason = 'VGM 중량 오차 범위 초과'
  AND ROUND((o.total_weight_kg - o.empty_weight_kg) * 1.12, -1) <= c.max_gross_kg;

-- G-3. 축하중 10톤(10,000kg) 초과 행은 위반 처리. 변경된 행은 임시 테이블에 기록해 후속 정리에 사용
CREATE TEMP TABLE _g3_changed (check_id BIGINT PRIMARY KEY) ON COMMIT DROP;

WITH u AS (
  UPDATE overload_checks
  SET is_violation = TRUE,
      is_passed = FALSE,
      violation_reason = COALESCE(violation_reason, '축중량 제한 초과(축당 10t)')
  WHERE NOT is_violation
    AND GREATEST(axle1_weight, axle2_weight, axle3_weight, axle4_weight,
                 axle5_weight, axle6_weight, axle7_weight, axle8_weight) > 10000
  RETURNING check_id)
INSERT INTO _g3_changed SELECT check_id FROM u;

-- G-4. G-3으로 위반이 된 검사 행을 참조하는 PASSED 통과 기록 → BLOCKED(OVERLOAD),
--      해당 통행료 결제는 취소/환불 처리
CREATE TEMP TABLE _g4_blocked (passage_id BIGINT PRIMARY KEY) ON COMMIT DROP;

WITH u AS (
  UPDATE toll_passages p
  SET passage_status = 'BLOCKED', block_reason = 'OVERLOAD', passed_at = NULL
  WHERE p.passage_status = 'PASSED'
    AND p.check_id IN (SELECT check_id FROM _g3_changed)
  RETURNING p.passage_id)
INSERT INTO _g4_blocked SELECT passage_id FROM u;

UPDATE toll_payments
SET payment_status = CASE WHEN payment_status = 'PAID' THEN 'REFUNDED' ELSE 'CANCELED' END
WHERE passage_id IN (SELECT passage_id FROM _g4_blocked)
  AND payment_status IN ('PENDING', 'PAID');

-- G-5. trucks.planned_route 스냅샷을 현재 컨테이너/야드 위치에 맞춤
CREATE OR REPLACE FUNCTION pg_temp.container_json(c containers, sector text) RETURNS jsonb
LANGUAGE sql AS $f$
  SELECT jsonb_build_object(
      'container_no', c.container_no,
      'iso_size_type_code', c.iso_size_type_code,
      'container_type', c.container_type,
      'is_high_cube', c.is_high_cube,
      'max_gross_kg', c.max_gross_kg,
      'tare_kg', c.tare_kg,
      'max_payload_kg', c.max_payload_kg,
      'cubic_capacity_cbm', c.cubic_capacity_cbm,
      'csc_approval_no', c.csc_approval_no,
      'reserved_cargo_info', c.reserved_cargo_info,
      'loading_location_id', c.loading_location_id,
      'loading_sector', sector)
$f$;

-- G-5a. 위치가 바뀐 차량: 컨테이너 정보 + 가장 가까운 인천신항 게이트/거리 재계산
--       destination_gate는 원래 origin_gate와 같았던 경우에만 새 게이트로 맞추고, 달랐다면 그대로 둠
UPDATE trucks t
SET planned_route = jsonb_set(jsonb_set(jsonb_set(jsonb_set(t.planned_route,
        '{container}', pg_temp.container_json(c, l.sector)),
        '{origin_gate}', to_jsonb(g.label)),
        '{destination_gate}',
          CASE WHEN t.planned_route->>'destination_gate' IS NOT DISTINCT FROM t.planned_route->>'origin_gate'
               THEN to_jsonb(g.label)
               ELSE t.planned_route->'destination_gate' END),
        '{nearest_gate_distance_m}', to_jsonb(g.dist))
FROM containers c
JOIN loading_locations l ON l.location_id = c.loading_location_id
JOIN LATERAL (
    SELECT gt.gate_code || ' (' || gt.gate_name || ')' AS label,
           ROUND(distance_m(l.latitude, l.longitude, gt.latitude, gt.longitude))::int AS dist
    FROM gates gt
    WHERE gt.terminal_id = (SELECT terminal_id FROM terminals WHERE terminal_code = 'ICNP')
    ORDER BY distance_m(l.latitude, l.longitude, gt.latitude, gt.longitude)
    LIMIT 1
) g ON TRUE
WHERE t.planned_route->'container'->>'container_no' = c.container_no
  AND (t.planned_route->'container'->>'loading_location_id')::bigint IS DISTINCT FROM c.loading_location_id;

-- G-5b. 위치가 같은 차량: 컨테이너 정보(화물, 컬럼 개명 반영)만 갱신 (옛 키 net_kg가 남은 행만)
UPDATE trucks t
SET planned_route = jsonb_set(t.planned_route, '{container}', pg_temp.container_json(c, l.sector))
FROM containers c
JOIN loading_locations l ON l.location_id = c.loading_location_id
WHERE t.planned_route->'container'->>'container_no' = c.container_no
  AND jsonb_exists(t.planned_route->'container', 'net_kg');

-- ============================================================
-- P. 사전 점검: 새 제약/인덱스가 걸리는 기존 행을 한 번에 안내
-- ============================================================
DO $$
DECLARE msg TEXT := ''; n BIGINT;
BEGIN
  SELECT count(*) INTO n FROM (SELECT 1 FROM hub_inbound_receipts GROUP BY assignment_id HAVING count(*) > 1) x;
  IF n > 0 THEN msg := msg || format(E'\n - hub_inbound_receipts: 같은 구간 중복 입고 %s구간', n); END IF;

  SELECT count(*) INTO n FROM (SELECT 1 FROM hub_dispatches GROUP BY assignment_id HAVING count(*) > 1) x;
  IF n > 0 THEN msg := msg || format(E'\n - hub_dispatches: 같은 구간 중복 출고 %s구간', n); END IF;

  SELECT count(*) INTO n FROM (
    SELECT 1 FROM hub_inbound_receipts
    WHERE dock_id IS NOT NULL AND status IN ('ARRIVED', 'UNLOADING')
    GROUP BY dock_id HAVING count(*) > 1) x;
  IF n > 0 THEN msg := msg || format(E'\n - hub_inbound_receipts: 진행 중 접수가 2건 이상인 도크 %s개', n); END IF;

  SELECT count(*) INTO n FROM toll_passages
  WHERE passage_status = 'PASSED' AND NOT (recognition_status = 'SUCCESS' AND vehicle_no IS NOT NULL);
  IF n > 0 THEN msg := msg || format(E'\n - toll_passages: PASSED인데 OCR 실패/미등록 %s건', n); END IF;

  SELECT count(*) INTO n FROM toll_passages
  WHERE (block_reason = 'OCR_FAILED'   AND recognition_status <> 'FAILED')
     OR (block_reason = 'UNREGISTERED' AND vehicle_no IS NOT NULL)
     OR (block_reason = 'OVERLOAD'     AND check_id IS NULL);
  IF n > 0 THEN msg := msg || format(E'\n - toll_passages: 차단 사유와 실제 상태가 다른 행 %s건', n); END IF;

  SELECT count(*) INTO n FROM notifications WHERE account_id IS NULL AND company_id IS NULL;
  IF n > 0 THEN msg := msg || format(E'\n - notifications: 수신자 없는 알림 %s건', n); END IF;

  SELECT count(*) INTO n FROM logistics_centers
  WHERE region_id IS NOT NULL AND center_type <> 'DISTRIBUTION_CENTER';
  IF n > 0 THEN msg := msg || format(E'\n - logistics_centers: 물류센터가 아닌데 region_id가 있는 거점 %s건', n); END IF;

  SELECT count(*) INTO n FROM post_comments c JOIN post_comments p ON p.comment_id = c.parent_id
  WHERE p.post_id <> c.post_id;
  IF n > 0 THEN msg := msg || format(E'\n - post_comments: 다른 게시글의 댓글을 부모로 가진 대댓글 %s건', n); END IF;

  SELECT count(*) INTO n FROM items i JOIN shipments s ON s.shipment_id = i.shipment_id
  WHERE i.container_no IS NOT NULL AND i.container_no IS DISTINCT FROM s.container_no;
  IF n > 0 THEN msg := msg || format(E'\n - items: shipments.container_no와 다른 container_no %s건', n); END IF;

  IF msg <> '' THEN
    RAISE EXCEPTION E'새 제약을 적용할 수 없는 기존 데이터가 있습니다. 정리 후 다시 실행하세요.%', msg;
  END IF;
END $$;

-- H. 최종 검증
-- ============================================================
DO $$
DECLARE n BIGINT;
BEGIN
  -- G-1 대상 행: 축중 합계 = 총중량 = 공차 + VGM
  IF EXISTS (SELECT 1 FROM overload_checks o JOIN _vgm_targets t USING (check_id)
             WHERE o.total_weight_kg <> o.empty_weight_kg + o.vgm_kg
                OR o.total_weight_kg <> o.axle1_weight + o.axle2_weight + o.axle3_weight + o.axle4_weight
                                      + o.axle5_weight + o.axle6_weight + o.axle7_weight + o.axle8_weight) THEN
    RAISE EXCEPTION 'VGM 재계산 대상 행의 총중량/축중 합계가 어긋납니다';
  END IF;

  -- G-2: 사유 행의 신고 VGM 오차가 5% 초과인지
  SELECT count(*) INTO n FROM overload_checks
  WHERE violation_reason = 'VGM 중량 오차 범위 초과'
    AND (total_weight_kg - empty_weight_kg) > 0
    AND ABS(vgm_kg - (total_weight_kg - empty_weight_kg))::numeric / (total_weight_kg - empty_weight_kg) <= 0.05;
  IF n > 0 THEN
    RAISE WARNING 'VGM 오차 사유 행 %건의 실제 오차가 5%% 이하입니다 (컨테이너 최대총중량 제한으로 G-2가 건너뛰었을 수 있음)', n;
  END IF;

  -- G-3
  IF EXISTS (SELECT 1 FROM overload_checks WHERE NOT is_violation
             AND GREATEST(axle1_weight, axle2_weight, axle3_weight, axle4_weight,
                          axle5_weight, axle6_weight, axle7_weight, axle8_weight) > 10000) THEN
    RAISE EXCEPTION '축하중 10t 초과인데 위반이 아닌 행이 남아 있습니다';
  END IF;

  -- G-4: 위반 검사를 참조하는 PASSED 통과 기록 (정리 대상 외 데이터는 경고만)
  SELECT count(*) INTO n FROM toll_passages p JOIN overload_checks o ON o.check_id = p.check_id
  WHERE p.passage_status = 'PASSED' AND o.is_violation;
  IF n > 0 THEN
    RAISE WARNING '위반 판정된 검사를 참조하는 PASSED 통과 기록이 %건 있습니다', n;
  END IF;

  -- G-5
  IF EXISTS (SELECT 1 FROM trucks t JOIN containers c
               ON c.container_no = t.planned_route->'container'->>'container_no'
             WHERE (t.planned_route->'container'->>'loading_location_id')::bigint
                   IS DISTINCT FROM c.loading_location_id) THEN
    RAISE EXCEPTION 'trucks.planned_route의 컨테이너 위치가 containers와 다릅니다';
  END IF;
  IF EXISTS (SELECT 1 FROM trucks WHERE jsonb_exists(planned_route->'container', 'net_kg')) THEN
    RAISE EXCEPTION 'planned_route에 옛 키(net_kg)가 남아 있습니다';
  END IF;

  -- 축중량/VGM 오차 시나리오를 포함해 물품 합계와 VGM이 다른 검사 행 안내
  SELECT count(*) INTO n
  FROM overload_checks o
  JOIN containers c ON c.container_no = o.container_no
  JOIN (SELECT container_no, SUM(weight_kg) AS cargo FROM items
        WHERE container_no IS NOT NULL GROUP BY container_no) s ON s.container_no = o.container_no
  WHERE o.vgm_kg IS DISTINCT FROM ROUND(c.tare_kg + s.cargo)::int;
  IF n > 0 THEN
    RAISE WARNING '물품 합계와 VGM이 다른 검사 행이 %건 남아 있습니다 (축중량 및 VGM 오차 시나리오 행은 의도적으로 제외)', n;
  END IF;
END $$;

-- 트리거 자체 테스트: 항만 구간에 shipment_id 없이 기록을 넣으면 구간 값으로 채워지는지 4개 테이블 확인 후 취소
DO $$
DECLARE leg RECORD; got BIGINT; itm BIGINT;
BEGIN
  SELECT assignment_id, shipment_id, vehicle_no, from_center_id INTO leg
  FROM shipment_legs WHERE leg_type = 'PORT_TO_HUB' ORDER BY assignment_id LIMIT 1;
  IF NOT FOUND THEN
    RAISE NOTICE '항만 구간이 없어 자체 테스트를 건너뜁니다';
    RETURN;
  END IF;

  -- gate_logs
  BEGIN
    INSERT INTO gate_logs (gate_id, assignment_id)
    SELECT gate_id, leg.assignment_id FROM gates ORDER BY gate_id LIMIT 1
    RETURNING shipment_id INTO got;
    IF got IS DISTINCT FROM leg.shipment_id THEN
      RAISE EXCEPTION 'gate_logs 자동 채움 실패 (기대 %, 실제 %)', leg.shipment_id, got;
    END IF;
    RAISE EXCEPTION USING ERRCODE = 'P0001', MESSAGE = '__selftest_rollback__';
  EXCEPTION WHEN SQLSTATE 'P0001' THEN
    IF SQLERRM <> '__selftest_rollback__' THEN RAISE; END IF;
  END;

  -- overload_checks
  BEGIN
    INSERT INTO overload_checks (vehicle_no, is_violation, is_passed, assignment_id)
    VALUES (leg.vehicle_no, FALSE, TRUE, leg.assignment_id)
    RETURNING shipment_id INTO got;
    IF got IS DISTINCT FROM leg.shipment_id THEN
      RAISE EXCEPTION 'overload_checks 자동 채움 실패 (기대 %, 실제 %)', leg.shipment_id, got;
    END IF;
    RAISE EXCEPTION USING ERRCODE = 'P0001', MESSAGE = '__selftest_rollback__';
  EXCEPTION WHEN SQLSTATE 'P0001' THEN
    IF SQLERRM <> '__selftest_rollback__' THEN RAISE; END IF;
  END;

  -- item_history (shipment_id NOT NULL 컬럼이지만 BEFORE 트리거가 먼저 채움)
  BEGIN
    INSERT INTO item_history (center_id, assignment_id, handling_type)
    VALUES (leg.from_center_id, leg.assignment_id, 'UNLOAD')
    RETURNING shipment_id INTO got;
    IF got IS DISTINCT FROM leg.shipment_id THEN
      RAISE EXCEPTION 'item_history 자동 채움 실패 (기대 %, 실제 %)', leg.shipment_id, got;
    END IF;
    RAISE EXCEPTION USING ERRCODE = 'P0001', MESSAGE = '__selftest_rollback__';
  EXCEPTION WHEN SQLSTATE 'P0001' THEN
    IF SQLERRM <> '__selftest_rollback__' THEN RAISE; END IF;
  END;

  -- item_vehicle_assignments (해당 운송 건의 물품이 있을 때만)
  SELECT item_id INTO itm FROM items WHERE shipment_id = leg.shipment_id ORDER BY item_id LIMIT 1;
  IF itm IS NULL THEN
    RAISE NOTICE '해당 구간의 운송 건에 물품이 없어 item_vehicle_assignments 테스트는 건너뜁니다';
  ELSE
    BEGIN
      INSERT INTO item_vehicle_assignments (item_id, assignment_id)
      VALUES (itm, leg.assignment_id)
      RETURNING shipment_id INTO got;
      IF got IS DISTINCT FROM leg.shipment_id THEN
        RAISE EXCEPTION 'item_vehicle_assignments 자동 채움 실패 (기대 %, 실제 %)', leg.shipment_id, got;
      END IF;
      RAISE EXCEPTION USING ERRCODE = 'P0001', MESSAGE = '__selftest_rollback__';
    EXCEPTION WHEN SQLSTATE 'P0001' THEN
      IF SQLERRM <> '__selftest_rollback__' THEN RAISE; END IF;
    END;
  END IF;
END $$;

-- 5차 보완: 통과 이벤트와 검사 차량/구간/결과 일치 검증.
-- 검사 없는 정상 통과는 허용합니다. 검사가 연결되면 해당 검사 결과를 확인합니다.
CREATE OR REPLACE FUNCTION scargo_validate_passage_check()
RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE c RECORD;
BEGIN
 IF NEW.check_id IS NULL THEN RETURN NEW; END IF;
 SELECT vehicle_no,assignment_id,is_violation,is_passed INTO c
 FROM overload_checks WHERE check_id=NEW.check_id FOR SHARE;
 IF NOT FOUND THEN RAISE EXCEPTION '존재하지 않는 검사: %',NEW.check_id USING ERRCODE='23503'; END IF;
 IF NEW.vehicle_no IS DISTINCT FROM c.vehicle_no OR NEW.assignment_id IS DISTINCT FROM c.assignment_id THEN
  RAISE EXCEPTION '통과 기록과 검사 차량/운송 구간 불일치' USING ERRCODE='23514';
 END IF;
 IF NEW.passage_status='PASSED' AND (c.is_violation OR NOT c.is_passed) THEN
  RAISE EXCEPTION '위반 또는 불합격 검사로 통과 처리할 수 없습니다' USING ERRCODE='23514';
 END IF;
 IF NEW.block_reason='OVERLOAD' AND NOT c.is_violation THEN
  RAISE EXCEPTION '과적 차단은 위반 검사 결과가 필요합니다' USING ERRCODE='23514';
 END IF;
 RETURN NEW;
END $$;
DROP TRIGGER IF EXISTS trg_passage_check_consistency ON toll_passages;
CREATE TRIGGER trg_passage_check_consistency
 BEFORE INSERT OR UPDATE OF check_id,vehicle_no,assignment_id,passage_status,block_reason ON toll_passages
 FOR EACH ROW EXECUTE FUNCTION scargo_validate_passage_check();

-- 검사 결과를 나중에 변경해 기존 통과 기록과 모순되는 경우도 차단합니다.
-- 재검사는 새 검사 행으로 기록하는 것을 권장합니다.
CREATE OR REPLACE FUNCTION scargo_guard_check_passages()
RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
 IF EXISTS (SELECT 1 FROM toll_passages p WHERE p.check_id=OLD.check_id AND
    (p.vehicle_no IS DISTINCT FROM NEW.vehicle_no
     OR p.assignment_id IS DISTINCT FROM NEW.assignment_id
     OR (p.passage_status='PASSED' AND (NEW.is_violation OR NOT NEW.is_passed))
     OR (p.block_reason='OVERLOAD' AND NOT NEW.is_violation))) THEN
  RAISE EXCEPTION '검사 변경이 기존 통과 기록과 모순됩니다: %',OLD.check_id USING ERRCODE='23514';
 END IF;
 RETURN NEW;
END $$;
DROP TRIGGER IF EXISTS trg_guard_check_passages ON overload_checks;
CREATE TRIGGER trg_guard_check_passages
 BEFORE UPDATE OF vehicle_no,assignment_id,is_violation,is_passed ON overload_checks
 FOR EACH ROW EXECUTE FUNCTION scargo_guard_check_passages();

-- 새 트리거 설치 이전에 존재하던 기록도 검증합니다.
DO $$ BEGIN
 IF EXISTS (SELECT 1 FROM toll_passages p JOIN overload_checks c ON c.check_id=p.check_id
 WHERE p.vehicle_no IS DISTINCT FROM c.vehicle_no OR p.assignment_id IS DISTINCT FROM c.assignment_id
 OR (p.passage_status='PASSED' AND (c.is_violation OR NOT c.is_passed))
 OR (p.block_reason='OVERLOAD' AND NOT c.is_violation)) THEN
  RAISE EXCEPTION '기존 통과 기록과 검사 차량/구간/결과가 모순됩니다. 원본 확인 후 정리하세요.';
 END IF;
END $$;


-- 확인 조회: 의도적 축중량/VGM 오차 시나리오는 물품 합계와 다른 신고 VGM을 유지할 수 있습니다.
SELECT o.check_id, o.vehicle_no, o.container_no, o.violation_reason,
       o.vgm_kg, ROUND(c.tare_kg + s.cargo)::int AS expected_vgm
FROM overload_checks o
JOIN containers c ON c.container_no = o.container_no
JOIN (SELECT container_no, SUM(weight_kg) AS cargo FROM items
      WHERE container_no IS NOT NULL GROUP BY container_no) s ON s.container_no = o.container_no
WHERE o.vgm_kg IS DISTINCT FROM ROUND(c.tare_kg + s.cargo)::int;

-- 6차: 결제 연결 통과 기록 변경/삭제 방지

LOCK TABLE toll_passages, toll_payments IN SHARE ROW EXCLUSIVE MODE;

CREATE OR REPLACE FUNCTION scargo_freeze_paid_passage()
RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
 IF TG_OP='UPDATE' AND NEW IS NOT DISTINCT FROM OLD THEN RETURN NEW; END IF;
 IF EXISTS (SELECT 1 FROM toll_payments WHERE passage_id=OLD.passage_id OR entry_passage_id=OLD.passage_id) THEN
  RAISE EXCEPTION '결제 기록에 연결된 통과 기록은 수정/삭제할 수 없습니다 (passage_id=%)',OLD.passage_id
   USING ERRCODE='23514';
 END IF;
 IF TG_OP='DELETE' THEN RETURN OLD; END IF;
 RETURN NEW;
END $$;
DROP TRIGGER IF EXISTS trg_freeze_paid_passage ON toll_passages;
CREATE TRIGGER trg_freeze_paid_passage BEFORE UPDATE OR DELETE ON toll_passages
 FOR EACH ROW EXECUTE FUNCTION scargo_freeze_paid_passage();

-- 결제 등록과 통과 수정이 동시에 진행될 때도 같은 통과 행을 잠급니다.
-- 결제 링크를 제거/변경하여 통과 기록 잠금을 우회하는 것은 허용하지 않습니다.
-- payment_status 등 결제 처리 정보 변경은 기존 규칙에 따라 허용합니다.
CREATE OR REPLACE FUNCTION scargo_lock_payment_passages()
RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
 IF TG_OP='DELETE' THEN
  RAISE EXCEPTION '결제 이력은 삭제하지 않고 취소/환불 상태로 보존하세요' USING ERRCODE='23514';
 END IF;
 IF TG_OP='UPDATE' AND
   (NEW.passage_id IS DISTINCT FROM OLD.passage_id OR NEW.entry_passage_id IS DISTINCT FROM OLD.entry_passage_id) THEN
  RAISE EXCEPTION '결제에 연결된 진입/진출 통과 기록은 변경할 수 없습니다' USING ERRCODE='23514';
 END IF;
 PERFORM passage_id FROM toll_passages
 WHERE passage_id IN (NEW.passage_id,NEW.entry_passage_id)
 ORDER BY passage_id FOR UPDATE;
 RETURN NEW;
END $$;
DROP TRIGGER IF EXISTS trg_lock_payment_passages ON toll_payments;
CREATE TRIGGER trg_lock_payment_passages BEFORE INSERT OR UPDATE OR DELETE ON toll_payments
 FOR EACH ROW EXECUTE FUNCTION scargo_lock_payment_passages();
